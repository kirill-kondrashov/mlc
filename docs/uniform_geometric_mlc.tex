\section*{Обозначения}

Положим
\[
 \N:=\{0,1,2,\ldots\},\qquad f_c(z):=z^2+c,\qquad
 p_0(c):=0,\qquad p_{n+1}(c):=p_n(c)^2+c,
\]
\[
 \M:=\{c\in\C:\sup_{n\in\N}|p_n(c)|<\infty\},\qquad
 K:=\overline B(0,2),\qquad
 \D:=\{x+iy:x,y\in[-2,2]\},
\]
\[
 O_N:=K\cap\bigcap_{j=0}^{N}p_j^{-1}(K),\qquad
 \Pi(z):=\frac{2z}{\max\{2,|z|\}}.
\]
Для непустого \(A\subseteq\C\) и непустых компактных \(A,B\subseteq\C\)
положим
\[
 \dist(z,A):=\inf_{a\in A}|z-a|,\qquad
 \dH(A,B):=
 \max\left\{\sup_{a\in A}\dist(a,B),\
             \sup_{b\in B}\dist(b,A)\right\}.
\]

\section*{Основная гипотеза и её связь с факторизацией}

Пусть \(\mathcal K(\M)\) — множество непустых компактных подмножеств
\(\M\). Для \(C\in\mathcal K(\M)\) положим
\[
 \operatorname{diam}C:=\sup\{|u-v|:u,v\in C\}.
\]
Для \(x,y\in\M\) положим
\[
 \mathcal C_\M(x,y):=
 \{C\in\mathcal K(\M):C\text{ связно},\ x,y\in C\}.
\]
При \(r>0\) положим
\[
 \omega_\M(r):=
  \sup_{\substack{x,y\in\M\\|x-y|<r}}
  \inf_{C\in\mathcal C_\M(x,y)}\operatorname{diam}C
  \in[0,+\infty],\qquad \inf\varnothing:=+\infty.
 \]
 Для \(m\ge1\) положим
 \[
  \mathcal R_m:=
 \left\{G\in C(\D,K):
 \begin{array}{l}
 \exists N\in\N,\ N\ge m,\quad G|_{O_N}=\id_{O_N},\\[1mm]
 \displaystyle\sup_{z\in\D}\dist(G(z),O_N)\le2^{-m}
 \end{array}
 \right\}.
\]
\[
 \mathsf{Fin}:\Leftrightarrow
 \forall m\ge1,\quad\mathcal R_m\ne\varnothing.
 \]
\[
  \begin{gathered}
  \mathsf{EA}_{\mathrm{seq}}\Leftrightarrow
  \exists(m_j)_{j\ge1}\ \exists(G_j)_{j\ge1},\\
  1\le m_1<m_2<\cdots,\qquad m_j\longrightarrow\infty,\qquad
  G_j\in\mathcal R_{m_j}.
  \end{gathered}
 \]
 \[
   \begin{gathered}
   \mathsf{EAdata}\bigl(\rho,(m_j),(G_j)\bigr)\\
  \Leftrightarrow\\
  \begin{gathered}
  1\le m_1<m_2<\cdots,\qquad m_j\longrightarrow\infty,\qquad
  G_j\in\mathcal R_{m_j},\\
 \rho:(0,\infty)\to(0,\infty),\\
  \forall\eta>0,\qquad 0<\rho(\eta)\le\eta,\\
  \forall\eta>0\ \forall j\ge1\ \forall z,w\in\D,\qquad
  |z-w|\le\rho(\eta)\Rightarrow
  |G_j(z)-G_j(w)|\le\eta.
  \end{gathered}
 \end{gathered}
 \]
\[
 \begin{gathered}
 \mathsf{EA}\Leftrightarrow
 \exists\rho\ \exists(m_j)_{j\ge1}\ \exists(G_j)_{j\ge1},\\
 \mathsf{EAdata}\bigl(\rho,(m_j),(G_j)\bigr).
 \end{gathered}
 \]

Для \(F\subseteq\C\) обозначим через \(\pi_0(F)\) множество его
компонент связности. Если \(c\in F\), через
\(\operatorname{Comp}_c(F)\) обозначим компоненту, содержащую \(c\).
При \(c\in\M\), \(r>0\) и \(N\in\N\) положим
\[
 C_N(c,r):=
 \operatorname{Comp}_c\bigl(O_N\cap\overline B(c,r)\bigr).
\]
Для \(L\ge N\) и \(0<\delta<r\) включение
\[
 O_L\cap\overline B(c,\delta)
 \subseteq O_N\cap\overline B(c,r)
\]
индуцирует отображение множеств компонент
\[
 \varphi_{L,N}^{c;\delta,r}:
 \pi_0\bigl(O_L\cap\overline B(c,\delta)\bigr)
 \longrightarrow
 \pi_0\bigl(O_N\cap\overline B(c,r)\bigr),
\]
которое каждую компоненту меньшего множества переводит в единственную
компоненту большего множества, содержащую её.

\begin{definition}[Равномерный контроль компонент]
\label{def:uniform-component-control}
Условие \(\mathsf{UC}\) означает, что
\begin{equation}
 \begin{gathered}
 \forall c\in\M\ \forall\varepsilon>0\
 \exists r,\delta\in\R,\quad 0<\delta<r<\varepsilon,\qquad\\
 \forall N\in\N\ \exists L\in\N,\quad N\le L,\qquad
 \forall A\in\pi_0\bigl(O_L\cap\overline B(c,\delta)\bigr),\\
 \varphi_{L,N}^{c;\delta,r}(A)=C_N(c,r).
 \end{gathered}
 \label{eq:uniform-component-control}
\end{equation}
Радиусы \(r,\delta\) выбираются по \(c,\varepsilon\) и общие для всех
\(N\); индекс \(L\) может зависеть от \(c,\varepsilon,N\).
Для фиксированных \(c,\varepsilon,N,L\) равенство
\(\varphi_{L,N}^{c;\delta,r}(A)=C_N(c,r)\) для всех \(A\) равносильно
включению
\[
 O_L\cap\overline B(c,\delta)\subseteq C_N(c,r).
\]
\end{definition}

 \begin{proposition}[Диагональный критерий выбора с конечным числом оценок]
 \label{prop:diagonal-finite-scale}
 Условие \(\mathsf{EA}\) равносильно следующему условию: существуют
 положительные числа \((\delta_i)_{i\ge1}\), для которых
 \begin{equation}
  \begin{gathered}
  \forall k,M\in\N,\ k,M\ge1\ \exists m\in\N,\ m\ge M\
  \exists G\in\mathcal R_m,\quad\\
  \forall i\in\{1,\ldots,k\}\ \forall z,w\in\D,\qquad
  |z-w|\le\delta_i\Rightarrow
  |G(z)-G(w)|\le2^{-i}.
  \end{gathered}
  \label{eq:diagonal-finite-scale}
 \end{equation}
 \end{proposition}
 \begin{proof}
 Пусть выполнено \(\mathsf{EAdata}(\rho,(m_j),(G_j))\). Положим
 \(\delta_i:=\rho(2^{-i})\). Для любых \(k,M\ge1\) выберем \(j\ge k\)
 так, чтобы \(m_j\ge M\). Тогда \(G_j\in\mathcal R_{m_j}\), а оценка
 равностепенной непрерывности при \(\eta=2^{-i}\) даёт все неравенства
 в \eqref{eq:diagonal-finite-scale} для \(1\le i\le k\).

 Обратно, пусть существуют \((\delta_i)\), удовлетворяющие
 \eqref{eq:diagonal-finite-scale}. Положим \(m_0:=0\). Последовательно
 применяя это условие с \(k=j\) и \(M=m_{j-1}+1\), выберем
 \(m_j\ge m_{j-1}+1\) и \(G_j\in\mathcal R_{m_j}\), для которых
 \[
  \forall i\in\N,\ 1\le i\le j\ \forall z,w\in\D,\qquad
  |z-w|\le\delta_i\Rightarrow
  |G_j(z)-G_j(w)|\le2^{-i}.
 \]
 Тогда \(m_1<m_2<\cdots\), и \(m_j\to\infty\). Зафиксируем
 \(\eta>0\) и выберем \(i\ge1\) так, чтобы \(2^{-i}\le\eta\).
 Равномерная непрерывность конечного семейства
 \(G_1,\ldots,G_{i-1}\) даёт \(\sigma_\eta>0\), для которого
 \[
  |z-w|\le\sigma_\eta\Rightarrow
  |G_j(z)-G_j(w)|\le\eta
  \qquad(1\le j<i,\ z,w\in\D);
 \]
 при \(i=1\) положим \(\sigma_\eta:=1\). Положим
 \(\rho(\eta):=\min\{\eta,\delta_i,\sigma_\eta\}\).
 Если \(j\ge i\), требуемая оценка следует из неравенства с индексом
 \(i\); если \(j<i\), она следует из выбора \(\sigma_\eta\).
 Следовательно, \(\mathsf{EAdata}(\rho,(m_j),(G_j))\), то есть
 \(\mathsf{EA}\).
 \end{proof}
 \begin{equation}
  \forall\rho,(m_j),(G_j),\qquad
   \mathsf{EAdata}\bigl(\rho,(m_j),(G_j)\bigr)
 \Rightarrow
  \forall\eta>0,\qquad
  \omega_\M\!\left(\frac{\rho(\eta)}2\right)\le2\eta.
  \label{eq:quantitative-diameter-bound}
\end{equation}
\begin{equation}
  \begin{gathered}
  \mathsf{Fin},\qquad \mathsf{EA}_{\mathrm{seq}},\\
  \mathsf{EA}\quad\Leftrightarrow\quad
  \lim_{r\downarrow0}\omega_\M(r)=0
  \quad\Leftrightarrow\quad\LC(\M).
  \end{gathered}
 \label{eq:main-discrepancy}
\end{equation}

Теорема об аппроксимации из приложения~\ref{app:approximation-theorem}
задаёт требуемую в \(\mathsf{EA}_{\mathrm{seq}}\) строго возрастающую
последовательность: \(m_j=j\), \(G_j=T_j\in\mathcal R_j\). Её масштабы
используются в шагах~8 и~19. Условие \(\mathsf{EA}\) усиливает эту
конструкцию требованием общего модуля равностепенной непрерывности
\(\rho\) для выбранных отображений. По предложению
\ref{prop:diagonal-finite-scale} диагональный путь требует для каждого
конечного \(k\) выполнения первых \(k\) оценок осцилляции при сколь
угодно больших \(m\). Для завершения этого маршрута требуется доказать
равномерную факторизацию компонент \(\mathsf{UC}\), заданную в теореме
\ref{thm:remaining-component-factorization}. Теорема
\ref{thm:uniform-component-control} устанавливает эквивалентность
\(\mathsf{UC}\Longleftrightarrow\LC(\M)\), откуда также следует
\(\mathsf{EA}\). Следовательно, доказательство факторизации в данной
форме равносильно доказательству локальной связности \(\M\). Формулировка
цели вынесена в отдельный раздел, а черновой условный маршрут — в
следующий раздел «Черновое доказательство теоремы о факторизации»; в
шаге~2 утверждение сведено к включению
\eqref{eq:factorization-inclusion-form}.

\section*{Теорема о факторизации локальных компонент}
\label{sec:remaining-factorization}

В этом разделе \(\N=\{0,1,2,\ldots\}\), а
\(\overline B(a,\rho):=\{z\in\C:|z-a|\le\rho\}\) обозначает замкнутый
диск. Положим
\[
 K:=\overline B(0,2),\qquad f_c(z):=z^2+c,\qquad
 p_0(c):=0,\qquad p_{j+1}(c):=p_j(c)^2+c,
\]
\[
 \M:=\{c\in\C:\sup_{j\in\N}|p_j(c)|<\infty\},\qquad
 O_N:=\{c\in K:p_j(c)\in K\text{ для }0\le j\le N\}.
\]
Это семейство множеств соответствует радиусу выхода \(2\):
если \(|c|\le2\) и \(|z|>2\), то
\(|f_c(z)|\ge|z|^2-2>|z|\); если \(|c|>2\), то начиная с
\(|p_1(c)|=|c|\) имеем
\(|p_{j+1}(c)|\ge|p_j(c)|^2-|c|\ge|p_j(c)|^2-|p_j(c)|>|p_j(c)|\).
Поэтому \(\M=\bigcap_{N\in\N}O_N\). Из определения также следует
вложенность \(O_L\subseteq O_N\) при \(L\ge N\).
Подмножества \(\C\) рассматриваются с евклидовой подпространственной
топологией. Для \(Y\subseteq\C\) через \(\pi_0(Y)\) обозначим множество
его связных компонент, то есть максимальных связных подмножеств \(Y\).
При \(c\in Y\) через \(\operatorname{Comp}_c(Y)\) обозначим компоненту,
содержащую \(c\).
Для \(c\in\M\), \(r>0\) и \(N\in\N\) положим
\[
 F_N(c,r):=O_N\cap\overline B(c,r),\qquad
 C_N(c,r):=\operatorname{Comp}_c(F_N(c,r)).
\]
Иными словами, \(C_N(c,r)\) — связная компонента внешнего среза
\(F_N(c,r)\), содержащая выделенную точку \(c\).
Так как \(c\in\M=\bigcap_NO_N\), имеем \(c\in F_N(c,r)\), поэтому
\(C_N(c,r)\) определена.
Если \(L\ge N\) и \(0<\delta<r\), то
\[
 X_L(c,\delta):=O_L\cap\overline B(c,\delta)
 \subseteq F_N(c,r).
\]
\noindent
Будем называть \(F_N(c,r)\) локальным срезом внешнего приближения
\(O_N\), а \(X_L(c,\delta)\) — вложенным срезом глубины \(L\).
Обозначим через \(\iota_{L,N}^{c;\delta,r}:X_L(c,\delta)\hookrightarrow
F_N(c,r)\) это включение. Оно задаёт отображение компонент
\[
 \varphi_{L,N}^{c;\delta,r}:=
 \pi_0\!\left(\iota_{L,N}^{c;\delta,r}\right):
 \pi_0\bigl(X_L(c,\delta)\bigr)\longrightarrow
 \pi_0\bigl(F_N(c,r)\bigr),
\]
которое каждой компоненте вложенного среза ставит в соответствие
единственную компоненту среза \(F_N(c,r)\), содержащую её.

\begin{theorem}[Теорема о факторизации локальных компонент]
\label{thm:remaining-component-factorization}
Для введённой системы внешних множеств \((O_N)\) выполнено условие
\(\mathsf{UC}\), то есть
\begin{equation}
 \begin{gathered}
 \forall c\in\M\ \forall\varepsilon>0\
 \exists r,\delta\in\R,\quad 0<\delta<r<\varepsilon,\qquad\\
 \forall N\in\N\ \exists L\in\N,\quad N\le L,\qquad
 \forall A\in\pi_0\bigl(O_L\cap\overline B(c,\delta)\bigr),\\
 \varphi_{L,N}^{c;\delta,r}(A)=C_N(c,r).
 \end{gathered}
 \label{eq:factorization-target}
\end{equation}
\end{theorem}

\noindent\emph{Смысл формулировки.}
Индекс \(N\) задаёт последний проверяемый член орбиты. При переходе
от \(N\) к \(N+1\) добавляется условие \(p_{N+1}(c)\in K\), поэтому
\(O_{N+1}=O_N\cap p_{N+1}^{-1}(K)\subseteq O_N\). С ростом \(N\)
множества \(O_N\) уточняют конечные внешние приближения \(\M\).

Для каждого \(c\in\M\) и \(\varepsilon>0\) выбираются общие для всех
\(N\) радиусы \(0<\delta<r<\varepsilon\), а для каждого \(N\) —
индекс \(L\ge N\). Обозначим
\[
 X_L:=X_L(c,\delta),\qquad F_N:=F_N(c,r).
\]
Вложенный срез включён в срез \(F_N\), поэтому каждая его компонента
\(A\in\pi_0(X_L)\) содержится в единственной компоненте \(F_N\).
Теорема утверждает, что при подходящем \(L\) эта компонента для каждого
\(A\) равна \(C_N(c,r)\), то есть \(X_L\subseteq C_N(c,r)\).

Срезы на рисунках представлены сеточными выборками.\footnote{Сеточная
выборка \(E\subseteq\C\) —
растр конечной прямоугольной сетки параметров \(c=x+iy\), в котором
пиксель окрашен тогда и только тогда, когда соответствующий узел
принадлежит \(E\).}
Для конкретного примера выбрана граничная точка \(c_0=\frac14\) —
острие главной кардиоиды \(\M\). Сколь угодно близко к \(c_0\) есть
вещественные параметры вне \(\M\), поэтому здесь особенно важно
проследить, как конечные внешние приближения уточняют локальную
геометрию у границы.

\noindent\emph{Геометрический смысл теоремы.}
Рис.~\ref{fig:factorization-global} и~\ref{fig:factorization-target}
показывают срезы \(X_{20}\subseteq F_2\) у граничной точки
\(c_0=\frac14\) и выбранную компоненту \(A\in\pi_0(X_{20})\), которая
содержится в \(C_2(c_0,r)\). Теорема
распространяет это свойство на каждую компоненту \(X_L\) при достаточной
глубине \(L\):
\[
 X_L(c,\delta)\subseteq C_N(c,r),\qquad
 \varphi_{L,N}^{c;\delta,r}(A)=C_N(c,r)
 \quad\text{для каждого }A\in\pi_0(X_L(c,\delta)).
\]
У среза \(X_L\) могут быть разные компоненты, и все они переходят в
одну компоненту \(C_N(c,r)\). Поэтому
\[
 \operatorname{im}\varphi_{L,N}^{c;\delta,r}
 =\{C_N(c,r)\}.
\]
На множестве компонент \(\pi_0(X_L(c,\delta))\) зададим отношение
эквивалентности
\[
A\sim B
\quad\Longleftrightarrow\quad
\varphi_{L,N}^{c;\delta,r}(A)=\varphi_{L,N}^{c;\delta,r}(B).
\]
Фактор-множество \(\pi_0(X_L(c,\delta))/\sim\) состоит из классов
компонент с одинаковым образом. Множество \(X_L(c,\delta)\) содержит
\(c\), поэтому у него есть компонента \(A\). По теореме все компоненты
эквивалентны, и фактор-множество состоит из одного класса:
\[
\pi_0(X_L(c,\delta))/\sim=\{[A]\}.
\]
Элементами фактор-множества служат классы связных компонент \(X_L\).
На правой части рис.~\ref{fig:factorization-component-geometry}
области \(A_0'\) и \(A_1'\) остаются раздельными; соотношение
\(A_0'\sim A_1'\) означает, что их образы совпадают и обе компоненты
входят в один класс \([A]\).
\[
\begin{aligned}
 q &: \pi_0(X_L(c,\delta))\longrightarrow
      \pi_0(X_L(c,\delta))/\sim,
      &q(A)&=[A],\\
 \bar\varphi_{L,N}^{c;\delta,r}&:
      \pi_0(X_L(c,\delta))/\sim\longrightarrow\pi_0(F_N(c,r)),
      &\bar\varphi_{L,N}^{c;\delta,r}([A])&=C_N(c,r),\\
 \varphi_{L,N}^{c;\delta,r}&=\bar\varphi_{L,N}^{c;\delta,r}\circ q.
\end{aligned}
\]
Таким образом, одноэлементный объект здесь — фактор-множество по
\(\sim\); его единственный класс соответствует единственному значению
\(\{C_N(c,r)\}\) отображения компонент.
\(\mathcal R_N:=\pi_0(F_N(c,r))\setminus\{C_N(c,r)\}\) обозначает
остальные компоненты \(F_N\).
Для изображения ниже обозначим через \(D\in\mathcal R_N\) одну такую
компоненту.
Для наглядности рис.~\ref{fig:factorization-component-geometry}
сравнивает промежуточную глубину \(\ell\) и достаточную глубину \(L\),
где \(N\le\ell<L\). В изображённом случае срез \(X_\ell\) имеет
компоненты \(A_0,A_1\subseteq C_N(c,r)\) и \(B\subseteq D\), где
\(D\in\mathcal R_N\); поэтому
\(\operatorname{im}\varphi_{\ell,N}^{c;\delta,r}=\{C_N(c,r),D\}\).
Глубина \(L\) задаёт срез \(X_L\subseteq X_\ell\), в котором остаются
компоненты \(A_0',A_1'\) внутри \(C_N(c,r)\). Поскольку \(D\) отделена
от \(C_N(c,r)\), имеем \(B\cap X_L=\varnothing\), и
\(\operatorname{im}\varphi_{L,N}^{c;\delta,r}=\{C_N(c,r)\}\).
Так рисунок показывает, как дополнительное условие глубины сужает
множество значений отображения. Число и форма изображённых компонент
условны. Точка \(c\) на схеме расположена у края \(C_N(c,r)\), чтобы
связать её с граничным численным примером выше; это расположение также
схематично.

\section*{Черновое доказательство теоремы о факторизации}
\label{sec:factorization-draft-proof}
\pdfbookmark[1]{Черновое доказательство теоремы о факторизации}{factorization-draft-proof}

\noindent\emph{Статус.} Цель этого раздела — утверждение
\eqref{eq:factorization-target}. Шаги~1--19 задают постановку и
формулируют рабочую гипотезу \(\mathsf B\). Сразу за ней шаги~20--25
уточняют попытки её доказательства. Шаги~26--48 развивают исходную
условную цепочку, а шаги~49--68 — блоковую геометрию разрезов и
альтернативные мостовые оценки. Рабочая гипотеза \(\mathsf B\)
и все неустановленные оценки заключены в среду \emph{fuzzystep}: текст синий,
рядом проходит вертикальная синяя линия. Условные следствия
\(\mathsf B\) доказаны в шагах~44--48; пока
\eqref{eq:block-bridge-loss} не выведена, теорема не считается доказанной.

\subsection*{Шаг 1. Сведение выбора глубины к боковым компонентам}

Ниже используются вложенность \(O_{N+1}\subseteq O_N\),
\(\M\subseteq O_N\) и равенство \(\bigcap_NO_N=\M\); они установлены
леммой~\ref{lem:outer-sets} в шаге~1.
Для \(c\in\M\), \(0<\delta<r\) и \(N\in\N\) положим
\begin{equation}
 \label{eq:factorization-local-data}
 F_N(c,r):=O_N\cap\overline B(c,r),\qquad
 \mathcal B_N(c;r,\delta):=
 \bigl(F_N(c,r)\setminus C_N(c,r)\bigr)\cap\overline B(c,\delta).
\end{equation}
Множество \(F_N(c,r)\) компактно и полуалгебраично: \(p_j\) — полиномы,
а условия \(z\in K\), \(|p_j(z)|\le2\) и \(|z-c|\le r\) задаются
полиномиальными неравенствами по вещественным координатам. По теореме о
конечности числа компонент полуалгебраического множества
\cite[теорема 2.4.5]{BCR} множество \(F_N(c,r)\) имеет конечное число
компонент. Поэтому \(F_N(c,r)\setminus C_N(c,r)\), а значит и
\(\mathcal B_N(c;r,\delta)\), компактно.

\begin{proposition}[Компактностное сведение факторизации]
\label{prop:factorization-compactness-reduction}
Для фиксированных \(c\in\M\), \(0<\delta<r\) и \(N\in\N\) эквивалентны
условия
\begin{equation}
 \label{eq:factorization-finite-inclusion}
 \exists L\ge N,\qquad
 O_L\cap\overline B(c,\delta)\subseteq C_N(c,r)
\end{equation}
и
\begin{equation}
 \label{eq:no-bad-limit}
 \mathcal B_N(c;r,\delta)\cap\M=\varnothing.
\end{equation}
\end{proposition}
\begin{proof}
Из~\eqref{eq:factorization-finite-inclusion} следует
\(\mathcal B_N(c;r,\delta)\cap\M=\varnothing\) по
\eqref{eq:factorization-local-data} и \(\M\subseteq O_L\). Обратно, пусть
выполнено~\eqref{eq:no-bad-limit}. Если бы
\(\mathcal B_N(c;r,\delta)\cap O_L\ne\varnothing\) для каждого \(L\ge N\),
то вложенные непустые компактные множества
\(\mathcal B_N(c;r,\delta)\cap O_L\) имели бы непустое пересечение.
По \(\bigcap_{L\ge N}O_L=\M\) это пересечение равно
\(\mathcal B_N(c;r,\delta)\cap\M\), которое пусто по
\eqref{eq:no-bad-limit}. Следовательно, для некоторого \(L\ge N\) имеем
\(\mathcal B_N(c;r,\delta)\cap O_L=\varnothing\). Так как \(O_L\subseteq
O_N\) и \(\delta<r\), каждая точка множества
\(\bigl(O_L\cap\overline B(c,\delta)\bigr)\setminus C_N(c,r)\) лежит в
\(\mathcal B_N(c;r,\delta)\cap O_L\). Следовательно, требуемое
включение выполнено.
\end{proof}

По~\eqref{eq:factorization-target} достаточно выбрать \(r,\delta\) так,
чтобы~\eqref{eq:no-bad-limit} выполнялось при каждом \(N\).

\subsection*{Шаг 2. Перевести факторизацию в включение}

Зафиксируем \(c\in\M\), \(0<\delta<r\), \(N\le L\) и положим
\(X_L:=O_L\cap\overline B(c,\delta)\). Компоненты из \(\pi_0(X_L)\)
покрывают \(X_L\). Для каждой такой компоненты \(A\) отображение
\(\varphi_{L,N}^{c;\delta,r}\) переводит \(A\) в компоненту
\(O_N\cap\overline B(c,r)\), содержащую \(A\). Поэтому
\(\varphi_{L,N}^{c;\delta,r}(A)=C_N(c,r)\) для всех \(A\) тогда и
только тогда, когда \(A\subseteq C_N(c,r)\) для всех \(A\), то есть
\[
 X_L\subseteq C_N(c,r).
\]
Следовательно, теорема~\ref{thm:remaining-component-factorization}
равносильна прямой форме
\begin{equation}
 \begin{gathered}
 \forall c\in\M\ \forall\varepsilon>0\
 \exists\,0<\delta<r<\varepsilon\ \forall N\in\N\
 \exists L\ge N,\qquad
 O_L\cap\overline B(c,\delta)\subseteq C_N(c,r).
 \end{gathered}
 \label{eq:factorization-inclusion-form}
\end{equation}
Для внутренних точек проверка непосредственна: если
\(\overline B(c,r)\subseteq\M\), то \(C_N(c,r)=\overline B(c,r)\)
для каждого \(N\), и любое \(0<\delta<r\) даёт нужное включение.
Остаётся рассмотреть граничные точки \(c\in\partial\M\).

\subsection*{Шаг 3. Первый уровень разделения на границе}

Предположим нарушение~\eqref{eq:factorization-inclusion-form}. Тогда
найдутся \(c\in\partial\M\) и \(\varepsilon>0\), для которых при любых
\(0<\delta<r<\varepsilon\) существуют \(N\in\N\) и
\(y\in\mathcal B_N(c;r,\delta)\cap\M\). Зафиксируем \(r\) и выберем
последовательность \(\delta_k\downarrow0\). По
предложению~\ref{prop:factorization-compactness-reduction} для каждого
\(\delta_k\) выбираются уровень \(N_k\) и точка \(y_k\), так что
\begin{equation}
 \label{eq:factorization-failure-sequence}
 y_k\in\mathcal B_{N_k}(c;r,\delta_k)\cap\M.
\end{equation}
Последовательность~\eqref{eq:factorization-failure-sequence}
удовлетворяет \(y_k\to c\) и
\(y_k\in F_{N_k}(c,r)\setminus C_{N_k}(c,r)\).
Для каждого фиксированного \(N\) конечность числа компонент
\(F_N(c,r)\) даёт \(0<\eta_N<r\), для которого
\begin{equation}
 \label{eq:fixed-level-radius}
 F_N(c,r)\cap\overline B(c,\eta_N)\subseteq C_N(c,r).
\end{equation}
Для любого \(M_0\) минимум
\(\min_{0\le N\le M_0}\eta_N\) положителен. Так как \(y_k\in\M\) и
\(y_k\to c\), при достаточно больших \(k\) точка \(y_k\) лежит в
\(C_n(c,r)\) для каждого \(0\le n\le M_0\). Поскольку
\(y_k\in F_{N_k}(c,r)\setminus C_{N_k}(c,r)\), получаем \(N_k>M_0\), а значит,
\(N_k\to\infty\).
Положим
\begin{equation}
 \label{eq:first-splitting-index}
 n_k:=\min\{n\le N_k:y_k\in F_n(c,r)\setminus C_n(c,r)\}.
\end{equation}
Из определения~\eqref{eq:first-splitting-index}, \(c,y_k\in K=O_0\) и
выпуклости \(K\cap\overline B(c,r)\) следует
\(y_k\in C_0(c,r)\); следовательно, \(n_k\ge1\). Из выбора
\(\eta_N\) и \(y_k\in C_n(c,r)\) при \(0\le n\le M_0\) для всех
достаточно больших \(k\) следует \(n_k>M_0\), поэтому \(n_k\to\infty\).
\begin{equation}
 \label{eq:first-splitting-points}
 y_k\in C_{n_k-1}(c,r)\cap
 \bigl(F_{n_k}(c,r)\setminus C_{n_k}(c,r)\bigr).
\end{equation}
Множество \(C_{n_k-1}(c,r)\) связно и содержится в \(O_{n_k-1}\).
Оно содержит \(c,y_k\), а
\(y_k\in F_{n_k}(c,r)\setminus C_{n_k}(c,r)\); следовательно,
\(\exists z_k\in C_{n_k-1}(c,r)\setminus O_{n_k}\). Выберем \(z_k\)
из этого множества. Тогда
\begin{equation}
 \label{eq:first-splitting-witness}
 z_k\in C_{n_k-1}(c,r),\qquad
 |p_{n_k}(z_k)|>2,\qquad
 |p_{n_k}(c)|\le2,\qquad |p_{n_k}(y_k)|\le2.
\end{equation}
На глубинах \(n_k\to\infty\) новая обрезка отделяет точки
\(y_k\to c\). Положение свидетелей \(z_k\) относительно \(c\)
требует отдельной оценки.

\subsection*{Шаг 4. Сформулировать равномерный переход}

Для \(n\ge1\) обозначим
 \begin{equation}
  \label{eq:one-step-slice}
  A_n(c,r):=C_{n-1}(c,r)\cap p_n^{-1}(K).
 \end{equation}
 Из \(F_n=F_{n-1}\cap p_n^{-1}(K)\) следует
 \begin{equation}
  \label{eq:root-component-pullback}
  C_n(c,r)=\operatorname{Comp}_c(A_n(c,r)).
 \end{equation}
 Всякое связное подмножество \(F_n\), содержащее \(c\), лежит в
 \(C_{n-1}(c,r)\), а значит, в \(A_n(c,r)\). Обратно,
 \(A_n(c,r)\subseteq F_n\). Поэтому компоненты \(F_n\) и \(A_n(c,r)\),
 содержащие \(c\), совпадают.
Из~\eqref{eq:first-splitting-points}, включения \(\M\subseteq O_{n_k}\)
и равенства~\eqref{eq:root-component-pullback} получаем
 \(y_k\in A_{n_k}(c,r)\setminus C_{n_k}(c,r)\). Для замыкания черновика
достаточно найти общий по \(n\) радиус \(\delta>0\), при котором
\begin{equation}
 \M\cap\overline B(c,\delta)\cap
 \bigl(A_n(c,r)\setminus C_n(c,r)\bigr)=\varnothing
 \qquad(n\ge1).
 \label{eq:one-step-component-stability}
\end{equation}
Действительно, \(C_0(c,r)=K\cap\overline B(c,r)\), поэтому
\(\M\cap\overline B(c,\delta)\subseteq C_0(c,r)\). Если точка из этого
множества лежит в \(C_{n-1}(c,r)\), то она лежит и в \(p_n^{-1}(K)\);
условие~\eqref{eq:one-step-component-stability} вынуждает её лежать в
\(C_n(c,r)\). Индукция даёт
\begin{equation}
 \M\cap\overline B(c,\delta)\subseteq C_N(c,r)
 \qquad(N\in\N).
 \label{eq:inductive-local-inclusion}
\end{equation}
По~\eqref{eq:inductive-local-inclusion} и предложению
\ref{prop:factorization-compactness-reduction} это даёт
\eqref{eq:factorization-target}.

\paragraph{Коллард функции Грина.}

В обозначениях определения~\ref{def:green-retractions} для \(n\ge1\)
обозначим через \(\Sigma_n(c,r)\) возможное множество
нового разреза при переходе \(C_{n-1}\to C_n\):
\begin{equation}
 \label{eq:green-separator}
 g_n:=2^{2-n}\log3,\qquad
 \Sigma_n(c,r):=C_{n-1}(c,r)\cap\{z:|p_n(z)|=2\}.
\end{equation}
Так как \(C_{n-1}(c,r)\subseteq O_{n-1}\), леммы
\ref{lem:green-depth} и~\eqref{eq:equipotential-distance} дают
\begin{equation}
 \Sigma_n(c,r)\subseteq E_{g_n},\qquad
 \forall z\in\Sigma_n(c,r),\quad
 \dist(z,\M)\le\kappa(g_n).
 \label{eq:green-separator-distance}
\end{equation}
Действительно, \(|p_n|\) непрерывна на связном \(C_{n-1}\); значения
из~\eqref{eq:first-splitting-witness} по разные стороны от \(2\)
вынуждают её принять значение \(2\). Поэтому свидетель отделён от
\(c,y_k\) границей~\eqref{eq:green-separator}. Для локальных мостов
потребуется дополнительно контролировать взаимное положение этой
границы и компонент срезов. Оценка
\eqref{eq:green-separator-distance} задаёт масштаб толщины новых
разрезов.

Ячеечные графы и их точное отождествление с компонентами внешних
срезов, содержащими \(c\), установлены в шагах~8--10 и~15--16.
Их переходные разрезы лежат на
\(\Sigma_n(c,r)\); шаг~19 формулирует рабочую оценку их накопленного
эффекта.

\subsection*{Шаг 5. Представить \(O_N\) как нулевое множество конечного потенциала}

Положим \(\log^+0:=0\), а при \(t>0\) —
\(\log^+t:=\max\{0,\log t\}\). Для \(N\ge1\) определим
\begin{equation}
 \label{eq:finite-orbit-potential}
 \psi_N(z):=
 \max_{1\le j\le N}
 2^{-j}\log^+\!\left(\frac{|p_j(z)|}{2}\right).
\end{equation}
По определению~\eqref{eq:finite-orbit-potential} каждая слагаемая
непрерывна и субгармонична по \(z\), поэтому
\(\psi_N\) также непрерывна и субгармонична. Из \(p_1(z)=z\) следует,
что условие \(\psi_N(z)=0\) уже влечёт \(z\in K\); остальные нулевые
слагаемые означают \(|p_j(z)|\le2\). Следовательно,
\begin{equation}
 \label{eq:finite-potential-zero-set}
 O_N=\{z\in\C:\psi_N(z)=0\},\qquad
 \psi_N\le\psi_{N+1}.
\end{equation}
В~\eqref{eq:finite-potential-zero-set} каждый новый уровень добавляет
одно слагаемое; первая смена компоненты соответствует переходу
\eqref{eq:one-step-slice}.

\subsection*{Шаг 6. Свести общий радиус к положительности одного инфимума}

Для фиксированных \(c\in\M\) и \(r>0\) положим
\begin{equation}
 \label{eq:component-radius}
 \rho_N(c,r):=
 \sup\Bigl(\{0\}\cup
 \bigl\{\delta\in(0,r):
 \M\cap\overline B(c,\delta)\subseteq C_N(c,r)\bigr\}\Bigr).
\end{equation}
Из~\eqref{eq:fixed-level-radius} следует \(\rho_N(c,r)>0\).
Вложенность \(C_{N+1}(c,r)\subseteq C_N(c,r)\) даёт
\(\rho_{N+1}(c,r)\le\rho_N(c,r)\). Для данного \(r\) существование
одного \(\delta>0\), работающего при всех \(N\), равносильно
\begin{equation}
 \label{eq:uniform-component-radius}
 \inf_{N\in\N}\rho_N(c,r)>0.
\end{equation}
Обратное направление следует из определения супремума в
\eqref{eq:component-radius}.

\subsection*{Шаг 7. Оценить высоту Грина у свидетеля первого разделения}

По лемме~\ref{lem:green-depth} и включению
\(z_k\in C_{n_k-1}(c,r)\subseteq O_{n_k-1}\) из
\eqref{eq:first-splitting-witness} имеем
\begin{equation}
 \label{eq:green-depth-first-split}
 0\le G(z_k)\le 2^{2-n_k}\log3\longrightarrow0.
\end{equation}
Из~\eqref{eq:green-depth-first-split}, положив
\(s_k:=2^{2-n_k}\log3\), получаем \(z_k\in E_{s_k}\) и
\begin{equation}
 \label{eq:split-witness-distance}
 \dist(z_k,\M)\le\kappa(s_k)\longrightarrow0.
\end{equation}
По~\eqref{eq:split-witness-distance} разрезающие свидетели приближаются
к \(\M\); они лежат в сужающемся по потенциалу
колларде вокруг \(\M\). Их локализация около \(c\) рассматривается
отдельно.

\subsection*{Шаг 8. Перенести свидетеля в конечную систему ячеек}

Пусть \((\lambda_m)\) — последовательность глубин из
теоремы~\ref{thm:constructed-external-maps}, доказанной в приложении
\ref{app:approximation-proof}; по
\eqref{eq:external-depths} она строго возрастает к бесконечности.
Для больших \(k\) выберем
\begin{equation}
 \label{eq:split-grid-depth}
 m_k:=\max\{m\ge2:\lambda_m\le n_k-1\}.
\end{equation}
Так как \(\lambda_m\to\infty\) по~\eqref{eq:external-depths} и
\(n_k\to\infty\), множество в~\eqref{eq:split-grid-depth} непусто при
больших \(k\) и \(m_k\to\infty\).
По~\eqref{eq:first-splitting-witness} имеем
\(z_k\in O_{n_k-1}\subseteq O_{\lambda_{m_k}}\); значит, \(z_k\)
лежит в некоторой внешней ячейке \(U_{m_k,\alpha_k}\). Предложение
\ref{prop:cellwise-inner-outer} даёт один из двух вариантов: либо
ячейка содержит внутреннюю часть \(I_{m_k,\alpha_k}\) и
\(\dist(z_k,I_{m_k,\alpha_k})\le8\cdot2^{-m_k}\) по
\eqref{eq:dyadic-grid-diameter}, либо
\(\dist(z_k,\M)\le h_{\lambda_{m_k}}\). По
 \eqref{eq:external-mesh-bound}, \(h_{\lambda_{m_k}}\le2^{-m_k}\);
так как \(m_k\to\infty\), обе оценки стремятся к нулю. Внешний свидетель приближается к внутренней
 геометрии. Для следующего шага требуется установить связность его
 ячейки с ячейкой \(c\).

\subsection*{Шаг 9. Отождествить графовый кластер с компонентой внешнего уровня}

Для фиксированного \(m\) множества
\[
 A\in\pi_0\bigl(U_{m,\alpha}\cap\overline B(c,r)\bigr)
\]
являются компактными связными множествами, конечными по числу и
покрывают \(O_{\lambda_m}\cap\overline B(c,r)\). Их граф смежности
соединяет две вершины ровно тогда, когда соответствующие множества
пересекаются. Объединение вершин каждой связной компоненты графа
связно. Объединения для разных компонент графа попарно дизъюнктны
и компактны; поскольку таких объединений конечное число, они задают
разбиение на компоненты связности всего множества
\(O_{\lambda_m}\cap\overline B(c,r)\). Следовательно,
\begin{equation}
 H_m(c,r)=C_{\lambda_m}(c,r).
\label{eq:dyadic-graph-component}
\end{equation}
Граф точно задаёт компоненту \(O_{\lambda_m}\cap\overline B(c,r)\),
содержащую \(c\).

\subsection*{Шаг 10. Свести ячеечный граф к общему радиусу}

По~\eqref{eq:dyadic-graph-component} кластер графа на уровне \(m\)
равен \(C_{\lambda_m}(c,r)\). Так как \(\lambda_m\to\infty\) и
\(C_{N+1}(c,r)\subseteq C_N(c,r)\), включение
\(\M\cap\overline B(c,\delta)\subseteq C_{\lambda_m}(c,r)\) для всех
\(m\) равносильно требованию
\begin{equation}
 \label{eq:common-component-neighborhood}
 \M\cap\overline B(c,\delta)\subseteq C_N(c,r)
 \qquad\text{для всех }N.
\end{equation}
Действительно, \(\lambda_m\to\infty\), а компоненты \(C_N(c,r)\)
вложены; поэтому контроль на выбранных уровнях \(\lambda_m\) влечёт
контроль на каждом \(N\), и наоборот. Графовая формулировка переводит
каждый конечный срез в задачу о конечном графе. Для завершения требуется
показать, что кластер \(c\) содержит все близкие точки \(\M\) на
сколь угодно глубоких уровнях.

\subsection*{Шаг 11. Развернуть рекурсию производных}

Пусть \(D_n(c):=p_n'(c)\). Из
\(p_{n+1}(c)=p_n(c)^2+c\) следует
\begin{equation}
 \label{eq:parameter-derivative-recurrence}
 D_1(c)=1,\qquad D_{n+1}(c)=2p_n(c)D_n(c)+1,
\end{equation}
 последовательное раскрытие~\eqref{eq:parameter-derivative-recurrence}
 даёт точную формулу
\begin{equation}
 \label{eq:parameter-derivative-series}
 D_n(c)=
 \sum_{j=0}^{n-1}2^j\prod_{\ell=1}^{j}p_{n-\ell}(c),
 \qquad\text{(пустое произведение равно \(1\)).}
\end{equation}
Для \(c\in\M\) из~\eqref{eq:parameter-derivative-series} следует
\begin{equation}
 \label{eq:parameter-derivative-bound}
 |D_n(c)|\le\sum_{j=0}^{n-1}4^j=\frac{4^n-1}{3}.
\end{equation}
Граница~\eqref{eq:parameter-derivative-bound} растёт с \(n\), а
слагаемые в~\eqref{eq:parameter-derivative-series} могут взаимно
сокращаться. Для оценки устойчивости требуется отдельный контроль
производных на орбите и положительная нижняя граница радиуса.

\subsection*{Шаг 12. Получить обратную ветвь и изолировать её ограничение}

Если \(D_n(c)\ne0\), из~\eqref{eq:parameter-derivative-recurrence}
положим
\begin{equation}
 \label{eq:inverse-branch-radius}
 M_{2,n}:=\sup_{\zeta\in\overline B(c,r)}|p_n''(\zeta)|,\qquad
 \sigma_n:=\min\left\{r,\frac{|D_n(c)|}{2(1+M_{2,n})}\right\}.
\end{equation}
Для \(z,w\in\overline B(c,\sigma_n)\) интегрирование производной по
отрезку \([w,z]\) даёт
\begin{equation}
 \label{eq:inverse-branch-lipschitz}
 |p_n(z)-p_n(w)|\ge\frac{|D_n(c)|}{2}|z-w|.
\end{equation}
При радиусе \(\sigma_n\) из~\eqref{eq:inverse-branch-radius}
неравенство~\eqref{eq:inverse-branch-lipschitz} показывает, что \(p_n\)
инъективно на этом диске. Радиус \(\sigma_n\) зависит от \(n\) и может стремиться к нулю. Для
компонентного вывода требуется также контролировать связность множества
\(p_n(\overline B(c,\sigma_n))\cap K\) и его положение относительно
\(C_{n-1}(c,r)\).

\subsection*{Шаг 13. Сосчитать глобальные компоненты лемнискаты}

Из рекурсии \(p_{n+1}=p_n^2+c\) следует
\begin{equation}
 d_1=1,\qquad d_{n+1}=2d_n,\qquad d_n:=\deg p_n=2^{n-1}.
 \label{eq:parameter-polynomial-degree}
\end{equation}
По~\eqref{eq:parameter-polynomial-degree} \(p_n\) имеет степень \(d_n\).
Пусть окружность \(\partial K\) состоит из регулярных значений \(p_n\);
обозначим через \(k_n\) число компонент
\(p_n^{-1}(B(0,2))\), а через \(d_i\) степень ограничения \(p_n\) на
\(i\)-ю компоненту, \(1\le i\le k_n\). Тогда
\(\sum_{i=1}^{k_n}d_i=d_n\). Регулярная лемниската
\(\{|p_n|=2\}\) состоит из конечного числа гладких замкнутых кривых.
Если компонента \(p_n^{-1}(B(0,2))\) имела бы отверстие, область между
соседними граничными кривыми внутри этого отверстия лежала бы в
\(\{|p_n|>2\}\); принцип максимума на её замыкании дал бы
\(|p_n|\le2\), противоречие. Поэтому все компоненты односвязны, а
ограничения \(p_n\) являются собственными отображениями на \(B(0,2)\)
степеней \(d_i\). Формула
Римана--Гурвица \cite{Forster} даёт
\begin{equation}
 \sum_{p_n'(\zeta)=0,\ |p_n(\zeta)|<2}
       \operatorname{mult}_{\zeta}(p_n')
   =d_n-k_n,\qquad
 \sum_{p_n'(\zeta)=0,\ |p_n(\zeta)|>2}
       \operatorname{mult}_{\zeta}(p_n')
   =k_n-1.
 \label{eq:lemniscate-critical-count}
\end{equation}
Уравнение~\eqref{eq:lemniscate-critical-count} считает критические
точки глобальной лемнискаты. Для локального аргумента требуется также
оценить их расположение относительно \(c\).

\subsection*{Шаг 14. Отделить глобальную лемнискату от локального среза}

Для нужного перехода используется локальный срез
\(A_n(c,r)\) из~\eqref{eq:one-step-slice}. Его компоненты зависят от
взаимного положения \(p_n^{-1}(K)\) и \(C_{n-1}(c,r)\): пересечение
связных множеств может иметь несколько компонент. Формула
Римана--Гурвица даёт глобальный счёт; для
\eqref{eq:one-step-component-stability} требуется дополнить его
локальной геометрической оценкой.

\subsection*{Шаг 15. Представить каждый локальный срез конечным графом}

Для сетки \(Q_{m,\alpha}\) из~\eqref{eq:dyadic-grid-cells} при
\(m\ge2\) положим
\begin{equation}
 \label{eq:grid-cell-slices}
 V_{n,m,\alpha}:=F_n(c,r)\cap Q_{m,\alpha}.
\end{equation}
Полуалгебраичность даёт конечное число компонент у каждого
\(V_{n,m,\alpha}\) \cite[теорема 2.4.5]{BCR}. В графе
\(\mathsf G_{n,m}(c,r)\) вершины —
компоненты~\eqref{eq:grid-cell-slices}, ребро означает непустое
пересечение.
Объединение вершин одной графовой компоненты связно; разные
объединения образуют конечное замкнутое разбиение \(F_n(c,r)\).
Пусть \(H_{n,m}(c,r)\) — объединение вершин графовой компоненты,
содержащей вершину с \(c\in A\). Тогда
\begin{equation}
 H_{n,m}(c,r)=C_n(c,r).
 \label{eq:grid-root-component}
\end{equation}
Включение \(F_n\hookrightarrow F_{n-1}\)
задаёт отображение компонент; потеря нужной связности происходит
ровно при переходе от уровня \(n-1\) к \(n\).

\subsection*{Шаг 16. Сформулировать требуемую оценку перехода графов}

Зафиксируем \(c\in\M\), \(\varepsilon>0\). Искомое равномерное
условие в клеточной форме:
\begin{equation}
 \begin{gathered}
 \exists\,0<\delta<r<\varepsilon,\qquad
 \forall n\ge1\ \forall m\ge2\ \forall\alpha\in J_m^2\
 \forall A\in\pi_0(V_{n,m,\alpha}),\\
 A\cap\M\cap\overline B(c,\delta)\ne\varnothing
 \Rightarrow A\subseteq H_{n,m}(c,r).
 \end{gathered}
 \label{eq:grid-transition-stability}
\end{equation}
Здесь \(J_m^2\) и \(Q_{m,\alpha}\) заданы в
\eqref{eq:dyadic-grid-cells}. Для фиксированных \(n,m\),
\eqref{eq:grid-root-component} отождествляет \(H_{n,m}(c,r)\) с
\(C_n(c,r)\). Условие~\eqref{eq:grid-transition-stability}
достаточно для~\eqref{eq:common-component-neighborhood}: для
\(y\in\M\cap\overline B(c,\delta)\) берём клетку, содержащую \(y\),
и компоненту \(A\) этой клетки, содержащую \(y\); тогда
\eqref{eq:grid-transition-stability} даёт \(A\subseteq C_n(c,r)\),
а значит, \(y\in C_n(c,r)\). Условие~\eqref{eq:common-component-neighborhood} контролирует точки
\(\M\), тогда как вывод~\eqref{eq:grid-transition-stability}
охватывает всю клеточную компоненту \(A\), включая её точки вне
\(\M\). Поэтому~\eqref{eq:grid-transition-stability} является более
сильным достаточным условием.
Уровень \(N=0\) включён автоматически, поскольку
\(\M\cap\overline B(c,\delta)\subseteq C_0(c,r)\).
По шагу~4 это равносильно
\eqref{eq:one-step-component-stability}. Оценки
\eqref{eq:dyadic-grid-diameter} и
\(h_{\lambda_m}\to0\) по~\eqref{eq:external-mesh-bound} контролируют
размеры и расстояния вершин. Для завершения требуется оценить цепи
смежных рёбер при всех \(n\).

\subsection*{Шаг 17. Кодировать локальную компоненту связными мостами}

Для \(y\in\M\cap\overline B(c,r)\) положим
\begin{equation}
 \label{eq:bridge-family}
 \mathfrak B_N(c,r;y):=
 \left\{\Gamma\subseteq F_N(c,r):
 \Gamma\text{ компактно и связно},\ c,y\in\Gamma\right\}.
\end{equation}
Тогда
\begin{equation}
 \label{eq:bridge-component-equivalence}
 y\in C_N(c,r)\quad\Leftrightarrow\quad
 \mathfrak B_N(c,r;y)\ne\varnothing.
\end{equation}
В прямом направлении берём \(\Gamma=C_N(c,r)\); обратное следует из
определения компоненты. Графовый путь строит \(\Gamma\) как конечное
объединение пересекающихся клеточных компонент.

\subsection*{Шаг 18. Замкнуть условный шаг через компактностное сведение}

Для фиксированных \(c\in\M\) и \(\varepsilon>0\), если
\begin{equation}
 \label{eq:all-level-bridges}
 \exists\,0<\delta<r<\varepsilon\ \forall N\in\N\
 \forall y\in\M\cap\overline B(c,\delta),\qquad
 \mathfrak B_N(c,r;y)\ne\varnothing,
\end{equation}
то~\eqref{eq:bridge-component-equivalence} даёт
\begin{equation}
 \M\cap\overline B(c,\delta)\subseteq C_N(c,r)
 \qquad(N\in\N).
 \label{eq:bridge-to-components}
\end{equation}
Включение~\eqref{eq:bridge-to-components} есть ровно
условие~\eqref{eq:common-component-neighborhood} для
выбранных \(c,r,\delta\). По~\eqref{eq:no-bad-limit} и
предложению~\ref{prop:factorization-compactness-reduction}
получаем~\eqref{eq:factorization-target}.

 \subsection*{Шаг 19. Сформулировать равномерный бюджет потерь}

 Пусть \(c\in\partial\M\), \(\varepsilon>0\), а
 \((\lambda_m,s_m)_{m\ge2}\) — последовательности из
 теоремы~\ref{thm:constructed-external-maps}. Тогда
 \[
  \lambda_m\ge m,\qquad
  2^{1-\lambda_m}\log3\le s_m,\qquad
  \kappa(s_m)\le2^{-m}.
 \]
 Для выбранного \(0<r<\varepsilon\) положим
 \begin{equation}
  R_m(c,r):=\rho_{\lambda_m}(c,r),\qquad
  \mu_m:=8\cdot2^{-m}+\kappa(s_m).
  \label{eq:block-radius-scales}
 \end{equation}
 В частности, \(0<\mu_m\le9\cdot2^{-m}\). Если
 \(\lambda_m<n\le\lambda_{m+1}\), то
 \[
  \Sigma_n(c,r)\subseteq C_{n-1}(c,r)
  \subseteq O_{n-1}\subseteq O_{\lambda_m}\subseteq E_{s_m}.
 \]
 Следовательно,
 \(\dist(\Sigma_n(c,r),\M)\le\kappa(s_m)\); также
 \(\operatorname{diam}Q_{m,\alpha}\le8\cdot2^{-m}\) для ячеек
 сетки~\eqref{eq:dyadic-grid-diameter}.
 Эти оценки описывают близость разрезов к \(\M\) и размер ячеек.
 Число \(\mu_m\) в~\eqref{eq:block-radius-scales} служит масштабом
 для оценки потери связности.

 Для предложенного количественного бюджета положим
 \begin{equation}
  \eta_m(c,r):=
  C_{c,r}\left(
  \sqrt{\frac{\mu_m}{r}}+
  \sum_{n>\lambda_m}\sqrt{g_n}\right),
  \qquad C_{c,r}>0.
  \label{eq:block-radius-loss}
 \end{equation}
 Для любой фиксированной константы \(C_{c,r}\), независимой от \(m\),
 геометрический ряд даёт
 \begin{equation}
  \sum_{n>\lambda_m}\sqrt{g_n}
  \le
  \frac{\sqrt{2\log3}}{1-2^{-1/2}}\,2^{-\lambda_m/2}
  \le
  \frac{\sqrt{2\log3}}{1-2^{-1/2}}\,2^{-m/2}.
  \label{eq:tail-critical-loss}
 \end{equation}
 Отсюда \(\eta_m(c,r)\le C'_{c,r}2^{-m/2}\) и
 \(\sum_{m\ge2}\eta_m(c,r)<\infty\). Для применения этого бюджета к
 радиусам требуется установить оценку
 \(R_m(c,r)-R_{m+1}(c,r)\le\eta_m(c,r)R_m(c,r)\).

 \begin{fuzzystep}
 \textbf{Рабочая гипотеза \(\mathsf B\) (отдельное условие).}
 \begin{equation}
  \begin{gathered}
  \forall c\in\partial\M\ \forall\varepsilon>0\
  \exists r\in(0,\varepsilon)\ \exists C_{c,r}>0\
  \exists m_0\ge2\ \forall m\ge m_0,\\
  \eta_m(c,r)\le\tfrac12,\qquad
  R_{m+1}(c,r)\ge
  \bigl(1-\eta_m(c,r)\bigr)R_m(c,r),
  \end{gathered}
  \label{eq:uniform-radius-loss-hypothesis}
 \end{equation}
 где \(R_m\) и \(\eta_m\) заданы в
 \eqref{eq:block-radius-scales} и~\eqref{eq:block-radius-loss}.
 Доказательство~\eqref{eq:uniform-radius-loss-hypothesis} должно связать
 расстояние разрезов до \(\M\) с нижней оценкой сохранённого радиуса.

 Оценка требует равномерного контроля обратного образа в параметре
 \(c\). Локальная нормальная форма \(u\mapsto u^2\) и компактность
 семейства \((c,u)\mapsto(c,u^2+c)\) задают модели для такого анализа.
 При этом \(D_n(c)=p_n'(c)\) может обращаться в нуль, а радиус
 действия~\eqref{eq:inverse-branch-lipschitz} зависит от \(n\).
 Глобальный счёт~\eqref{eq:lemniscate-critical-count} охватывает
 критические точки всей лемнискаты; локальная оценка требует также
 контролировать их расположение около рассматриваемой компоненты.
 Поскольку \(\deg p_n=2^{n-1}\) по
\eqref{eq:parameter-polynomial-degree}, равномерный бюджет по блокам
 должен учитывать глубину критических точек. Условие
\eqref{eq:grid-transition-stability} усиливает требуемое включение, а
оценки диаметров ячеек дополняются оценкой цепей мостов. Сложение
угловых потерь и выбор общей константы \(C_{c,r}\) входят в эту
исследовательскую программу.
\end{fuzzystep}

\factorizationstep{20}{Нормировать динамическую высоту интерфейса}
Положим \(\ell(w):=\log^+|w|\). Для \(|c|\le2\) и \(w\in\C\)
\[
 \bigl|\ell(f_c(w))-2\ell(w)\bigr|\le\log4.
\]
При \(|w|\le1\) это следует из \(|f_c(w)|\le3\). Если
\(1\le|w|\le2\), верхняя оценка следует из
\(|f_c(w)|\le|w|^2+2\le3|w|^2\), а нижняя — из
\(\ell(f_c(w))\ge0\) и \(2\ell(w)\le\log4\). При \(|w|\ge2\)
имеем
\[
 \frac{|w|^2}{2}\le |f_c(w)|\le\frac32|w|^2,
\]
что даёт требуемое неравенство. Телескопирование определяющего предела
динамической функции Грина даёт равномерную оценку
\begin{equation}
 \bigl|g_c(w)-\ell(w)\bigr|
 \le\sum_{k=0}^\infty2^{-(k+1)}\log4
 =\log4.
 \label{eq:green-vs-log-radius}
\end{equation}
Так как \(p_n(c)=f_c^{\circ n}(0)\), соотношение
\eqref{eq:green-normalization} даёт
\[
 g_c(p_n(c))=2^n g_c(0)=2^{n-1}G(c).
\]
На интерфейсе \(\Sigma_n\) выполнено \(\ell(p_n)=\log2\). Поэтому
\eqref{eq:green-vs-log-radius} влечёт более точную локализацию
\[
 \Sigma_n(c,r)\subseteq E_{\widehat g_n},\qquad
 \dist(\Sigma_n(c,r),\M)\le\kappa(\widehat g_n),\qquad
 \widehat g_n:=2^{1-n}\log8<g_n.
\]
Она улучшает константу в колларе шага~19, сохраняя порядок \(2^{-n}\).

\factorizationstep{21}{Проверить масштаб перехода к образу}
Для \(z\in E_{\widehat g_n}\cap K\) точное соотношение динамических и
параметрической функций Грина даёт
\[
 g_z(p_n(z))=2^{n-1}G(z)\le\log8.
\]
Из~\eqref{eq:green-vs-log-radius} следует лишь
\(\ell(p_n(z))\le\log32\), то есть равномерная ограниченность
\(|p_n(z)|\), независимая от \(n\). На интерфейсе
\(|p_n(\zeta)|=2\), но эта оценка не даёт для окрестности карты
\[
 |p_n(z)-p_n(\zeta)|\le t_{n,j}
 \quad\text{с }t_{n,j}\longrightarrow0.
\]
Именно такой малый образный радиус требуется для суммирования весов
из~\eqref{eq:local-degree-pullback}. Поэтому подстановка
\(t_{n,j}=g_n\) или \(t_{n,j}=\sqrt{g_n}\) из одной лишь оценки высоты
Грина не обоснована. Это не исключает более сильной локальной оценки;
её необходимо получить отдельно.

\factorizationstep{22}{Разбить контакты по динамической высоте}
Для \(\zeta\in\Sigma_n\) введём динамическую высоту
\[
 u_n(\zeta):=g_\zeta(p_n(\zeta))=2^{n-1}G(\zeta)\in[0,\log8].
\]
При \(0<\tau\le\log8\) разделим интерфейс на
\[
 \Sigma_n^-(\tau):=\{\zeta\in\Sigma_n:u_n(\zeta)\le\tau\},
 \qquad
\Sigma_n^+(\tau):=\{\zeta\in\Sigma_n:u_n(\zeta)>\tau\}.
 \]
 Из точного равенства для \(u_n\) следуют включения
 \[
\Sigma_n^-(\tau)\subseteq E_{2^{1-n}\tau},\qquad
\Sigma_n^+(\tau)\subseteq
E_{\widehat g_n}\setminus E_{2^{1-n}\tau}.
 \]
 Это задаёт два разных режима для карт шага~67: малую динамическую
 высоту около невылетающего множества и параметрический слой между
двумя уровнями функции Грина.
\begin{fuzzystep}
После выбора \(\tau_m\) в шаге~23 кандидатный план состоит в том,
 чтобы оценить низковысотную часть через локальную геометрию динамических
 подуровней, а высоковысотную — через упаковку карт в параметрическом
 слое. Для
 применения первого механизма потребуется равномерно контролировать
 изменение \(g_c(w)\) при одновременном изменении параметра и образа;
 для второго — оценить число и кратность карт в слое
 \(E_{\widehat g_n}\setminus E_{2^{1-n}\tau_m}\). Ни одно из этих
 утверждений пока не следует из конечности графа или формулы
Римана--Гурвица. Поэтому разбиение уточняет недостающие леммы, но само
по себе ещё не даёт оценки \(\mathfrak W_m\).
\end{fuzzystep}

\factorizationstep{23}{Получить строгий бюджет низковысотных контактов}
Для \(\zeta\in\Sigma_n^-(\tau_m)\) точное равенство
\(G(\zeta)=2^{1-n}u_n(\zeta)\) даёт
\[
 G(\zeta)\le2^{1-n}\tau_m.
\]
Функция \(\kappa\) не убывает, поскольку \(E_s\) вложены по \(s\).
Зафиксируем глубины \(\lambda_m\), положим \(\tau_1:=\log8\) и
последовательно выберем \(\tau_m\) из условия
\[
 0<\tau_m\le\min\{2^{-m},\tau_{m-1}/2,\log8\},\qquad
 \kappa(2^{-\lambda_m}\tau_m)\le2^{-2m}.
\]
Такой выбор возможен из \(\kappa(s)\to0\), и \(\tau_m\downarrow0\).
Тогда для каждого \(\lambda_m<n\le\lambda_{m+1}\) и каждого
\(\zeta\in\Sigma_n^-(\tau_m)\)
\begin{equation}
 \dist(\zeta,\M)
 \le\kappa(2^{1-n}\tau_m)
 \le\kappa(2^{-\lambda_m}\tau_m)
 \le2^{-2m}.
 \label{eq:low-height-collar}
\end{equation}
Кроме того, если
\[
 a_{n,m}^-:=
 \sup\bigl(\{G(\zeta):\zeta\in\Sigma_n^-(\tau_m)\}\cup\{0\}\bigr),
\]
то
\begin{equation}
 \sum_{\lambda_m<n\le\lambda_{m+1}}\sqrt{a_{n,m}^-}
 \le
  \frac{\sqrt{\tau_m}}{1-2^{-1/2}}\,2^{-\lambda_m/2}
  \le\frac{2^{-m}}{1-2^{-1/2}}.
 \label{eq:low-height-budget}
\end{equation}
Итак, малая динамическая высота даёт строгие поперечный и суммарный
Green-бюджеты для низковысотных контактов. Эти оценки пока не
контролируют число контактных ячеек или образные радиусы карт.

\factorizationstep{24}{Сопоставить полную цену разрезов с мерой контактов}
Семейство \(I_{n,m}\) покрывает \(\Sigma_n\), поэтому его выбор удаляет
все ребра между вершинами \(A\) и \(\overline U^{\,C_{n-1}}\). Оно
допустимо и в задаче шага~32. Следовательно,
\[
 \operatorname{cap}_{n,m}(c,r)
 \le\sum_{\alpha\in I_{n,m}}\operatorname{diam}Q_{m,\alpha}.
\]
Суммируя по уровням блока и переставляя конечные суммы, получаем
\begin{equation}
 \mathfrak C_m(c,r)
 \le\sum_{\alpha\in J_m^2}
       v_{m,\alpha}\operatorname{diam}Q_{m,\alpha}
 =\mathfrak N_m(\C),
 \label{eq:capacity-by-contact-mass}
\end{equation}
где \(\mathfrak N_m\) определена в шаге~61.
В силу~\eqref{eq:capacity-by-contact-mass} для второго неравенства
шага~43 достаточно доказать
\[
 \mathfrak N_m(\C)
 \le B_{c,r}R_m(c,r)
 \left(\sqrt{\frac{\mu_m}{r}}+
       \sum_{\lambda_m<n\le\lambda_{m+1}}\sqrt{g_n}\right).
\]
Это точное достаточное условие обходит локальные обратные карты и
Green-to-image оценку,
но требует оценки всей массы контактов, более сильной, чем требуемая
оценка минимальной цены разреза. Потенциальная оценка из шага~61 сама
по себе её не влечёт: ядро в интеграле не превосходит единицы.

\factorizationstep{25}{Сосредоточить разрез на ближайшей потере}
При \(\Delta_m>0\) выберем ближайшую точку
\(y_m\in\mathcal L_m(c,r)\) и положим
\(n_m:=\nu_m(y_m)\). Обозначим через \(A_m^*\) компоненту
\(A_{n_m}(c,r)\), содержащую \(y_m\). Она M-содержащая и отлична от
\(C_{n_m}(c,r)\). Пусть \(\mathcal S_{n_m,m}(A_m^*)\) состоит из семейств
\(S\subseteq I_{n_m,m}\), которые по правилу шага~32 удаляют все пути
от вершины \(C_{n_m}\) к \(A_m^*\). По шагу~24 это семейство непусто;
конечность сетки обеспечивает достижение минимума. Целевую цену зададим
формулой
\[
\begin{aligned}
 \operatorname{cap}_{n_m,m}(A_m^*)&:=
 \min_{S\in\mathcal S_{n_m,m}(A_m^*)}
 \sum_{\alpha\in S}\operatorname{diam}Q_{m,\alpha},\\
 \widehat{\mathfrak C}_m&:=
 \operatorname{cap}_{n_m,m}(A_m^*).
\end{aligned}
\]
Сравнение допустимых семейств даёт
\[
 0\le\widehat{\mathfrak C}_m
 \le\operatorname{cap}_{n_m,m}(c,r)
 \le\mathfrak C_m(c,r).
\]
Если \(\Delta_m=0\), положим \(\widehat{\mathfrak C}_m=0\); тогда
\(0\le\widehat{\mathfrak C}_m\le\mathfrak C_m(c,r)\) также выполнено.
Если \(\Delta_m>0\), вершины \(C_{n_m}\) и \(A_m^*\) различны.
Связность графа интерфейсов из шага~30 показывает, что пустое семейство
ячеек не может быть допустимым разрезом, поэтому
\(\widehat{\mathfrak C}_m>0\). Каждая ячейка сетки~\eqref{eq:dyadic-grid-cells}
имеет сторону \(4\cdot2^{-m}\) и диаметр \(4\sqrt2\,2^{-m}\); значит,
\[
 \widehat{\mathfrak C}_m\ge4\sqrt2\,2^{-m},\qquad
 \Delta_m\le R_m(c,r)\le r
 \le\frac{r\,2^m}{4\sqrt2}\widehat{\mathfrak C}_m.
\]
Это строгое конечномасштабное сравнение, однако его коэффициент
растёт с \(m\) и потому не даёт требуемой равномерной оценки.
\begin{fuzzystep}
В блоках с \(\Delta_m>0\) для доказательства \(\mathsf B\) предлагается
оценить один целевой разрез:
\[
 \Delta_m\overset{?}{\le}A_{c,r}\widehat{\mathfrak C}_m,\qquad
 \widehat{\mathfrak C}_m\overset{?}{\le}
 B_{c,r}R_m(c,r)
 \left(\sqrt{\frac{\mu_m}{r}}+\sqrt{g_{n_m}}\right),
 \quad A_{c,r},B_{c,r}\text{ не зависят от }m.
\]
Обе целевые оценки пока открыты.
\end{fuzzystep}
При условии, что обе оценки верны и \(\Delta_m>0\), имеем
\(n_m>\lambda_m\) и
\(\sqrt{g_{n_m}}\le\sum_{n>\lambda_m}\sqrt{g_n}\). Положим
\(C_{c,r}:=\max\{1,A_{c,r}B_{c,r}\}\).
\[
 \Delta_m\le A_{c,r}\widehat{\mathfrak C}_m
 \le A_{c,r}B_{c,r}R_m
   \left(\sqrt{\frac{\mu_m}{r}}+\sqrt{g_{n_m}}\right)
 \le R_m\eta_m.
\]
При \(\Delta_m=0\) последнее неравенство выполняется автоматически.
Следовательно, доказательство обеих целевых оценок дало бы
\(\mathsf B\) по шагам~45--48. Сам вывод условной импликации строгий;
условия, от которых он зависит, остаются недоказанными.

\begin{fuzzystep}
Этот маршрут устраняет суммирование цен всех уровней блока и может
сделать разбиение шага~22 ненужным.
\end{fuzzystep}

\paragraph{Ограничение конечной графовой модели.}
На пути \(P_{2k}\) с вершинами \(v_j=j/(4k)\), \(0\le j\le2k\), назначим
каждому ребру цену \(1/(4k)\). Разделяющий набор должен содержать хотя
бы одно ребро, а удаление одного ребра достаточно, поэтому
\(\operatorname{cap}(P_{2k})=1/(4k)\). При радиусах
\(R=1\), \(R'=|v_{2k}-v_0|=1/2\) отношение
\((R-R')/\operatorname{cap}(P_{2k})=2k\) не ограничено. Следовательно,
одна лишь конечная графовая цена не даёт равномерной радиальной оценки.
Это не контрпример для системы срезов \((O_N)\): для неё нужна
дополнительная геометрическая связь радиального обхода с контактами.

\subsection*{Шаг 26. Эвристический план доказательства рабочей гипотезы}

\begin{fuzzystep}
Цель — вывести рабочую гипотезу
\eqref{eq:uniform-radius-loss-hypothesis} из конечных оценок без её
постулирования. После точного выражения потери через множество
\(\mathcal L_m\) в шаге~28 достаточно получить
две оценки из~\eqref{eq:cut-to-radius-and-budget}:
\[
 \bigl(R_m-\dist(c,\mathcal L_m)\bigr)_+
 \le A_{c,r}\mathfrak C_m,\qquad
 \mathfrak C_m\le B_{c,r}R_m
 \left(\sqrt{\frac{\mu_m}{r}}+
 \sum_{\lambda_m<n\le\lambda_{m+1}}\sqrt{g_n}\right).
\]
Для первой оценки следует выбрать ближайшую потерянную точку и её
первый уровень выхода. Путь в предшествующей компоненте пересекает
интерфейс этого уровня; контактный граф должен превратить такой путь
в разрез с контролируемым радиальным положением. Одной конечности
графа для этого мало: требуется оценить обход пути либо обеспечить
внутреннюю доступность относительно внешнего разреза.

Для второй оценки интерфейсные контакты нужно покрыть регулярными и
критическими картами. Сначала требуется перевести Green-высоту
\(g_n\) в образный радиус \(t\) отображения \(p_n\), затем применить
локальные оценки с учётом степеней \(q_n\). Суммирование вкладов должно
контролировать кратность контактов в ячейках \(v_{m,\alpha}\),
перекрытие карт и их критические константы; предполагаемая форма
такого контроля — локальный Carleson-бюджет для меры контактов.

Если обе оценки доказаны с константами, не зависящими от \(m\), точное
равенство~\eqref{eq:block-radius-loss-exact} и верхняя оценка блочной
суммы хвостом из шага~19 дают
\(\beta_m\le\eta_m\), если \(C_{c,r}\) выбрана не меньше произведения
\(A_{c,r}B_{c,r}\). Геометрическое убывание тогда обеспечивает
\(\eta_m\le\tfrac12\) при больших \(m\), что и завершает
\eqref{eq:uniform-radius-loss-hypothesis}. Сами радиальный мост,
Green-to-image переход и Carleson-оценка пока остаются открытыми.
\end{fuzzystep}

\Needspace{8\baselineskip}
\subsection*{Шаг 27. Эвристически реализовать мостовой бюджет}

Положим
\(\Delta_m:=R_m(c,r)-R_{m+1}(c,r)\ge0\).
\begin{fuzzystep}
Для каждого уровня блока граф интерфейсов задаёт допустимые ячеечные
разрезы; сумма их минимальных цен есть \(\mathfrak C_m\). На карте
локальной степени \(q_{n,j}\) формула~\eqref{eq:local-degree-pullback}
даёт вклад \((t_{n,j}/a_{n,j})^{1/q_{n,j}}\), а на регулярной карте
оценка шага~36 даёт \(2t_{n,j}/a_{n,j}\). Сумму по картам с учётом
перекрытий обозначим \(\mathfrak W_m\).

Эта эвристическая реализация приводит к цепочке
\[
 \Delta_m
 \overset{?}{\le} A_{c,r}\mathfrak C_m,\qquad
 \mathfrak C_m
 \overset{?}{\le} B_{c,r}\mathfrak W_m,\qquad
 \mathfrak W_m
 \overset{?}{\le} D_{c,r}R_m
 \left(\sqrt{\frac{\mu_m}{r}}+
 \sum_{\lambda_m<n\le\lambda_{m+1}}\sqrt{g_n}\right).
\]
Три вопросительных знака отмечают недоказанные переходы: радиальную
оценку потери через разрез, сравнение цены разреза с весом карт и
Green-to-image/Carleson-бюджет. Поэтому цепочка пока остаётся
эвристической; условное умножение оценок уже записано в шагах~25 и~44.
\end{fuzzystep}

\Needspace{8\baselineskip}
\subsection*{Шаг 28. Локализовать потерю радиуса в блоке глубин}

Положим \(Z_N(c,r):=\M\setminus C_N(c,r)\) и
\(\inf\varnothing:=+\infty\). Из определения
\eqref{eq:component-radius} следует точное выражение
\begin{equation}
 d_N(c,r):=
 \min\left\{r,\inf_{y\in Z_N(c,r)}|y-c|\right\},\qquad
 \rho_N(c,r)=d_N(c,r).
 \label{eq:radius-as-first-lost-distance}
\end{equation}
Действительно, всякий радиус \(\delta<d_N\) допустим в
\eqref{eq:component-radius}, а если \(d_N<\delta<r\), то по свойству
инфимума в замкнутом шаре радиуса \(\delta\) есть точка
\(\M\setminus C_N(c,r)\). Поэтому супремум допустимых радиусов равен
\(d_N\), независимо от того, достигается ли инфимум.

Для блока \([\lambda_m,\lambda_{m+1}]\) обозначим множество точек
\(\M\), впервые потерявших принадлежность выделенной компоненте на
этом блоке, через
\begin{equation}
 \mathcal L_m(c,r):=
 \M\cap\bigl(C_{\lambda_m}(c,r)
                  \setminus C_{\lambda_{m+1}}(c,r)\bigr).
 \label{eq:block-newly-separated-set}
\end{equation}
Так как компоненты \(C_N(c,r)\) вложены, из
\eqref{eq:radius-as-first-lost-distance} получаем
\begin{equation}
 \begin{aligned}
 R_{m+1}(c,r)
   &=\min\{R_m(c,r),\dist(c,\mathcal L_m(c,r))\},\\
 \frac{R_m(c,r)-R_{m+1}(c,r)}{R_m(c,r)}
   &=\frac{\bigl(R_m(c,r)-\dist(c,\mathcal L_m(c,r))\bigr)_+}
           {R_m(c,r)},
 \end{aligned}
 \label{eq:block-radius-loss-exact}
\end{equation}
где \(\dist(c,\varnothing):=+\infty\) и \(x_+:=\max\{x,0\}\).
Поскольку \(R_m(c,r)>0\) для каждого конечного уровня, обозначим эту
относительную потерю через
\[
 \beta_m(c,r):=
 \frac{R_m(c,r)-R_{m+1}(c,r)}{R_m(c,r)}\in[0,1).
\]
Тем самым \(\beta_m(c,r)\) является относительной радиальной потерей,
заданной инфимумом расстояний от \(c\) до новых отделённых точек
\(\mathcal L_m(c,r)\).

Эта точка привязывает потерю к конечному блоку уровней. Для
\(y\in\mathcal L_m(c,r)\) положим
\[
 \nu_m(y):=\min\{n:\lambda_m<n\le\lambda_{m+1},\
                       y\notin C_n(c,r)\}.
\]
Тогда
\[
 y\in C_{\nu_m(y)-1}(c,r)\cap p_{\nu_m(y)}^{-1}(K),
 \qquad y\notin C_{\nu_m(y)}(c,r).
\]
Связная компонента
\(C_{\nu_m(y)-1}(c,r)\) полуалгебраична и потому линейно связна
\cite[теорема 2.4.5]{BCR}. Возьмём в ней путь от \(c\) до \(y\).
Если бы путь целиком лежал в
\(A_{\nu_m(y)}(c,r)\), он соединил бы \(c\) с \(y\) в одной компоненте
этого множества, вопреки выбору \(\nu_m(y)\). Путь выходит из
\(p_{\nu_m(y)}^{-1}(K)\); поскольку его концы отображаются в \(K\),
непрерывность \(p_{\nu_m(y)}\) вынуждает его пересечь
\(\Sigma_{\nu_m(y)}(c,r)\) из~\eqref{eq:green-separator}.
Значит, каждая новая потеря в блоке связана с разрезом некоторого
уровня \(\lambda_m<n\le\lambda_{m+1}\).

Шаг~19 и~\eqref{eq:green-separator-distance} локализуют эти разрезы:
\(\Sigma_n(c,r)\subseteq E_{s_m}\) в рассматриваемом блоке. Ячейка
сетки уровня \(m\), встречающая \(\Sigma_n\), имеет диаметр не более
\(8\cdot2^{-m}\), поэтому вся такая ячейка лежит в
\(\mu_m\)-окрестности \(\M\). Это даёт пространственный масштаб
кандидатных мостов, но само по себе не ограничивает их суммарное
перекрытие и не контролирует, насколько далеко внутрь \(R_m\) может
дотянуться новый разрез. Именно этот переход от близости разрезов к
радиальной потере должен быть количественным.

\subsection*{Шаг 29. Разложить потерю по первому отделяющему уровню}

Для \(n\ge1\) положим
\begin{equation}
 \mathcal L_n(c,r):=
 \M\cap\bigl(C_{n-1}(c,r)\setminus C_n(c,r)\bigr).
 \label{eq:one-step-newly-separated}
\end{equation}
Так как \(\M\subseteq p_n^{-1}(K)\), имеем
\[
 \mathcal L_n(c,r)
 =\M\cap\bigl(A_n(c,r)\setminus C_n(c,r)\bigr).
\]
Множество \(A_n(c,r)\) компактно и полуалгебраично, поэтому имеет
конечное число компонент; все они компактны. Следовательно,
\(\mathcal L_n(c,r)\) есть пересечение \(\M\) с конечным объединением
компактных компонент \(A_n(c,r)\), отличных от корневой, и тоже
компактно.

Обозначим \(Z_n(c,r):=\M\setminus C_n(c,r)\). Вложенность компонент
даёт точное разбиение
\[
 Z_n(c,r)=Z_{n-1}(c,r)\,\dot\cup\,\mathcal L_n(c,r).
\]
По формуле радиуса шага~28 получаем
\begin{equation}
 \rho_n(c,r)=
 \min\{\rho_{n-1}(c,r),\dist(c,\mathcal L_n(c,r))\}.
 \label{eq:one-step-radius-recursion}
\end{equation}
Здесь и далее \(\dist(c,\varnothing)=+\infty\). Положим
\[
 \gamma_n(c,r):=
 \frac{\rho_{n-1}(c,r)-\rho_n(c,r)}{\rho_{n-1}(c,r)}
 \in[0,1).
\]
В каждом конечном уровне \(\rho_{n-1}>0\) по
\eqref{eq:fixed-level-radius}. Для блока глубин
\[
 \mathcal L_m(c,r)
 =\bigsqcup_{\lambda_m<n\le\lambda_{m+1}}
   \mathcal L_n(c,r),\qquad
 \frac{R_{m+1}(c,r)}{R_m(c,r)}
 =\prod_{\lambda_m<n\le\lambda_{m+1}}(1-\gamma_n(c,r)).
\]
Итак, блоковая потеря является точной композицией одношаговых потерь;
это равенство пока не даёт их количественной оценки.

\subsection*{Шаг 30. Построить конечный граф интерфейсов одного среза}

Рассмотрим множества
\[
 A_n(c,r)=C_{n-1}(c,r)\cap\{|p_n|\le2\},\qquad
 U_n(c,r)=C_{n-1}(c,r)\cap\{|p_n|>2\}.
\]
Они покрывают \(C_{n-1}(c,r)\). Полуалгебраичность даёт конечное число
компонент у \(A_n\) и \(U_n\). Обозначим их семейства через
\(\mathcal A_n\) и \(\mathcal U_n\). Каждая компонента из
\(\mathcal A_n\) компактна; замыкание в \(C_{n-1}\) каждой компоненты
\(U\in\mathcal U_n\) компактно и связно.

Определим конечный граф \(\mathsf I_n(c,r)\). Его вершины —
\(\mathcal A_n\) и множества \(\overline U^{\,C_{n-1}}\),
\(U\in\mathcal U_n\); две вершины соединены ребром, когда задаваемые
ими множества пересекаются. Эти связные компактные множества покрывают
связное \(C_{n-1}\), поэтому граф связен: иначе объединения вершин
двух графовых компонент задавали бы непустое разделение \(C_{n-1}\)
на непересекающиеся замкнутые множества.

Компоненты из \(\mathcal A_n\) попарно не пересекаются. Если
\(A\in\mathcal A_n\) пересекает \(\overline U^{\,C_{n-1}}\), точка
пересечения удовлетворяет \(|p_n|=2\): на \(A\) модуль не превосходит
двух, а в \(U\) он строго больше двух. Поэтому каждое ребро между
вершиной \(A\) и вершиной \(\overline U^{\,C_{n-1}}\) имеет непустое
множество контакта, лежащее в \(\Sigma_n(c,r)\). Удаление всех таких
рёбер изолирует вершины \(\mathcal A_n\). В частности, оно отделяет
корневую компоненту \(C_n(c,r)\) от каждой другой компоненты \(A_n\).

\subsection*{Шаг 31. Перенести контакты графа в сетку}

Для \(m\ge2\) и \(\lambda_m<n\le\lambda_{m+1}\) положим
\[
 I_{n,m}:=\{\alpha\in J_m^2:
              Q_{m,\alpha}\cap\Sigma_n(c,r)\ne\varnothing\},
 \qquad
 \mathcal Q_{n,m}:=\{Q_{m,\alpha}:\alpha\in I_{n,m}\}.
\]
Это конечное семейство ячеек. Для любой его ячейки выберем
\(\zeta\in Q_{m,\alpha}\cap\Sigma_n(c,r)\). По шагу~19
\(\dist(\zeta,\M)\le\kappa(s_m)\); поэтому для \(z\in Q_{m,\alpha}\)
\[
 \dist(z,\M)\le |z-\zeta|+\dist(\zeta,\M)
 \le8\cdot2^{-m}+\kappa(s_m)=\mu_m.
\]
Значит,
\begin{equation}
 \bigcup_{\lambda_m<n\le\lambda_{m+1}}
 \ \bigcup_{Q\in\mathcal Q_{n,m}}Q
 \subseteq\{z:\dist(z,\M)\le\mu_m\}.
 \label{eq:block-contact-cell-collar}
\end{equation}
Это точная локализация ячеек контакта в колларе.

\subsection*{Шаг 32. Определить конечную цену графового разреза}

Для ребра \(e=(A,\overline U)\) графа \(\mathsf I_n(c,r)\) обозначим
его множество контакта через
\[
 P_e:=A\cap\overline U^{\,C_{n-1}}\subseteq\Sigma_n(c,r).
\]
Положим
\[
 \mathcal A_n^\M:=
 \{A\in\mathcal A_n:A\ne C_n(c,r),\ A\cap\M\ne\varnothing\}.
\]
Для \(S\subseteq I_{n,m}\) удалим ребро \(e\) только тогда, когда
выбранные ячейки покрывают всё его множество контакта:
\[
P_e\subseteq\bigcup_{\alpha\in S}Q_{m,\alpha}.
\]
Назовём \(S\) допустимым разрезом, если после этого удаления в графе
нет пути от вершины \(C_n(c,r)\) к вершинам из \(\mathcal A_n^\M\).
Все рёбра \(A\)--\(\overline U\) попадают под удаление при
\(S=I_{n,m}\), поэтому допустимые разрезы существуют. Определим
\begin{equation}
 \operatorname{cap}_{n,m}(c,r):=
 \min_{S\text{ допустим}}
 \sum_{\alpha\in S}\operatorname{diam}Q_{m,\alpha}.
 \label{eq:finite-cut-capacity}
\end{equation}
Минимум достигается, поскольку \(I_{n,m}\) конечно. Если
\(\mathcal A_n^\M=\varnothing\), полагаем
\(\operatorname{cap}_{n,m}=0\). Из оценки диаметра сетки следует
\[
 0\le\operatorname{cap}_{n,m}(c,r)
 \le8\cdot2^{-m}|I_{n,m}|.
\]
Каждая точка \(\mathcal L_n(c,r)\) лежит в одной из вершин,
отделяемых таким разрезом.

\subsection*{Шаг 33. Сложить цены внутри блока}

Определим суммарную цену всех одношаговых разрезов блока:
\begin{equation}
 \mathfrak C_m(c,r):=
 \sum_{\lambda_m<n\le\lambda_{m+1}}
 \operatorname{cap}_{n,m}(c,r).
 \label{eq:block-cut-capacity}
\end{equation}
Она конечна и удовлетворяет
\[
 \mathfrak C_m(c,r)
 \le8\cdot2^{-m}
       \sum_{\lambda_m<n\le\lambda_{m+1}}|I_{n,m}|.
\]
Если
\[
 v_{m,\alpha}:=
 \#\{n:\lambda_m<n\le\lambda_{m+1},\
             Q_{m,\alpha}\cap\Sigma_n(c,r)\ne\varnothing\},
\]
то двойной подсчёт даёт точное равенство
\[
 \sum_{\lambda_m<n\le\lambda_{m+1}}|I_{n,m}|
 =\sum_{\alpha\in J_m^2}v_{m,\alpha}.
\]
Тем самым стоимость разреза сведена к числу ячеек-контактов с учётом
их кратности по уровням. Оценки шага~19 ограничивают положение каждой
такой ячейки, но пока не ограничивают \(v_{m,\alpha}\).

\subsection*{Шаг 34. Отделить коллард от радиальной оценки}

Из~\eqref{eq:block-contact-cell-collar} следует
\[
 \sup_{z\in Q}\dist(z,\M)\le\mu_m
 \quad(Q\in\mathcal Q_{n,m}).
\]
Это поперечная оценка расстояния от контактов до множества
\(\M\). Радиальная потеря в~\eqref{eq:block-radius-loss-exact}
определяется величиной \(\dist(c,\mathcal L_m)\), то есть положением
потерянных точек вдоль \(\M\) относительно \(c\). Для перехода между
этими двумя величинами нужны контроль графовой кратности и теорема,
связывающая разрез с расстоянием до потерянного множества.

\subsection*{Шаг 35. Вычислить локальную степень параметрического полинома}

Для фиксированного \(n\) и \(\zeta\in\C\) определим локальную степень
\[
 q_n(\zeta):=\min\{q\ge1:p_n^{(q)}(\zeta)\ne0\}.
\]
Полином \(p_n\) неконстантен, поэтому это число конечно. Разложение
Тейлора имеет вид
\[
 p_n(\zeta+w)-p_n(\zeta)=w^{q_n(\zeta)}h_{n,\zeta}(w),
 \qquad h_{n,\zeta}(0)=\frac{p_n^{(q_n(\zeta))}(\zeta)}
                              {q_n(\zeta)!}\ne0.
\]
Выберем \(\delta_{n,\zeta}>0\) и \(0<a_{n,\zeta}\le
b_{n,\zeta}<\infty\) так, чтобы
\[
 a_{n,\zeta}\le|h_{n,\zeta}(w)|\le b_{n,\zeta}
 \qquad(|w|\le\delta_{n,\zeta}).
\]
Для \(0<t\le a_{n,\zeta}\delta_{n,\zeta}^{q_n(\zeta)}\) получаем
\begin{equation}
 \overline B\!\left(\zeta,
       (t/b_{n,\zeta})^{1/q_n(\zeta)}\right)
 \subseteq
 \{z:|z-\zeta|\le\delta_{n,\zeta},\
       |p_n(z)-p_n(\zeta)|\le t\}
 \subseteq
 \overline B\!\left(\zeta,
       (t/a_{n,\zeta})^{1/q_n(\zeta)}\right).
 \label{eq:local-degree-pullback}
\end{equation}
Это локальная оценка обратного образа в параметрической координате.

\subsection*{Шаг 36. Получить равномерную оценку в регулярной карте}

Пусть \(p_n'(\zeta)\ne0\), \(a:=|p_n'(\zeta)|\), и
\(M_2:=\sup_{\overline B(\zeta,\sigma_0)}|p_n''|\). Возьмём
\[
 0<\sigma\le
 \min\left\{\sigma_0,\frac{a}{2(1+M_2)}\right\}.
\]
Для \(z,w\in\overline B(\zeta,\sigma)\) интегрирование производной
по отрезку \([w,z]\) даёт
\[
 |p_n(z)-p_n(w)-p_n'(\zeta)(z-w)|
 \le M_2\sigma|z-w|\le\frac a2|z-w|.
\]
Отсюда
\[
 |p_n(z)-p_n(w)|\ge\frac a2|z-w|.
\]
Следовательно, \(p_n\) инъективен на этом диске, а обратная ветвь
имеет константу Липшица не больше \(2/a\). Для блока глубин данная
оценка полезна только при контроле снизу \(a\) и сверху \(M_2\),
равномерном по всем рассматриваемым уровням.

\subsection*{Шаг 37. Учесть степень в критической карте}

Если \(q_n(\zeta)=q\ge2\), формула~\eqref{eq:local-degree-pullback}
показывает масштаб обратного образа \(t^{1/q}\). При \(q=2\) он
равен квадратному корню из \(t\). При \(q>2\) отношение
\[
 \frac{t^{1/q}}{\sqrt t}=t^{1/q-1/2}
\]
неограниченно при \(t\downarrow0\). Поэтому квадратичный масштаб
\(\sqrt t\) не даёт верхней оценки для ветвей локальной степени
\(q>2\). Для применения такого масштаба необходимо отдельно
контролировать локальные степени и их суммарные веса.

\subsection*{Шаг 38. Проверить критичность на явном параметре}

Имеем \(p_2(c)=c^2+c\), поэтому \(p_2'(-1/2)=0\) и
\(q_2(-1/2)=2\). Параметр \(-1/2\) принадлежит \(\M\): отображение
\(x\mapsto x^2-\tfrac12\) переводит отрезок
\([-\tfrac12,\tfrac12]\) в себя, и орбита нуля остаётся в этом
отрезке. Следовательно, производная параметрического полинома
\(p_n\) нельзя считать всюду ненулевой даже на \(\M\). Этот пример
показывает необходимость критических карт; он сам по себе не
утверждает, что данная критическая точка участвует в каждом разрезе.

\subsection*{Шаг 39. Сопоставить локальные степени с глобальным счётом}

Для \(\zeta\) с \(p_n'(\zeta)=0\) выполнено
\[
 q_n(\zeta)-1=\operatorname{ord}_{\zeta}(p_n').
\]
Так как \(\deg p_n=2^{n-1}\), теорема о числе нулей полинома даёт
\begin{equation}
 \sum_{p_n'(\zeta)=0}\bigl(q_n(\zeta)-1\bigr)
 =\deg p_n'=2^{n-1}-1.
 \label{eq:total-parameter-critical-multiplicity}
\end{equation}
Это точный глобальный бюджет кратностей. Он растёт с \(n\) и не
локализует критические точки в ячейках разреза; оценку для
\(\mathfrak C_m(c,r)\) из~\eqref{eq:finite-cut-capacity} он пока
не даёт.

\subsection*{Шаг 40. Построить конечные локальные карты фиксированного уровня}

Для фиксированных \(n,c,r\) множество \(\Sigma_n(c,r)\) компактно.
Окрестности из~\eqref{eq:local-degree-pullback} покрывают каждую его
точку; компактность даёт конечное подпокрытие
\[
 \Sigma_n(c,r)\subseteq
 \bigcup_{j=1}^{J_n}U_{n,j}.
\]
Каждой карте соответствуют конечные константы и локальная степень
\(q_{n,j}\). На регулярных картах действует оценка шага~36, на
критических — оценка шага~37. Это доказывает существование конечного
описания на каждом фиксированном уровне. Полученные числа пока зависят
от \(n,c,r\); из компактности для каждого отдельного \(n\) не следует
равномерность по блоку глубин.

\subsection*{Шаг 41. Перевести оценку Грина в образную шкалу}

Локальная оценка~\eqref{eq:local-degree-pullback} относится к
евклидову радиусу \(t\) в образе полинома \(p_n\). В шаге~19 величина
\(g_n\) ограничивает значение параметрической функции Грина на
\(O_{n-1}\), а \(\kappa(g_n)\) ограничивает расстояние от этой
области до \(\M\). Сами по себе эти данные не дают радиуса \(t\) в
\(p_n\)-образе локальной карты: подстановка \(t=g_n\) требует отдельной
оценки, связывающей Green-коллард, образ \(p_n\) и графовые контакты
равномерно по \(n\).

\subsection*{Шаг 42. Сформулировать бюджет кратностей и перекрытий}

\begin{fuzzystep}
Для каждого уровня надо выбрать карты, покрывающие контакты
\(\Sigma_n\), и сопоставить их рёбрам минимального разреза. Локальная
модель предсказывает вклад порядка \(t^{1/q_{n,j}}\) для карты степени
\(q_{n,j}\), если радиус образа \(t\) уже оценён. Кроме того, одну
ячейку могут встречать разрезы многих уровней; точная кратность этой
ситуации записана через \(v_{m,\alpha}\) в шаге~33. Рабочая цель —
получить суммируемую верхнюю оценку суммарного веса карт после
устранения такого перекрытия. Ни конечность графа, ни глобальная
формула~\eqref{eq:total-parameter-critical-multiplicity} сами по себе
такой оценки не дают.
\end{fuzzystep}

\subsection*{Шаг 43. Поставить графово-метрическую задачу}

\begin{fuzzystep}
Требуется оценить расстояние до новых потерянных точек через стоимость
разрезов, а затем оценить эту стоимость по функциям Грина. В терминах
\(\mathfrak C_m\) это распадается на две количественные задачи:
\begin{equation}
 \begin{aligned}
 \bigl(R_m-\dist(c,\mathcal L_m)\bigr)_+
   &\le A_{c,r}\,\mathfrak C_m(c,r),\\
 \mathfrak C_m(c,r)
   &\le B_{c,r}R_m
      \left(\sqrt{\frac{\mu_m}{r}}+
      \sum_{\lambda_m<n\le\lambda_{m+1}}\sqrt{g_n}\right).
 \end{aligned}
 \label{eq:cut-to-radius-and-budget}
\end{equation}
Первая строка переводит комбинаторный разрез в радиальную потерю.
Вторая требует контроля локальных карт, кратностей и перекрытий.
Ни одна из строк пока не следует из шага~19 и шагов~26--42.
\end{fuzzystep}

\subsection*{Шаг 44. Вывести требуемую оценку из двух бюджетов}

Предположим, что обе оценки~\eqref{eq:cut-to-radius-and-budget}
выполнены с константами \(A_{c,r},B_{c,r}\), не зависящими от \(m\).
Выберем в~\eqref{eq:block-radius-loss} константу
\(C_{c,r}:=A_{c,r}B_{c,r}\). Подстановка двух оценок даёт
\begin{equation}
 \bigl(R_m(c,r)-\dist(c,\mathcal L_m(c,r))\bigr)_+
 \le C_{c,r}R_m(c,r)
 \left(\sqrt{\frac{\mu_m}{r}}+
 \sum_{\lambda_m<n\le\lambda_{m+1}}\sqrt{g_n}\right),
 \label{eq:block-bridge-loss}
\end{equation}
где \(C_{c,r}=A_{c,r}B_{c,r}\). По
\eqref{eq:block-radius-loss-exact} это даёт
\[
 \beta_m(c,r)\le C_{c,r}\left(
 \sqrt{\frac{\mu_m}{r}}+
 \sum_{\lambda_m<n\le\lambda_{m+1}}\sqrt{g_n}\right).
\]
Так как сумма по блоку не превосходит хвоста
\(\sum_{n>\lambda_m}\sqrt{g_n}\), правая часть не превосходит
\(\eta_m(c,r)\) шага~19. По суммируемости бюджета существует
\(m_0\ge2\) такое, что \(\eta_m(c,r)\le\tfrac12\) при \(m\ge m_0\).
Следовательно, \eqref{eq:uniform-radius-loss-hypothesis} вытекает из
\eqref{eq:cut-to-radius-and-budget}.
\begin{fuzzystep}
Это доказательство импликации условно. Сами две оценки
\eqref{eq:cut-to-radius-and-budget}, сформулированные в шаге~43,
пока не доказаны.
\end{fuzzystep}

\subsection*{Шаг 45. Проверить суммируемость целевого бюджета}

Для \(\eta_m\), определённой в шаге~19,
из \(\mu_m\le9\cdot2^{-m}\) и оценки
\eqref{eq:tail-critical-loss} следует
\[
\eta_m(c,r)\le C'_{c,r}2^{-m/2}
\]
с некоторой константой \(C'_{c,r}>0\). Поэтому
\(\sum_{m\ge2}\eta_m(c,r)<\infty\), \(\eta_m(c,r)\to0\), и
\(\eta_m(c,r)\le\tfrac12\) при всех достаточно больших \(m\).

\subsection*{Шаг 46. Сохранить положительный радиус при суммируемых потерях}

Пусть \((R_m)\) — положительные числа и
\[
 \beta_m:=\frac{R_m-R_{m+1}}{R_m},\qquad
 0\le\beta_m\le\eta_m\le\tfrac12,\qquad
 \sum_{m\ge m_0}\eta_m<\infty.
\]
Тогда \(R_{m+1}=R_m(1-\beta_m)\), и для \(m>m_0\)
\[
 R_m\ge R_{m_0}\prod_{j=m_0}^{m-1}(1-\eta_j)
 \ge R_{m_0}\exp\!\left(-2\sum_{j=m_0}^{\infty}\eta_j\right)>0,
\]
где использовано \(\log(1-x)\ge-2x\) при \(0\le x\le1/2\).
Применяя это к~\eqref{eq:block-radius-loss-exact} и оценке шага~44,
получаем
\begin{equation}
 \inf_{m\ge m_0}R_m(c,r)>0.
 \label{eq:positive-limiting-radius}
\end{equation}
Это следствие суммируемости; оно не использует локальную связность
\(\M\).

\subsection*{Шаг 47. Перейти от выбранных глубин ко всем уровням}

Последовательность \(\rho_N(c,r)\) невозрастает, а
\(\lambda_m\to\infty\). Поэтому
\[
 \inf_{N\in\N}\rho_N(c,r)
 =\inf_{m\ge2}\rho_{\lambda_m}(c,r)
 =\inf_{m\ge2}R_m(c,r).
\]
Вместе с~\eqref{eq:positive-limiting-radius} это даёт
\begin{equation}
 \inf_{N\in\N}\rho_N(c,r)>0.
 \label{eq:all-level-radius-positive}
\end{equation}
Выберем \(0<\delta<\min\{r,\inf_N\rho_N(c,r)\}\). Для каждого \(N\) из
определения супремума~\eqref{eq:component-radius} существует
допустимый радиус \(\delta_N>\delta\); следовательно,
\[
 \M\cap\overline B(c,\delta)\subseteq C_N(c,r)
 \qquad(N\in\N).
\]

\subsection*{Шаг 48. Завершить условное доказательство факторизации}

Для внутренних точек \(c\in\operatorname{int}\M\) целевое утверждение
установлено в шаге~2. Для \(c\in\partial\M\), если выполнена
\(\mathsf B\) из~\eqref{eq:uniform-radius-loss-hypothesis}, шаги~45--47
дают \(r,\delta\) с
\[
 \M\cap\overline B(c,\delta)\subseteq C_N(c,r)
 \qquad\text{для всех }N.
\]
Предложение~\ref{prop:factorization-compactness-reduction} тогда
применяется к каждому \(N\) и даёт \(L\ge N\), для которого
\[
 O_L\cap\overline B(c,\delta)\subseteq C_N(c,r).
\]
По шагу~2 это равносильно равенству
\(\varphi_{L,N}^{c;\delta,r}(A)=C_N(c,r)\) для всех компонент
\(A\in\pi_0(O_L\cap\overline B(c,\delta))\); значит,
\(\mathsf B\Rightarrow\)\eqref{eq:factorization-target}.

\begin{fuzzystep}
Для завершения исходной теоремы остаётся доказать обе оценки
\eqref{eq:cut-to-radius-and-budget}, либо непосредственно
\eqref{eq:block-bridge-loss}. Существующие в этом черновике оценки
Грина, конечные графы и локальные разложения Тейлора дают точные
входные данные, но единого равномерного контроля их радиального
эффекта пока нет. Следовательно, шаги~45--48 завершают условную цепочку;
строгое доказательство факторизации по-прежнему открыто в этом тексте.
\end{fuzzystep}

\subsection*{Блоковая редукция мостовой оценки: шаги 49--68}

Шаги~49--68 связывают конечную потерю радиуса с геометрией интерфейсов,
локальными картами параметрических полиномов и суммированием по блокам.
Точные конечные утверждения отделены от равномерных оценок, которые
пока остаются открытыми и отмечены синей средой.

\factorizationstep{49}{Найти ближайшую потерю и её первый уровень}
По обозначению шага~27
\(\Delta_m=R_m(c,r)-R_{m+1}(c,r)\ge0\).
По~\eqref{eq:block-radius-loss-exact} множество
\(\mathcal L_m(c,r)\) компактно и при \(\Delta_m>0\) содержит точку
\(y_m\), для которой
\[
 \begin{gathered}
 |y_m-c|=\dist(c,\mathcal L_m(c,r))=R_{m+1}(c,r),\\
 R_{m+1}(c,r)<R_m(c,r).
 \end{gathered}
\]
Обозначим через \(\nu_m(y_m)\) первый индекс блока, на котором
\(y_m\notin C_{\nu_m(y_m)}(c,r)\). Тогда
\(y_m\in C_{\nu_m(y_m)-1}(c,r)\cap\M\) и лежит в M-содержащей
некорневой компоненте среза \(A_{\nu_m(y_m)}\).

\factorizationstep{50}{Провести путь через интерфейс первого выхода}
Множество \(C_{\nu_m(y_m)-1}(c,r)\) связно и полуалгебраично,
поэтому существует путь \(\gamma\) из \(c\) в \(y_m\), целиком лежащий
в нём. Путь не может целиком содержаться в
\(A_{\nu_m(y_m)}=C_{\nu_m(y_m)-1}\cap\{|p_{\nu_m(y_m)}|\le2\}\),
поскольку \(c\) и \(y_m\) лежат в разных компонентах этого среза.
Следовательно, путь выходит в область \(|p_{\nu_m(y_m)}|>2\); по
непрерывности он пересекает интерфейс \(\Sigma_{\nu_m(y_m)}\).

Для \(y\in C_{n-1}(c,r)\cap\M\) положим
\[
 \begin{aligned}
 \mathcal P_n(c,r;y)&:=
   \{\gamma\in C^0([0,1],C_{n-1}):
      \gamma(0)=c,\ \gamma(1)=y\},\\
 H_n(c,r;y)&:=
   \inf_{\gamma\in\mathcal P_n(c,r;y)}
   \max_{t\in[0,1]}|\gamma(t)-c|.
 \end{aligned}
\]
Связность \(C_{n-1}\) обеспечивает непустоту
 \(\mathcal P_n(c,r;y)\).
 Если \(y\notin C_n\), каждый такой путь пересекает \(\Sigma_n\), откуда
\[
 |y-c|\le H_n(c,r;y),\qquad
 \dist(c,\Sigma_n(c,r))\le H_n(c,r;y).
\]
Обе оценки следуют сначала для каждого пути, затем берётся инфимум;
минимизирующий путь не требуется.

\factorizationstep{51}{Сформулировать радиальный мост для потери}
Положим
\[
 \tau_n(c,r;y):=H_n(c,r;y)-|y-c|\ge0,
 \qquad \beta_m:=\frac{\Delta_m}{R_m(c,r)}.
\]
Для ближайшей потери \(y_m\) имеем
\(R_{m+1}=|y_m-c|\). Если всякий путь к \(y_m\) пересекает разделитель,
то есть множество \(S\subset C_{\nu_m-1}\), пересекаемое каждым путём
\(\gamma\in\mathcal P_{\nu_m}(c,r;y_m)\) и не пересекающее
\(B(c,R_m)\), то
\(H_{\nu_m}(c,r;y_m)\ge R_m\); поэтому
\[
 \tau_{\nu_m}(c,r;y_m)\le \eta_m R_m
 \quad\Longrightarrow\quad
 R_{m+1}\ge(1-\eta_m)R_m.
\]
\begin{fuzzystep}
Одновременно построить такие внешние разделители на каждом блоке и
получить для них суммируемую оценку избытка пути пока не удалось.
Без этих двух равномерных оценок минимаксная формула не даёт
\(\beta_m\le\eta_m\).
\end{fuzzystep}

\factorizationstep{52}{Связать граф интерфейсов с ячеечным разрезом}
В конечном графе \(\mathsf I_n(c,r)\) вершины задаются компонентами
\(A_n\) и замыканиями компонент открытой части \(U_n\). Для ребра
\(e=(A,\overline U)\) его контакт
\(P_e=A\cap\overline U^{\,C_{n-1}}\) лежит в \(\Sigma_n\).
Выбранные ячейки удаляют ребро только при полном покрытии \(P_e\);
допустимость разреза означает отделение корневой вершины от всех
M-содержащих некорневых компонент. Полное семейство контактных ячеек
даёт допустимый разрез, а конечность графа и сетки обеспечивает
достижение минимума в~\eqref{eq:finite-cut-capacity}.

Такой разрез действительно блокирует путь в \(C_{n-1}\), обходящий
выбранные ячейки. Иначе прообразы конечного покрытия вершинами графа
на отрезке пути образовали бы связную цепь пересекающихся множеств;
каждый её переход между вершинами \(A\) и \(\overline U\) задавал бы
неудалённое ребро. Сумма минимальных цен по уровням блока есть
\(\mathfrak C_m(c,r)\) из~\eqref{eq:block-cut-capacity}.

\factorizationstep{53}{Отделить внешний разрез от локальной доступности}
Для \(0\le\eta_m<1\) положим \(t=(1-\eta_m)R_m\) и назовём внешними ячейки
\[
 I^{\mathrm{out}}_{n,m}(t):=
 \{\alpha\in I_{n,m}:Q_{m,\alpha}\cap\overline B(c,t)=\varnothing\}.
\]
Если допустимый разрез состоит из таких ячеек, его объединение не
пересекает \(\overline B(c,t)\). Введём условие внутренней доступности
\[
 \mathsf{Acc}_n(t;y):\quad
  \exists\gamma:[0,1]\to C_{n-1},\quad
  \gamma(0)=c,\quad\gamma(1)=y,\qquad
 \gamma([0,1])\subseteq\overline B(c,t).
\]
При наличии такого пути никакой внешний разрез не может отделить
\(c\) от \(y\): путь избегает выбранных ячеек, что противоречит
шагу~52. Значит, внешний допустимый разрез вместе с
\(\mathsf{Acc}_n(t;y)\) исключает потерянную точку \(y\in B(c,t)\).
\begin{fuzzystep}
Равномерное существование внешних разрезов и нужная внутренняя
доступность для всех уровней блока не следуют из конечной графовой
модели; эта совместная оценка остаётся недоказанной.
\end{fuzzystep}

\factorizationstep{54}{Отличить фиксированный уровень от целого блока}
При фиксированном \(n\) срез \(A_n\) имеет конечное число компактных
компонент. Если \(\mathcal A_n^\M\ne\varnothing\), то
\[
 \sigma_n:=\dist\!\left(C_n,
      \bigcup_{A\in\mathcal A_n^\M}A\right)>0,
 \qquad \dist(c,\mathcal L_n)\ge\sigma_n.
\]
Одношаговая рекурсия даёт
\(\rho_n\ge\min\{\rho_{n-1},\sigma_n\}\); для пустого семейства
потерь полагаем \(\sigma_n=+\infty\). На любом конечном блоке минимум
по уровням положителен, если в блоке есть потери.
\begin{fuzzystep}
Эта конечноблоковая положительность не даёт оценки отношения
\(R_{m+1}/R_m\), равномерной по \(m\). Для~\(\mathsf B\) требуется
контроль по целым блокам; разделения компонент на отдельных уровнях
недостаточно.
\end{fuzzystep}

\factorizationstep{55}{Вывести точную рекурсию производной}
Для \(D_n(c):=p_n'(c)\) дифференцирование
\(p_{n+1}(c)=p_n(c)^2+c\) даёт
\[
 D_1(c)=1,\qquad D_{n+1}(c)=2p_n(c)D_n(c)+1.
\]
На \(\M\) имеем \(|p_n(c)|\le2\), поэтому
\(|D_{n+1}(c)|\le4|D_n(c)|+1\). Это лишь верхняя оценка роста:
сложение комплексных слагаемых не даёт равномерной нижней границы
\(|D_n(c)|\), требуемой для семейства обратных карт.

\factorizationstep{56}{Учесть локальные степени и критические параметры}
В регулярной точке локальная степень \(q_n(\zeta)=1\); в критической
она равна порядку отображения \(p_n\) и удовлетворяет
\(\operatorname{ord}_{\zeta}p_n'=q_n(\zeta)-1\). Формула
\eqref{eq:local-degree-pullback} ограничивает радиус прообраза
образного диска размера \(t\) величиной
\((t/a_{n,\zeta})^{1/q_n(\zeta)}\); в регулярной карте шаг~36 даёт
линейную оценку \(2t/a_{n,j}\). Суммарная кратность критических точек
равна
\[
 \sum_{p_n'(\zeta)=0}(q_n(\zeta)-1)=2^{n-1}-1.
\]
Уже \(p_2'(-1/2)=0\) при \(-1/2\in\M\), поэтому глобальную
невырожденность ветвей предполагать нельзя; нужны локальные бюджеты
по степени и константам карт.

\factorizationstep{57}{Покрыть интерфейс локальными картами}
Для фиксированного \(n\) критические точки \(p_n\) на компактном
\(\Sigma_n\) образуют конечное множество. Их можно окружить
критическими картами, а компактную оставшуюся часть интерфейса покрыть
регулярными картами, где \(|p_n'|\) отделена от нуля. Поэтому на каждом
фиксированном уровне существует конечное покрытие, к которому применимы
локальные оценки шага~36 и~\eqref{eq:local-degree-pullback}.

Радиусы карт, коэффициенты \(a_{n,\zeta}\), степени \(q_n(\zeta)\) и
число перекрытий зависят от \(n,c,r\). Конечность покрытий по отдельным
уровням не даёт общего контроля для всего блока.

\factorizationstep{58}{Согласовать Green-высоту с образным радиусом}
По~\eqref{eq:block-radius-scales} и шагу~19 имеем
\(\mu_m=8\cdot2^{-m}+\kappa(s_m)\le9\cdot2^{-m}\); все контактные
ячейки блока лежат в \(\mu_m\)-окрестности \(\M\). Кроме того,
\(g_n=2^{2-n}\log3\), откуда
\[
 \sum_{n>\lambda_m}\sqrt{g_n}\le C_0 2^{-\lambda_m/2}
 \le C_0 2^{-m/2}
\]
с абсолютной константой \(C_0\). Эти величины дают суммируемый
кандидатный бюджет по глубинам.
\begin{fuzzystep}
Переменная \(t\) в~\eqref{eq:local-degree-pullback} есть евклидов радиус
в образе \(p_n\); условие \(G\le g_n\) и оценка
\(\dist(z,\M)\le\kappa(g_n)\) его не определяют. Равномерная оценка,
переносящая Green-высоту в образные радиусы локальных карт, не выведена.
\end{fuzzystep}

\factorizationstep{59}{Подсчитать встречи интерфейсов с сеткой}
Положим
\[
 v_{m,\alpha}:=\#\{n:\lambda_m<n\le\lambda_{m+1},\
                       Q_{m,\alpha}\cap\Sigma_n\ne\varnothing\}.
\]
Перестановка конечных сумм даёт точное равенство и его взвешенный вариант:
\[
 \sum_{\lambda_m<n\le\lambda_{m+1}}|I_{n,m}|
 =\sum_{\alpha\in J_m^2}v_{m,\alpha},\qquad
 \sum_{\lambda_m<n\le\lambda_{m+1}}
  \sum_{\alpha\in I_{n,m}}w_{m,\alpha}
 =\sum_{\alpha\in J_m^2}v_{m,\alpha}w_{m,\alpha}
 \quad(w_{m,\alpha}\ge0).
\]
Отсюда следует лишь грубая оценка
\[
 \mathfrak C_m\le8\cdot2^{-m}\sum_{\alpha\in J_m^2}v_{m,\alpha};
\]
границы \(v_{m,\alpha}\le\lambda_{m+1}-\lambda_m\) и
\(|J_m^2|=4^m\) не дают нужного контроля через \(R_m\).

\factorizationstep{60}{Сравнить ячеечную и реберную цены}
Для ребра \(e\) положим
\[
 w_{n,m}(e):=
 \min\left\{\sum_{\alpha\in S}\operatorname{diam}Q_{m,\alpha}:
 S\subseteq I_{n,m},\
 P_e\subseteq\bigcup_{\alpha\in S}Q_{m,\alpha}\right\}.
\]
Объединение покрытий любого графового
разреза удаляет все его рёбра, поэтому ячеечная цена не превосходит
суммарную цену выбранных реберных покрытий. Обратная оценка требует
контроля кратности: одна ячейка может покрывать контакты многих рёбер.

Теоремы max-flow/min-cut и Менгера дают точную дискретную информацию
о сети и числе реберно-непересекающихся путей. Они сами по себе не
оценивают геометрическую цену ячеек и не ограничивают указанную
кратность.

\factorizationstep{61}{Поставить локальный бюджет меры контактов}
Рассмотрим конечную атомарную меру
\[
 \mathfrak N_m:=\sum_{\alpha\in J_m^2}
 v_{m,\alpha}\operatorname{diam}(Q_{m,\alpha})\,
 \delta_{z_{m,\alpha}},
\]
где \(z_{m,\alpha}\) — центр ячейки. Её поддержка лежит в
\(\mu_m\)-окрестности \(\M\), однако близость поддержки не ограничивает
массу или радиальный потенциал.
\begin{fuzzystep}
Естественная цель — получить локальную оценку вида
\[
 \int\frac{R_m(c,r)}{R_m(c,r)+|z-c|}\,d\mathfrak N_m(z)
 \le C_{c,r}R_m(c,r)2^{-m/2}.
\]
Ни эта оценка, ни сопоставление её левой части с минимальной ценой
корневого разреза пока не доказаны.
\end{fuzzystep}

\factorizationstep{62}{Избежать круговой нормировки на неизвестный радиус}
Полная масса меры равна
\(\sum_{\alpha}e_{m,\alpha}\), где
\(e_{m,\alpha}:=v_{m,\alpha}\operatorname{diam}Q_{m,\alpha}\);
она имеет размерность длины, как и \(\mathfrak C_m\). После деления
на \(R_m\) диаметр ячейки становится не больше
\(8\cdot2^{-m}/R_m\). Из \(\mu_m\to0\) не следует убывание этой
относительной величины, поскольку нижняя граница для \(R_m\) и есть
часть доказываемого утверждения.

\factorizationstep{63}{Выделить недостающее условие упаковки по масштабам}
Сетка измельчается по четверичному дереву, но интерфейсы \(\Sigma_n\)
не вложены друг в друга. Поэтому конечное дерево ячеек и точный двойной подсчёт
шага~59 не дают оценки числа потомков, несущих контакты, с нужным
весом по глубинам.
\begin{fuzzystep}
Для перехода от конечных сумм к бюджету шага~61 нужна равномерная
локальная оценка упаковки контактов в каждом диске \(B(c,t)\),
совместимая с Green-хвостом и не использующая заранее
\(\inf_mR_m>0\). Такой оценки пока нет.
\end{fuzzystep}

\factorizationstep{64}{Отличить путь от его равномерной геометрии}
Связность компактного полуалгебраического множества
\(C_{n-1}(c,r)\) обеспечивает соединяющую полуалгебраическую дугу;
такая дуга спрямляема и имеет конечную длину на каждом фиксированном
уровне. Внутреннее расстояние
\[
 d_{C_{n-1}}(c,y):=
 \inf\{\operatorname{length}(\gamma):
 \gamma\in C^0([0,1],C_{n-1}),\
 \gamma(0)=c,\ \gamma(1)=y,\ \gamma\text{ спрямляема}\}
\]
удовлетворяет для \(y\in\mathcal L_n\)
\[
 d_{C_{n-1}}(c,y)\ge
 \max\{|y-c|,\dist(c,\Sigma_n)\}.
\]
Эти факты не дают
равномерной верхней оценки длины или минимаксного радиуса при
\(n\to\infty\); длина и радиальный обход также являются разными
величинами.

\factorizationstep{65}{Отделить связность \(\M\) от локальной связности}
Для фиксированного \(n\) и
\(x\in\M\cap C_{n-1}\) обозначим через \(\ell_n(x)\in\pi_0(A_n)\)
компоненту \(A_n\), содержащую \(x\). Эта метка локально постоянна:
компоненты \(A_n\)
образуют конечное семейство попарно непересекающихся компактных
множеств. Радиус постоянства зависит от \(n\). Глобальная связность
\(\M\) не влечёт связности \(\M\cap\overline B(c,t)\); для факторизации
нужна устойчивость меток
на одном радиусе для всех глубин блока.
\begin{fuzzystep}
Равномерная по \(n\) локальная устойчивость меток здесь не доказана.
Её доказательство должно следовать из оценок блоков и не может
предполагать заранее целевую окрестность из~\eqref{eq:factorization-target}.
\end{fuzzystep}

\factorizationstep{66}{Перевести графовый разрез в геометрический барьер}
Объединение выбранных ячеек компактно, но может иметь много связных
компонент. Графовая цена учитывает удалённые контакты. Затраты на
соединение компонент выбранного множества в неё не входят; добавочные
ячейки могут увеличить радиальный бюджет. Там, где планарный аргумент
требует связного барьера, необходима отдельная конструкция с
контролируемой связностью и расположением относительно \(c\).
\begin{fuzzystep}
Не выведено существование барьера, который одновременно отделяет
корневую и M-содержащие части, лежит вне нужного диска и имеет цену,
суммируемую по блокам. Сам факт минимальности конечного графового
разреза этого не обеспечивает.
\end{fuzzystep}

\factorizationstep{67}{Собрать локальные веса критических и регулярных карт}
Для выбранных конечных покрытий критическими картами
\(\mathcal J^{\mathrm{crit}}_{n,m}\) и регулярными картами
\(\mathcal J^{\mathrm{reg}}_{n,m}\) определим кандидат
\[
 \mathfrak W_m:=
 \sum_{\lambda_m<n\le\lambda_{m+1}}
 \left(
 \sum_{j\in\mathcal J^{\mathrm{crit}}_{n,m}}
   a_{n,j}^{-1/q_{n,j}}t_{n,j}^{1/q_{n,j}}
 +\sum_{j\in\mathcal J^{\mathrm{reg}}_{n,m}}
   \frac{2t_{n,j}}{a_{n,j}}
 \right),
\]
где \(t_{n,j}\) — радиус образа, \(q_{n,j}\) — локальная степень,
а перекрывающиеся карты учитываются с кратностью.
Положим
\(\mathcal J_{n,m}:=\mathcal J^{\mathrm{crit}}_{n,m}
\sqcup\mathcal J^{\mathrm{reg}}_{n,m}\) и для каждой карты
\(j\in\mathcal J_{n,m}\) зафиксируем центр
\(\zeta_{n,j}\in\Sigma_n(c,r)\). Для критической карты положим
\(a_{n,j}:=a_{n,\zeta_{n,j}}\) из шага~35 и
\(q_{n,j}:=q_n(\zeta_{n,j})\); для регулярной карты положим
\(a_{n,j}:=|p_n'(\zeta_{n,j})|\) и \(q_{n,j}:=1\).
Величина \(t_{n,j}>0\) — выбранный радиус образного диска,
допустимый для соответствующей оценки шага~35 или~36.
Для знака \(\sigma\in\{-,+\}\) положим
\[
\begin{aligned}
\mathfrak W_m^\sigma(\tau_m):=
\sum_{\lambda_m<n\le\lambda_{m+1}}
\Bigg(&
\sum_{\substack{j\in\mathcal J^{\mathrm{crit}}_{n,m}\\
                \zeta_{n,j}\in\Sigma_n^\sigma(\tau_m)}}
  a_{n,j}^{-1/q_{n,j}}t_{n,j}^{1/q_{n,j}}\\
&+
\sum_{\substack{j\in\mathcal J^{\mathrm{reg}}_{n,m}\\
                \zeta_{n,j}\in\Sigma_n^\sigma(\tau_m)}}
  \frac{2t_{n,j}}{a_{n,j}}
\Bigg).
\end{aligned}
\]
Каждый центр лежит ровно в одном из множеств
\(\Sigma_n^-(\tau_m)\), \(\Sigma_n^+(\tau_m)\); следовательно,
\[
\mathfrak W_m=
\mathfrak W_m^-(\tau_m)+\mathfrak W_m^+(\tau_m).
\]
\begin{fuzzystep}
Для доказательства \(\mathfrak W_m\le C_{c,r}R_m2^{-m/2}\) нужны
одновременно Green-to-image оценка шага~58, контроль критических
степеней и равномерная оценка перекрытий. Конечность отдельного
покрытия ни одного из этих блоковых контролей не даёт.
\end{fuzzystep}

\factorizationstep{68}{Свести оставшуюся задачу к трём мостовым оценкам}
Из~\eqref{eq:block-radius-loss} и геометрических оценок шагов~19 и~51
следует \(\eta_m=O_{c,r}(2^{-m/2})\), поэтому
\(\sum_m\eta_m<\infty\). Если бы с константами, не зависящими от \(m\),
были доказаны оценки
\[
 \mathfrak W_m\le C_1R_m2^{-m/2},\qquad
 \mathfrak C_m\le C_2\mathfrak W_m,\qquad
 \Delta_m\le C_3\mathfrak C_m,
\]
то после увеличения \(C_{c,r}\) они дали бы
\(\Delta_m\le R_m\eta_m\). Шаги~45--48 тогда обеспечили бы
положительный предельный радиус и целевую факторизацию.
\begin{fuzzystep}
Все три перехода в этой цепочке остаются недоказанными: контроль
локальных весов, сравнение весов с ценой графового разреза и
радиальная оценка потери через цену разреза. Поэтому доказательство
теоремы в этом черновике по-прежнему не завершено.
\end{fuzzystep}

Обозначим через \(\mathbf{CompHaus}\) категорию компактных
хаусдорфовых пространств и непрерывных отображений. Для компактных

 \operatorname{Hom}(X,Y):=C(X,Y).
\]
Если \(Y\) снабжено метрикой, пространство морфизмов наделим
равномерной метрикой. Пусть \(\mathbf N\) — категория, соответствующая
частично упорядоченному множеству натуральных чисел. Вложения
\(\iota_N:O_N\hookrightarrow\D\) задают обратную систему
\(\mathcal O:\mathbf N^{\mathrm{op}}\to\mathbf{CompHaus}\);
в шаге~1 её предел отождествляется с \(\M\).

\begin{approxthm}[Последовательность эндоморфизмов, фиксирующих уровни]
Существуют \(\lambda=(\lambda_n)_{n\in\N}\in\N^\N\) и непрерывные
отображения \(T_n:\D\to\D\), \(n\in\N\), для которых
\begin{gather}
 \lambda_0=0,\qquad\lambda_1=1,\qquad
 \forall n\in\N,\quad \lambda_n\ge n,\quad
 \lambda_{n+1}>\lambda_n,\label{eq:opening-depths}\\
 T_0=\id_{\D},\qquad T_1=\Pi|_{\D},\qquad
 \forall n\ge1,\quad T_n(\D)\subseteq K,\label{eq:opening-maps}\\
 \forall n\in\N,\quad
 T_n\circ\iota_{\lambda_n}=\iota_{\lambda_n},
 \label{eq:opening-fixing}\\
 \forall n\in\N,\quad
 \sup_{z\in\D}\dist(T_n(z),O_{\lambda_n})
                 \le2^{-n},
 \qquad
 \forall n\ge1,\quad T_n\in\mathcal R_n,
 \label{eq:opening-proximity}\\
 \forall n\in\N,\quad
 \M\subseteq T_n(\D),\qquad
 \forall n\ge2,\quad
 \dH(T_n(\D),\M)\le2^{-n}.
 \label{eq:opening-image-distance}
\end{gather}
\end{approxthm}

\label{app:approximation-proof}

\subsection*{Локальная связность через компактные связные множества малого диаметра}

\begin{theorem}[Критерий локальной связности через компактные связные множества]
\label{thm:small-connected-compacts}
Пусть \(X\) — компактное метрическое пространство с метрикой \(d\).
Тогда \(X\) локально связно тогда и только тогда, когда
\[
 \forall\varepsilon>0\ \exists\delta>0\ \forall x,y\in X,\qquad
 d(x,y)<\delta\ \Rightarrow\
 \exists C\subseteq X:
 \begin{cases}
 C\text{ компактно и связно},\\
 x,y\in C,\\
 \operatorname{diam}(C)<\varepsilon.
 \end{cases}
\]
\end{theorem}
\begin{proof}
Если \(X\) локально связно, то у каждой точки есть связная открытая
окрестность диаметра меньше \(\varepsilon\). Компактность даёт
конечное подпокрытие этих окрестностей, а лемма Лебега — число
\(\delta>0\). Если \(d(x,y)<\delta\), точки \(x,y\) лежат в одной
такой окрестности; её замыкание в \(X\) компактно, связно и имеет
диаметр меньше \(\varepsilon\).

Обратно, пусть \(x\in V\), где \(V\) открыто в \(X\). Выберем
\(\varepsilon>0\) так, чтобы \(B(x,\varepsilon)\cap X\subseteq V\).
Для соответствующего \(\delta\) каждая точка \(y\in B(x,\delta)\cap X\)
лежит вместе с \(x\) в связном множестве \(C\) диаметра меньше
\(\varepsilon\), поэтому \(C\subseteq V\). Значит, компонента \(V\),
содержащая \(x\), содержит шар \(B(x,\delta)\cap X\). Компоненты
открытых множеств открыты, что равносильно локальной связности.
\end{proof}

\begin{proposition}[Критерий локальной связности через функцию \(\omega_\M\)]
\label{prop:omega-criterion}
\[
 \lim_{r\downarrow0}\omega_\M(r)=0
 \quad\Leftrightarrow\quad
 \LC(\M).
\]
\end{proposition}
\begin{proof}
Пусть \(\lim_{r\downarrow0}\omega_\M(r)=0\). Для каждого
\(\varepsilon>0\) существует \(\delta>0\) такое, что
\(\omega_\M(\delta)<\varepsilon\). При \(|x-y|<\delta\) имеем
\[
 \inf_{C\in\mathcal C_\M(x,y)}\operatorname{diam}C
 \le\omega_\M(\delta)<\varepsilon.
\]
Значит, существует \(C\in\mathcal C_\M(x,y)\) с
\(\operatorname{diam}C<\varepsilon\). Теорема
\ref{thm:small-connected-compacts} даёт \(\LC(\M)\).

Обратно, пусть \(\LC(\M)\). Применим теорему
\ref{thm:small-connected-compacts} к \(\varepsilon/2\) и получим
\(\delta>0\). При \(0<r\le\delta\) для каждой пары
\(x,y\in\M\), \(|x-y|<r\), существует
\(C\in\mathcal C_\M(x,y)\) с \(\operatorname{diam}C<\varepsilon/2\).
Следовательно, \(\omega_\M(r)\le\varepsilon/2<\varepsilon\), откуда
\(\lim_{r\downarrow0}\omega_\M(r)=0\).
\end{proof}

\subsection*{Шаг 1. Представить конечные орбитальные ограничения компактными объектами.}

\begin{lemma}[Описание \(\M\) через множества \(O_N\)]
\label{lem:outer-sets}
\begin{gather}
 \M=\{c\in\C:\forall n\in\N,\ |p_n(c)|\le2\},\label{eq:escape}\\
 \M\subseteq O_{N+1}\subseteq O_N\subseteq K\subseteq\D,\qquad
 \bigcap_{N\in\N}O_N=\M,\label{eq:outer-nesting}\\
 \varprojlim_{\mathbf N^{\mathrm{op}}}\mathcal O
       \cong\M.\label{eq:outer-limit}
\end{gather}
Множества \(\M,O_N,K,\D\) компактны и замкнуты, а \(\D\) выпукло.
Переходные морфизмы \(\mathcal O\) — это вложения
\(\iota_{N+1,N}:O_{N+1}\hookrightarrow O_N\).
\end{lemma}
\begin{proof}
Пусть \(q:=|p_n(c)|>2\) и \(|c|\le q\). Индукцией получаем
\[
 |p_{n+j}(c)|\ge q(q-1)^j\qquad(j\in\N).
\]
На индукционном шаге, если \(r:=|p_{n+j}(c)|\ge q\), то
\[
 |p_{n+j+1}(c)|
 \ge r^2-|c|
 \ge r^2-r
 \ge(q-1)r.
\]
Следовательно, модуль орбитальных точек принимает сколь угодно
большие значения. Если \(|c|>2\), берём \(n=1\) и
\(q=|p_1(c)|=|c|\). Если \(|c|\le2\) и некоторый член орбиты имеет
модуль больше двух, берём его в качестве \(q\). Это доказывает одно
направление равенства \eqref{eq:escape}; обратное следует из
определения ограниченности с \(B=2\).

Каждое \(p_j\) непрерывно, поэтому \(p_j^{-1}(K)\) замкнуто.
Следовательно, \(O_N\) замкнуто в компактном \(K\). Из равенств
\(p_0(c)=0\) и \(p_1(c)=c\) следует
\(O_{N+1}\subseteq O_N\subseteq K\). Равенство~\eqref{eq:escape}
даёт \(\M\subseteq O_N\) и
\[
 \bigcap_{N\in\N}O_N
 =K\cap\bigcap_{j\in\N}p_j^{-1}(K)=\M.
\]
Последнее равенство также показывает, что \(\M\) замкнуто в \(K\),
а значит, компактно. Наконец, \(K\subseteq\D\), где \(\D\) —
замкнутый ограниченный выпуклый квадрат. Пусть \(X\) — компактное хаусдорфово пространство и
\(u_N:X\to O_N\) — отображения, удовлетворяющие
\(\forall N,\ \iota_{N+1,N}\circ u_{N+1}=u_N\). Тогда композиции
\(\iota_N\circ u_N:X\to\D\) совпадают. Их общий
образ лежит в \(\bigcap_NO_N=\M\), поэтому они единственным образом
факторизуются через вложение \(\M\hookrightarrow\D\). Это и есть
универсальное свойство в \(\mathbf{CompHaus}\), утверждаемое в
\eqref{eq:outer-limit}.
\end{proof}

\subsection*{Шаг 2. Доказать сходимость в пространстве компактных подмножеств.}

\begin{definition}[Расстояние Хаусдорфа до множества \(\M\)]
\label{def:outer-distance}
Пусть \(\mathcal K(K)\) — пространство непустых компактных
подмножеств \(K\) с метрикой Хаусдорфа.
Для \(N\in\N\) положим
\[
 h_N:=\dH(O_N,\M)
     =\sup_{z\in O_N}\dist(z,\M).
\]
Последовательность \((O_N)\) сходится в \(\mathcal K(K)\) к \(\M\).
\end{definition}

\begin{theorem}[Сходимость расстояния Хаусдорфа]\label{thm:outer-distance}
\[
 0\le h_{N+1}\le h_N,\qquad \lim_{N\to\infty}h_N=0.
\]
Определим
\begin{gather}
 \ell_0:=0,\qquad\ell_1:=1,\qquad
 \ell_n:=\min\{N\in\N:N>\ell_{n-1},\ N\ge n,\ h_N\le2^{-n}\}
 \quad(n\ge2),\label{eq:lower-depths}\\
 \Lambda:=
 \{\lambda\in\N^\N:
   \lambda_0=0,\ \lambda_1=1,\
   \forall n,\ \lambda_n\ge\ell_n,\
   \lambda_{n+1}>\lambda_n\}.
 \label{eq:depth-class}
\end{gather}
Тогда все минимумы в \eqref{eq:lower-depths} существуют,
\(\ell\in\Lambda\), и
\[
 \forall\lambda\in\Lambda\ \forall n\ge2,\qquad
 \lambda_n\ge n,\quad h_{\lambda_n}\le2^{-n}.
\]
Для каждой \(\lambda\in\Lambda\) имеем \(\lambda_n\to\infty\) и
\(\bigcap_{n\in\N}O_{\lambda_n}=\M\). Поэтому предел
переиндексированной обратной системы
\(\mathcal O\circ\lambda^{\mathrm{op}}\) равен \(\M\).
\end{theorem}
\begin{proof}
Из вложенности следует \(h_{N+1}\le h_N\). Обозначим предел через
\(h_\infty\). Для каждого \(N\) выберем \(z_N\in O_N\) так, чтобы
\(\dist(z_N,\M)=h_N\). Если \(h_\infty>0\), компактность \(O_0\)
даёт подпоследовательность \(z_{N_k}\to z_\infty\in O_0\).
Для каждого фиксированного \(m\) при достаточно больших \(k\)
имеем \(N_k\ge m\), а значит, \(z_{N_k}\in O_m\). Замкнутость даёт
\(z_\infty\in\bigcap_mO_m=\M\).  Непрерывность расстояния до непустого множества тогда даёт
\[
 h_{N_k}=\dist(z_{N_k},\M)\longrightarrow0,
\]
что противоречит \(h_{N_k}\to h_\infty>0\). Следовательно,
\(h_N\to0\), и требуемые целые числа \(\ell_n\) можно построить
рекурсивно.
\end{proof}

\subsection*{Равномерный по глубине контроль локальных компонент}

Обозначим через \(\operatorname{Disc}(S)\) дискретную категорию с
множеством объектов \(S\). Для фиксированных \(c\in\M\) и
\(0<\delta<r\) положим
\[
 \mathbf P_L(c,\delta):=
 \operatorname{Disc}\bigl(\pi_0(O_L\cap\overline B(c,\delta))\bigr),
 \qquad
 \mathbf C_N(c,r):=
 \operatorname{Disc}\bigl(\pi_0(O_N\cap\overline B(c,r))\bigr).
\]
При \(L\ge N\) включение
\[
 O_L\cap\overline B(c,\delta)
 \subseteq O_N\cap\overline B(c,r)
\]
переводит каждую компоненту связности левой части в единственную
компоненту правой части и задаёт функтор
\[
 F_{L,N}^{c;\delta,r}:
 \mathbf P_L(c,\delta)\longrightarrow\mathbf C_N(c,r).
\]
Включения при уточнении уровней также задают функторы
\[
 u_{L+1,L}:\mathbf P_{L+1}(c,\delta)\to\mathbf P_L(c,\delta),
\qquad
 v_{N+1,N}:\mathbf C_{N+1}(c,r)\to\mathbf C_N(c,r).
\]
Для \(L\ge N\) они удовлетворяют равенству
\[
 v_{N+1,N}\circ F_{L+1,N+1}^{c;\delta,r}
 =
 F_{L,N}^{c;\delta,r}\circ u_{L+1,L}.
\]
Равенство следует из того, что оба функториальных перехода посылают
компоненту более глубокого уровня в компоненту более грубого уровня,
содержащую её.

Определим функторы реализации
\[
 \mathcal Y_L:\mathbf P_L(c,\delta)\to\mathbf{CompHaus},
 \qquad
 \mathcal X_N:\mathbf C_N(c,r)\to\mathbf{CompHaus}
\]
отображением каждого объекта в соответствующую компактную компоненту;
компоненты компактных множеств замкнуты и потому компактны.
Включения компонент задают естественные преобразования
\[
 \mathcal Y_{L+1}\Rightarrow\mathcal Y_L\circ u_{L+1,L},\qquad
 \mathcal X_{N+1}\Rightarrow\mathcal X_N\circ v_{N+1,N},\qquad
 \mathcal Y_L\Rightarrow\mathcal X_N\circ F_{L,N}^{c;\delta,r}.
\]

Компонента \(C_N(c,r)\), содержащая \(c\), является выделенным объектом
\(\mathbf C_N(c,r)\). Условие \(\mathsf{UC}\) равносильно требованию,
чтобы для выбранных \(r,\delta\) при каждом \(N\) существовал \(L\ge N\),
для которого
\begin{equation}
 \forall A\in\pi_0(O_L\cap\overline B(c,\delta)),\qquad
 F_{L,N}^{c;\delta,r}(A)=C_N(c,r).
 \label{eq:component-functor-factorization}
\end{equation}
Иными словами, функтор от категории более глубоких локальных компонент
должен факторизоваться через полную подкатегорию, состоящую из одного
выделенного объекта. Коммутативность переходных функторов следует из
вложенности множеств; факторизация~\eqref{eq:component-functor-factorization}
есть дополнительное геометрическое утверждение.

\begin{theorem}[Эквивалентность равномерного контроля компонент и локальной связности]
\label{thm:uniform-component-control}
\[
 \mathsf{UC}\Longleftrightarrow\LC(\M).
\]
\end{theorem}
\begin{proof}
Сначала предположим \(\LC(\M)\). Зафиксируем \(c\in\M\) и
\(\varepsilon>0\). Локальная связность даёт связную окрестность \(V\) точки \(c\),
открытую в \(\M\), для которой
\[
 c\in V\subseteq \M\cap B(c,\varepsilon/2).
\]
Так как \(V\) открыто в \(\M\), существует \(d>0\) такое, что
\(\M\cap B(c,d)\subseteq V\). Положим
\[
 \delta:=\min\{d/2,\varepsilon/4\},\qquad r:=3\varepsilon/4,
 \qquad W:=\overline V^{\,\M}.
\]
Тогда \(0<\delta<r<\varepsilon\),
\(\M\cap\overline B(c,\delta)\subseteq V\), а \(W\) компактно и
связно, содержит \(c\) и лежит в
\(\M\cap\overline B(c,\varepsilon/2)\subseteq
\overline B(c,r)\). Поэтому \(W\subseteq C_N(c,r)\) для каждого
\(N\): это связное подмножество \(F_N(c,r)\), содержащее \(c\).
В частности,
\[
 \mathcal B_N(c;r,\delta)\cap\M=\varnothing
 \qquad(N\in\N).
\]
Предложение~\ref{prop:factorization-compactness-reduction} даёт для
каждого \(N\) индекс \(L\ge N\), для которого
\[
 O_L\cap\overline B(c,\delta)\subseteq C_N(c,r).
\]
Значит, выполнено \(\mathsf{UC}\).

Теперь предположим \(\mathsf{UC}\). Зафиксируем \(c\in\M\) и
относительную открытую окрестность \(U\) точки \(c\) в \(\M\).
Выберем \(\varepsilon>0\) так, чтобы
\(\M\cap B(c,\varepsilon)\subseteq U\), и применим
\eqref{eq:uniform-component-control}. Положим
\[
 C:=\bigcap_{N\in\N}C_N(c,r).
\]
Из \(O_{N+1}\subseteq O_N\) следует
\(C_{N+1}(c,r)\subseteq C_N(c,r)\). Множество
\(O_N\cap\overline B(c,r)\) компактно; его компонента \(C_N(c,r)\)
замкнута, поэтому компактна, связна и содержит \(c\). Теорема о
пересечении убывающей последовательности непустых компактных связных
множеств показывает, что \(C\) непусто, компактно и связно.
Кроме того, \(C\subseteq\bigcap_NO_N=\M\) и
\(C\subseteq\overline B(c,r)\).

Пусть \(y\in\M\cap\overline B(c,\delta)\). Для каждого \(N\) условие
\eqref{eq:uniform-component-control} даёт \(L\ge N\) с
\[
 O_L\cap\overline B(c,\delta)\subseteq C_N(c,r).
\]
Так как \(y\in\M\subseteq O_L\), имеем \(y\in C_N(c,r)\). Это верно
для каждого \(N\), следовательно,
\(\M\cap\overline B(c,\delta)\subseteq C\). Итак,
\[
 c\in\M\cap B(c,\delta)\subseteq C
 \subseteq\M\cap\overline B(c,r)
 \subseteq\M\cap B(c,\varepsilon)\subseteq U.
\]
Значит, \(C\) — связная окрестность \(c\), содержащаяся в \(U\).
Поскольку \(c\) и \(U\) произвольны, \(\M\) локально связно.
\end{proof}

Таким образом, реализация выделенных объектов вместе с функториальными
вложениями даёт связное множество
\(\bigcap_N C_N(c,r)\subseteq\M\). Факторизация
\eqref{eq:component-functor-factorization} обеспечивает, что это
множество содержит \(\M\cap\overline B(c,\delta)\), то есть является
окрестностью \(c\) в \(\M\). После этого предложение
\ref{prop:equicontinuous-selection} даёт последовательность отображений
с общим модулем непрерывности.

В частности, \(\mathsf{UC}\) влечёт \(\mathsf{EA}\) по
предложению~\ref{prop:equicontinuous-selection}, а значит, по
предложению~\ref{prop:diagonal-finite-scale} даёт требуемые конечные
оценки осцилляции при сколь угодно больших уровнях. Для конкретных
внешних множеств \(O_N\) завершение маршрута требует геометрической
факторизации~\eqref{eq:component-functor-factorization}; естественность
функторов задаёт согласованность переходов между уровнями.

\subsection*{Шаг 3. Построить внешнее конформное отображение.}

\begin{theorem}[Внешнее конформное отображение и функция Грина]
\label{thm:external-conformal-map}
\(\M\) нетривиально, связно и имеет связное дополнение. В категории
\(\mathbf{Riem}\) римановых поверхностей и бихоломорфизмов существуют
взаимно обратные изоморфизмы
\[
 \Phi:\C\setminus\M\longrightarrow\{w\in\C:|w|>1\},
 \qquad \Psi:=\Phi^{-1},\qquad
 \frac{\Phi(c)}{c}\longrightarrow1\quad(|c|\to\infty).
\]
Для \(k\in\N\) положим
\[
 f_c^{\circ0}:=\id_{\C},\qquad
 f_c^{\circ(k+1)}:=f_c\circ f_c^{\circ k},\qquad
 g_c(z):=\lim_{k\to\infty}2^{-k}
             \log\max\{1,|f_c^{\circ k}(z)|\}.
\]
Определим
\[
 G(c):=
 \begin{cases}
  \log|\Phi(c)|,&c\notin\M,\\
  0,&c\in\M.
 \end{cases}
\]
Функции \(g_c\) и \(G\) удовлетворяют соотношениям
 \begin{gather}
  G\in C(\C,[0,\infty)),\qquad G^{-1}(0)=\M,\qquad
  G(c)-\log|c|\longrightarrow0\quad(|c|\to\infty),\label{eq:green}\\
  g_c\circ f_c=2g_c,\qquad G(c)=g_c(c)=2g_c(0).
 \label{eq:green-normalization}
 \end{gather}
\end{theorem}
\begin{proof}
Связность \(\M\) и существование \(\Phi\) следуют из
\href{https://pi.math.cornell.edu/~hubbard/OrsayEnglish.pdf#page=64}
{теоремы 8.1, с.~64--65} и
\href{https://pi.math.cornell.edu/~hubbard/OrsayEnglish.pdf#page=65}
{следствия 8.3, с.~65}.\par
Непрерывность продолженной динамической функции Грина установлена в
\href{https://pi.math.cornell.edu/~hubbard/OrsayEnglish.pdf#page=62}
{предложении 8.1(b), с.~62}.\par
Нормировка параметрической функции приведена в
\href{https://pi.math.cornell.edu/~hubbard/OrsayEnglish.pdf#page=65}
{следствии 8.4(a), с.~65}.
Функциональное уравнение следует из сдвига индекса в определяющем
пределе.
\end{proof}

\subsection*{Шаг 4. Построить ретракции на подуровни функции Грина.}

\begin{definition}[Линии уровня функции Грина и ретракции]\label{def:green-retractions}
При \(s>0\) положим
\begin{gather}
 E_s:=\{c\in\C:G(c)\le s\},\qquad
 \kappa(s):=\dH(E_s,\M),\qquad
 \Gamma_s(\theta):=\Psi(e^{s+i\theta}),\label{eq:green-levels}\\
 P_s(c):=
 \begin{cases}
  c,&G(c)\le s,\\
  \displaystyle\Psi\left(e^s\frac{\Phi(c)}{|\Phi(c)|}\right),
       &G(c)>s,
 \end{cases}
 \qquad Q_s:=\Pi\circ P_s|_{\D}.
 \label{eq:green-projection}
\end{gather}
Пусть \(e_s:E_s\hookrightarrow\C\) — вложение. Отображение \(P_s\)
единственным образом факторизуется в непрерывное отображение
\(\bar P_s:\C\to E_s\), для которого
\(e_s\circ\bar P_s=P_s\).
\end{definition}

\begin{theorem}[Ретракции на подуровни функции Грина]
\label{thm:green-projection}
При \(0<s<t\) выполняются
\begin{gather}
 E_s\subseteq\operatorname{int}E_t,\qquad
 \bigcap_{s>0}E_s=\M,\qquad
 \lim_{s\downarrow0}\kappa(s)=0,\label{eq:equipotential-distance}\\
 P_s(\C)=E_s,\qquad P_s|_{E_s}=\id_{E_s},\qquad
 \bar P_s\circ e_s=\id_{E_s},\qquad
 P_s\circ P_t=P_{\min\{s,t\}},\label{eq:retraction-composition}\\
 Q_s\in C(\D,\D),\qquad Q_s|_\M=\id_\M,\qquad
 \dH(Q_s(\D),\M)\le\kappa(s).\label{eq:Q-hausdorff-estimate}
\end{gather}
Граница \(E_s\) — регулярная вещественно-аналитическая кривая
Жордана \(\Gamma_s\). Отображение \((s,c)\mapsto P_s(c)\) непрерывно
на \((0,\infty)\times\C\). Для каждого \(s>0\)
\[
 Q_s(\D)\setminus\M\ne\varnothing.
\]
При \(0<u<s<t\) переходные морфизмы
\[
 r_{t,s}:=\bar P_s\circ e_t:E_t\to E_s
\]
удовлетворяют \(r_{t,s}\circ e_{s,t}=\id_{E_s}\) и
\(r_{s,u}\circ r_{t,s}=r_{t,u}\), где \(e_{s,t}:E_s\hookrightarrow E_t\) — вложение. Следовательно,
отображения \(r_{t,s}\) образуют согласованную систему ретракций.
\end{theorem}
\begin{proof}
Собственность \(G\) обеспечивает компактность каждого \(E_s\).
Вложенность следует из определения, а равенство
\(G^{-1}(0)=\M\) даёт \(\bigcap_{s>0}E_s=\M\).
Если \(\limsup_{s\downarrow0}\kappa(s)>0\), выберем
\(s_j\downarrow0\) и \(c_j\in E_{s_j}\), для которых
\(\dist(c_j,\M)\ge\varepsilon\) при некотором \(\varepsilon>0\).
Компактность \(E_{s_1}\) даёт сходящуюся подпоследовательность
с пределом \(c_\infty\in G^{-1}(0)=\M\). Непрерывность расстояния
даёт \(\dist(c_\infty,\M)=0\), тогда как выбранная нижняя граница
даёт \(\dist(c_\infty,\M)\ge\varepsilon\).

Отображение \(\Gamma_s\) инъективно на окружности, причём
\[
 \Gamma_s'(\theta)=ie^{s+i\theta}\Psi'(e^{s+i\theta})\ne0.
\]
Неограниченная компонента дополнения этой кривой равна
\(\Psi(\{|w|>e^s\})=\{G>s\}\). По теореме Жордана её заполненная
внутренняя область совпадает с \(E_s\).

Для \(c\notin\M\) отображение \(P_s\) можно записать в виде
\[
 P_s(c)=
 \Psi\left(e^{\min\{s,G(c)\}}\frac{\Phi(c)}{|\Phi(c)|}\right).
\]
Это доказывает непрерывность вне \(\M\), в том числе при переходе
через \(G=s\). В точке \((s_0,c_0)\), где \(c_0\in\M\),
непрерывность \(G\) даёт окрестность, в которой \(G(c)<s_0/2\).
При \(s\), близком к \(s_0\), отображение \(P_s\) тождественно в
этой окрестности. Полугрупповое тождество следует из замены
потенциала \(G(c)\) на \(\min\{s,G(c)\}\) без изменения внешнего
угла.

Отображение \(\Pi\) — нерастягивающая метрическая проекция на \(K\);
оно фиксирует \(\M\subseteq K\). Поэтому
\[
 \dist(\Pi(z),\M)\le\dist(z,\M).
\]
Так как \(P_s(\D)\subseteq E_s\), а \(Q_s\) фиксирует \(\M\),
получаем \eqref{eq:Q-hausdorff-estimate}.

Наконец, \(\M\) — непустое собственное замкнутое подмножество
связного множества \(K\), поскольку \(2\in K\setminus\M\).
Относительно открытое множество \(K\cap\{G<s\}\) содержит \(\M\).
Так как \(K\) связно, а \(\M\) — его непустое собственное замкнутое
подмножество, это множество также содержит точку из \(K\setminus\M\).
Выберем
\(c\in K\cap\{G<s\}\setminus\M\). Тогда \(Q_s(c)=c\), откуда
\(c\in Q_s(\D)\setminus\M\).
\end{proof}

\subsection*{Шаг 5. Вложить множества \(O_N\) в подуровни функции Грина.}

\begin{lemma}[Вложение \(O_N\) в подуровень функции Грина]
\label{lem:green-depth}
Для каждого \(N\in\N\)
\begin{equation}\label{eq:green-depth-bound}
 \sup_{c\in O_N}G(c)\le2^{1-N}\log3.
\end{equation}
В частности,
\begin{equation}\label{eq:green-depth-condition}
 \begin{gathered}
 2^{1-N}\log3\le s\Rightarrow
  \exists\eta_{N,s}:O_N\to E_s\text{ такая, что}\\
  e_s\circ\eta_{N,s}=j_N^\C,\qquad
  \bar P_s\circ j_N^\C=\eta_{N,s},\\
  Q_s|_{O_N}=\id_{O_N},\qquad h_N\le\kappa(s).
 \end{gathered}
\end{equation}
\noindent
Здесь \(j_N^\C:O_N\hookrightarrow\C\) обозначает вложение в
окружающее пространство.
\end{lemma}
\begin{proof}
При \(|c|\le2\) и \(|z|\le2\) индукцией получаем
\[
 |f_c^{\circ k}(z)|\le3^{2^k}-1.
\]
База индукции — \(k=0\). Если \(A=3^{2^k}\), следующий шаг следует
из \((A-1)^2+2\le A^2-1\). Поэтому для таких \(c,z\)
\(g_c(z)\le\log3\). Если \(c\in O_N\), то \(c\) и \(p_N(c)\)
принадлежат \(K\). Из равенства~\eqref{eq:green-normalization}
получаем
\[
 G(c)=2g_c(0)=2^{1-N}g_c(p_N(c))
                         \le2^{1-N}\log3.
\]
Так получается вложение \(\eta_{N,s}:O_N\to E_s\).
Отображение \(\bar P_s\) удовлетворяет
\(\bar P_s\circ j_N^\C=\eta_{N,s}\). Поскольку \(O_N\subseteq K\),
отображение \(\Pi\) фиксирует \(O_N\), поэтому
\(Q_s|_{O_N}=\id_{O_N}\). Вложение \(O_N\hookrightarrow E_s\)
даёт \(h_N\le\kappa(s)\).
\end{proof}

\subsection*{Шаг 6. Построить последовательность приближённых эндоморфизмов.}

\begin{theorem}[Конструктивные внешние морфизмы]
\label{thm:constructed-external-maps}
Положим
\begin{gather}
 a:=1+\max_{z\in\D}G(z),\qquad j_1:=0,\qquad s_1:=a,\notag\\
 j_n:=\min\{j\in\N:j>j_{n-1},\
                       \kappa(a2^{-j})\le2^{-n}\},\qquad
 s_n:=a2^{-j_n}\quad(n\ge2),\label{eq:height-choice}\\
 \lambda_0:=0,\qquad\lambda_1:=1,\notag\\
 \lambda_n:=\min\{N\in\N:N>\lambda_{n-1},\
                N\ge n,\ 2^{1-N}\log3\le s_n\}\quad(n\ge2),
 \label{eq:external-depths}\\
 T_0:=\id_{\D},\qquad T_1:=\Pi|_{\D}=Q_a,\qquad
 T_n:=Q_{s_n}\quad(n\ge2).\label{eq:external-maps}
\end{gather}
Тогда \(\lambda\in\Lambda\), каждое \(T_n\) является эндоморфизмом
\(\D\) в \(\mathbf{CompHaus}\), и выполняются все утверждения
начальной теоремы.
\end{theorem}
\begin{proof}
Минимумы, задающие \(j_n\), существуют, поскольку
\(\kappa(s)\to0\); следовательно, \(s_n\downarrow0\).
Минимумы, задающие \(\lambda_n\), существуют, так как
\(2^{1-N}\log3\to0\). Лемма~\ref{lem:green-depth} даёт
\begin{equation}
 h_{\lambda_n}\le\kappa(s_n)\le2^{-n}\qquad(n\ge2).
 \label{eq:external-mesh-bound}
\end{equation}
Из рекурсивного определения и \eqref{eq:lower-depths} следует
\(\lambda_n\ge\ell_n\), поэтому \(\lambda\in\Lambda\) и
\(\lambda_n\to\infty\). Последовательность \((O_{\lambda_n})\)
состоит из множеств конечных орбит.

Так как \(G<a\) на \(\D\), имеем \(P_a|_{\D}=\id_{\D}\) и
\(Q_a=\Pi|_{\D}\). На уровнях ноль и один \(O_0=O_1=K\);
тождественное отображение и \(\Pi\) фиксируют \(K\).
При \(n\ge2\) морфизм факторизации
\(\eta_n:O_{\lambda_n}\to E_{s_n}\) из
леммы~\ref{lem:green-depth} удовлетворяет
\(\bar P_n\circ\iota_{\lambda_n}=\eta_n\). Посткомпозиция с
\(\Pi\circ e_{s_n}\) сохраняет тождество на уровне
\(T_n\circ\iota_{\lambda_n}=\iota_{\lambda_n}\).
Для \(z\in\D\)
\[
 \dist(T_n(z),O_{\lambda_n})
 \le\dist(T_n(z),\M)
 \le\kappa(s_n)\le2^{-n}.
\]
При \(n=0\) имеем \(\dist(z,O_0)\le2\sqrt2-2<1\); при \(n=1\)
\(T_1(\D)=O_1=K\). Так как каждый \(T_n\) фиксирует \(\M\), имеем
\(\M\subseteq T_n(\D)\).
Наконец, при \(n\ge2\)
\[
 \dH(T_n(\D),\M)\le\kappa(s_n)\le2^{-n}.
\]
\end{proof}

\begin{proposition}[Сравнение конечных систем]
\label{prop:inverse-system-comparison}
Для \(\lambda,s_n\) из теоремы~\ref{thm:constructed-external-maps} при
\(n\ge2\) положим \(O_n^\lambda:=O_{\lambda_n}\) и \(E_n:=E_{s_n}\).
Вложения задают обратные функторы
\[
 \mathcal O^\lambda,\mathcal E:
 \mathbf N_{\ge2}^{\mathrm{op}}\longrightarrow\mathbf{CompHaus}.
\]
Морфизмы факторизации \(\eta_n:O_n^\lambda\to E_n\) образуют
естественное преобразование и индуцируют тождество на пределах:
\[
 \iota_n^{\mathcal E}\circ\eta_{n+1}
       =\eta_n\circ\iota_n^{\mathcal O},\qquad
 \varprojlim\mathcal O^\lambda\cong\M
       \cong\varprojlim\mathcal E,\qquad
 \varprojlim\eta=\id_\M.
\]
Если \(\bar P_n:=\bar P_{s_n}|_{\D}:\D\to E_n\), то
\[
 r_{n,m}:=\bar P_{s_m}|_{E_n}:E_n\to E_m\qquad(n\le m)
 \]
 задают ковариантный функтор \(\mathcal S\), а \((\bar P_n)\)
 образует естественное преобразование \(\underline{\D}\to\mathcal S\):
\[
 r_{n,n}=\id_{E_n},\qquad
 r_{m,k}\circ r_{n,m}=r_{n,k},\qquad
 r_{n,m}\circ\bar P_n=\bar P_m,\qquad
 r_{n,n+1}\circ\iota_n^{\mathcal E}=\id_{E_{n+1}}.
\]
При \(n\ge2\) эндоморфизмы \(T_n\) из теоремы об аппроксимации имеют вид
\(T_n=\Pi\circ e_{s_n}\circ\bar P_n\).
\end{proposition}
\begin{proof}
Лемма~\ref{lem:green-depth} даёт вложения
\(O_{\lambda_n}\hookrightarrow E_{s_n}\), на которых \(\bar P_{s_n}\)
тождественно. Поэтому \(\eta_n\) являются
вложениями, естественность следует из коммутативности диаграмм
вложений. Из \(\lambda_n\to\infty\) и \(s_n\downarrow0\) следуют
указанные пределы. Индуцированное отображение \(\varprojlim\eta\)
тождественно на каждом \(c\in\M\).

Остальные тождества следуют из
\eqref{eq:retraction-composition}: переходы \(r_{n,m}\) композиционно
совместимы и являются левыми обратными к вложениям \(E_m\hookrightarrow E_n\);
равенство \(r_{n,m}\circ\bar P_n=\bar P_m\) задаёт естественность
\((\bar P_n)\). Последняя формула для \(T_n\) следует из
\eqref{eq:external-maps} и определения \(Q_{s_n}\).
\end{proof}

\subsection*{Внутренние и внешние ячейки}

\begin{equation}
 \label{eq:dyadic-grid-cells}
 \begin{gathered}
 n\ge2,\qquad J_n:=\{0,\ldots,2^n-1\},\qquad J_n^2:=J_n\times J_n,\\
 q_{n,a}:=
 \left[-2+\frac{4a}{2^n},-2+\frac{4(a+1)}{2^n}\right],\\
 Q_{n,(a,b)}:=
 \{x+iy:x\in q_{n,a},\ y\in q_{n,b}\}
 \qquad((a,b)\in J_n^2).
 \end{gathered}
\end{equation}
Каждая ячейка \(Q_{n,\alpha}\) компактна и
\begin{equation}
 \operatorname{diam}Q_{n,\alpha}\le8\cdot2^{-n}.
 \label{eq:dyadic-grid-diameter}
\end{equation}
Действительно, разности вещественных и мнимых частей двух точек одной
ячейки удовлетворяют оценке \(4\cdot2^{-n}\), а
\(|x+iy|\le|x|+|y|\).
Положим
\[
 U_{n,\alpha}:=O_{\lambda_n}\cap E_{s_n}\cap Q_{n,\alpha},
 \qquad
 I_{n,\alpha}:=\M\cap Q_{n,\alpha}.
\]
Здесь \(U_{n,\alpha}\) — внешняя ячейка общего уточнения систем
\((O_{\lambda_n})\) и \((E_{s_n})\), а \(I_{n,\alpha}\) — её внутренняя
часть. Обозначим
\[
 \mathcal A_n^{\mathrm{out}}:=
   \{\alpha\in J_n^2:U_{n,\alpha}\ne\varnothing\},
 \qquad
 \mathcal A_n^{\mathrm{in}}:=
   \{\alpha\in J_n^2:I_{n,\alpha}\ne\varnothing\}.
\]
Каждое \(I_{n,\alpha}\) компактно, а
\(\mathcal A_n^{\mathrm{in}}\subseteq\mathcal A_n^{\mathrm{out}}\).
Рассмотрим конечные дискретные категории
\(\mathbf C_n^{\mathrm{in}}\) и \(\mathbf C_n^{\mathrm{out}}\) с объектами
\(\mathcal A_n^{\mathrm{in}}\) и \(\mathcal A_n^{\mathrm{out}}\),
соответственно. Отображение родителя
\[
 p_n(a',b'):=\left(\left\lfloor\frac{a'}2\right\rfloor,
                       \left\lfloor\frac{b'}2\right\rfloor\right)
 \qquad((a',b')\in J_{n+1}^2)
\]
 задаёт функторы перехода между уровнями: вложения
 \(O_{\lambda_{n+1}}\subseteq O_{\lambda_n}\),
 \(E_{s_{n+1}}\subseteq E_{s_n}\) и
 \(Q_{n+1,\beta}\subseteq Q_{n,p_n(\beta)}\) дают вложения
 \(U_{n+1,\beta}\hookrightarrow U_{n,p_n(\beta)}\) и
 \(I_{n+1,\beta}\hookrightarrow I_{n,p_n(\beta)}\).
 Все \(U_{n,\alpha}\) и \(I_{n,\alpha}\) компактны. Реализации
 \(\mathcal U_n:\mathbf C_n^{\mathrm{out}}\to\mathbf{CompHaus}\) и
 \(\mathcal I_n:\mathbf C_n^{\mathrm{in}}\to\mathbf{CompHaus}\) задаются
 формулами \(\mathcal U_n(\alpha)=U_{n,\alpha}\) и
 \(\mathcal I_n(\alpha)=I_{n,\alpha}\). Вложения между ячейками задают
 естественные преобразования
 \(\mathcal U_{n+1}\Rightarrow\mathcal U_n\circ p_n\) и
 \(\mathcal I_{n+1}\Rightarrow\mathcal I_n\circ p_n\), а включения
 \(I_{n,\alpha}\hookrightarrow U_{n,\alpha}\) задают естественное
 преобразование \(\mathcal I_n\Rightarrow\mathcal U_n\circ j_n\), где
 \(j_n:\mathbf C_n^{\mathrm{in}}\hookrightarrow\mathbf C_n^{\mathrm{out}}\),
 совместимое с переходами между уровнями.

\begin{proposition}[Сближение внутренних и внешних ячеек]
\label{prop:cellwise-inner-outer}
Для каждого \(n\ge2\) выполняются
\[
 \bigcup_{\alpha\in\mathcal A_n^{\mathrm{in}}}I_{n,\alpha}=\M,\qquad
 \bigcup_{\alpha\in\mathcal A_n^{\mathrm{out}}}U_{n,\alpha}=O_{\lambda_n}.
\]
Для \(\alpha\in\mathcal A_n^{\mathrm{in}}\)
\[
 I_{n,\alpha}\subseteq U_{n,\alpha},
 \qquad
 \dH(I_{n,\alpha},U_{n,\alpha})\le8\cdot2^{-n}.
\]
Для \(\alpha\in\mathcal A_n^{\mathrm{out}}\setminus
                    \mathcal A_n^{\mathrm{in}}\)
\[
 \sup_{z\in U_{n,\alpha}}\dist(z,\M)
 \le h_{\lambda_n}\le\kappa(s_n)\le2^{-n}.
\]
В частности,
\[
 \dH\left(
   \bigcup_{\alpha\in\mathcal A_n^{\mathrm{in}}}I_{n,\alpha},
   \bigcup_{\alpha\in\mathcal A_n^{\mathrm{out}}}U_{n,\alpha}
 \right)=h_{\lambda_n}\longrightarrow0.
\]
\end{proposition}
\begin{proof}
Свойства покрытия следуют из покрытия \(\D\) ячейками \(Q_{n,\alpha}\)
и включений \(\M\subseteq O_{\lambda_n}\subseteq E_{s_n}\). Если
\(\alpha\in\mathcal A_n^{\mathrm{in}}\), то
\(I_{n,\alpha}\subseteq U_{n,\alpha}\).
Для \(z\in U_{n,\alpha}\) выберем \(w\in I_{n,\alpha}\). Обе точки
лежат в \(Q_{n,\alpha}\), поэтому
\(|z-w|\le\operatorname{diam}Q_{n,\alpha}\le8\cdot2^{-n}\).
Это даёт оценку Хаусдорфа.

Если \(\alpha\in\mathcal A_n^{\mathrm{out}}\setminus
                   \mathcal A_n^{\mathrm{in}}\), то
\(U_{n,\alpha}\subseteq O_{\lambda_n}\), откуда
\(\sup_{z\in U_{n,\alpha}}\dist(z,\M)\le h_{\lambda_n}\).
Оценки \(h_{\lambda_n}\le\kappa(s_n)\le2^{-n}\) следуют из
леммы~\ref{lem:green-depth} и выбора \(s_n,\lambda_n\).
Объединения ячеек равны \(\M\) и \(O_{\lambda_n}\), соответственно,
а \(\dH(O_{\lambda_n},\M)=h_{\lambda_n}\).
\end{proof}

\begin{theorem}[Предел категорий ячеек]
\label{thm:cell-category-limit}
Пусть
\[
 \mathfrak A^{\mathrm{out}}:=
 \varprojlim(\mathcal A_n^{\mathrm{out}},p_n),\qquad
 \mathfrak A^{\mathrm{in}}:=
 \varprojlim(\mathcal A_n^{\mathrm{in}},p_n),
\]
где конечные множества индексов наделены дискретной топологией.
Тогда \(\mathfrak A^{\mathrm{in}}=\mathfrak A^{\mathrm{out}}\).
Для каждого адреса \(a=(\alpha_n)\) пересечение
\[
 \bigcap_{n\ge2}U_{n,\alpha_n}
\]
состоит из одной точки \(\pi(a)\in\M\). Отображение
\(\pi:\mathfrak A^{\mathrm{out}}\to\M\) непрерывно и сюръективно;
отождествление адресов с одинаковой точкой реализации задаёт
гомеоморфизм
\[
 \mathfrak A^{\mathrm{out}}/{\sim}\ \cong\ \M,
 \qquad a\sim b\Leftrightarrow\pi(a)=\pi(b).
\]
\end{theorem}
\begin{proof}
Для каждого адреса множества \(U_{n,\alpha_n}\) непусты, компактны и
вложены друг в друга. По свойству конечного пересечения компактности
их пересечение непусто; оценки диаметра \(8\cdot2^{-n}\) показывают,
что оно состоит из одной точки. Эта точка принадлежит всем
\(O_{\lambda_n}\), а из \(\lambda_n\to\infty\) следует
\(\bigcap_nO_{\lambda_n}=\M\). Значит, каждый внешний адрес на каждом
уровне принадлежит \(\mathcal A_n^{\mathrm{in}}\), и
\(\mathfrak A^{\mathrm{out}}\subseteq\mathfrak A^{\mathrm{in}}\).
Обратное включение следует из
\(I_{n,\alpha}\subseteq U_{n,\alpha}\).

Для \(c=x+iy\in\M\) положим
\[
 a_n:=\min\left\{\left\lfloor\frac{2^n(x+2)}4\right\rfloor,2^n-1\right\},
 \qquad
 b_n:=\min\left\{\left\lfloor\frac{2^n(y+2)}4\right\rfloor,2^n-1\right\}.
\]
Тогда \((a_n,b_n)\in\mathcal A_n^{\mathrm{in}}\), и деление индексов на два даёт
\(p_n(a_{n+1},b_{n+1})=(a_n,b_n)\). Поэтому эти индексы задают адрес
с точкой реализации \(c\), и \(\pi\) сюръективно. Если два адреса
совпадают на уровне \(n\), их
точки реализации лежат в одной ячейке \(Q_{n,\alpha}\), где
\(\operatorname{diam}Q_{n,\alpha}\le8\cdot2^{-n}\); это доказывает
непрерывность \(\pi\). Пространство
\(\mathfrak A^{\mathrm{out}}\) компактно как обратный предел конечных
дискретных пространств. Непрерывная сюръекция \(\pi\) индуцирует
непрерывную биекцию
\(\mathfrak A^{\mathrm{out}}/{\sim}\to\M\); компактность области и
хаусдорфовость \(\M\) дают гомеоморфизм.
\end{proof}

Предложение~\ref{prop:cellwise-inner-outer} даёт явную оценку
расстояния Хаусдорфа для каждой внутренней ячейки и соответствующей
внешней ячейки. Для отображений \(G_j\in\mathcal R_{m_j}\) условие
\(\mathsf{EA}\) требует общего модуля равностепенной непрерывности;
по предложению~\ref{prop:equicontinuous-selection} его выполнимость
эквивалентна \(\LC(\M)\). Оценки Хаусдорфа описывают множества
\(I_{n,\alpha}\) и \(U_{n,\alpha}\); для диагонального условия
\eqref{eq:diagonal-finite-scale} дополнительно выбираются
\(\delta_i>0\) и для каждого \(k\) доказывается одновременное
выполнение первых \(k\) оценок для некоторого
\(G\in\mathcal R_m\) при сколь угодно больших \(m\).

\subsection*{Шаг 7. Оценить расстояния между отображениями \(Q_s\).}

\begin{lemma}[Равномерная оценка расстояния между \(Q_s\) и \(Q_t\)]
\label{lem:band-motion}
При \(s>0\) определим
\[
 V(s):=\max_{\theta\in\R/2\pi\mathbb Z}
                     e^s|\Psi'(e^{s+i\theta})|.
\]
Тогда \(V\) конечна и непрерывна на \((0,\infty)\). Кривая
\(c_\theta(s):=\Psi(e^{s+i\theta})\) удовлетворяет уравнению
\begin{equation}\label{eq:external-ray-ode}
 \frac{dc_\theta}{ds}
 =\frac{\Phi(c_\theta)}{\Phi'(c_\theta)}
 =\frac{\nabla G(c_\theta)}{|\nabla G(c_\theta)|^2},
 \qquad |\nabla G(c_\theta)|
       =\frac{|\Phi'(c_\theta)|}{|\Phi(c_\theta)|}>0.
\end{equation}
При \(0<s<t\) выполняется
\begin{equation}\label{eq:finite-band-motion}
 \|Q_t-Q_s\|_\infty
 \le\|\bar P_t|_{\D}-\bar P_s|_{\D}\|_{\infty,\C}
 \le\int_s^t V(u)\,du.
\end{equation}
\end{lemma}
\begin{proof}
Максимум в определении \(V(s)\) берётся по компактной окружности,
на которой \(\Psi'\) непрерывна и \(|\Psi'|>0\).
Дифференцирование равенства
\(\Phi(c_\theta(s))=e^{s+i\theta}\) даёт первое равенство
\eqref{eq:external-ray-ode}. Формула для градиента следует из
\(G=\log|\Phi|\) и уравнений Коши--Римана. Если \(G(c)\le s\),
оба отображения фиксируют \(c\). Иначе их значения лежат на одном и
том же внешнем луче между уровнями \(s\) и \(\min\{t,G(c)\}\).
Интегрирование скорости вдоль луча даёт вторую оценку
\eqref{eq:finite-band-motion}; нерастягиваемость \(\Pi\) даёт
первую.
\end{proof}

\begin{proposition}[Расстояния между соседними отображениями \(Q_{s_n}\)]
\label{prop:consecutive-retraction-distances}
Для высот из теоремы~\ref{thm:constructed-external-maps},
\(s_1=a\) и \(s_n\downarrow0\). При \(n\ge2\) положим
\[
 d_n:=
 \|\bar P_n-\iota_n^{\mathcal E}\circ\bar P_{n+1}\|_{\infty,\C}
 =\sup_{z\in\D}|\bar P_n(z)-\bar P_{n+1}(z)|.
\]
Тогда для \(n\ge2\), \(N\ge2\) и \(N\ge1\) соответственно
\begin{align}
 d_n&\le\int_{s_{n+1}}^{s_n}V(u)\,du,\label{eq:retraction-distance-bound}\\
 \sum_{n=2}^{N}d_n
   &\le\int_{s_{N+1}}^{s_2}V(u)\,du,\label{eq:retraction-distance-sum}\\
 \sum_{n=1}^{N}\|Q_{s_{n+1}}-Q_{s_n}\|_\infty
   &\le\int_{s_{N+1}}^aV(u)\,du.
   \label{eq:retraction-motion-sum}
\end{align}
В частности, если \(\int_0^aV(u)\,du<\infty\), обе бесконечные
суммы \(\sum_{n\ge2}d_n\) и
\(\sum_{n\ge1}\|Q_{s_{n+1}}-Q_{s_n}\|_\infty\) конечны.
\end{proposition}
\begin{proof}
При \(t=s_n\), \(s=s_{n+1}\) оценка
\eqref{eq:finite-band-motion} даёт первую границу. Суммирование по
непересекающимся интервалам \([s_{n+1},s_n]\) даёт вторую.
Поскольку \(s_1=a\), то же суммирование для расстояний между \(Q_{s_n}\)
даёт третью. Последнее утверждение следует из сходимости несобственного
интеграла.
\end{proof}

\subsection*{Шаг 8. Построить допустимую последовательность для каждого \(m\).}

\begin{definition}[Допустимые конечные последовательности отображений]
\label{def:finite-sequences}
Определим
\[
 \Delta:=
 \{\delta\in(0,\infty)^\N:
   \forall k,\ \delta_k\le2^{-k},\
   \delta_{k+1}\le\delta_k/2\}.
\]
Для \(\delta\in\Delta\) положим
\[
 \mathcal K_\delta:=
 \left\{F\in C(\D,\D):
   \forall k\ \forall z,w\in\D,\
   |z-w|\le\delta_k\Rightarrow|F(z)-F(w)|\le2^{-k}
 \right\}.
\]
Зафиксируем \(\lambda\in\Lambda\), \(\delta\in\Delta\) и положим
\[
 e_i(F):=\sup_{z\in\D}\dist(F(z),O_{\lambda_i}),\qquad
 \epsilon_i:=2^{-i},\qquad\beta_i:=8\,2^{-i}.
\]
Для \(m\in\N\) определим
\begin{align}
 \mathcal P_m(\lambda,\delta):=
 \left\{(F_0,\ldots,F_m)\in\mathcal K_\delta^{m+1}:
 \begin{gathered}
  F_0=\id_{\D},\quad
  (m\ge1\Rightarrow F_1=\Pi|_{\D}),\\
  \forall i\in\{1,\ldots,m\},\quad F_i(\D)\subseteq K,\\
  \forall i\in\{0,\ldots,m\}\ \forall c\in O_{\lambda_i},\
                             F_i(c)=c,\\
  \forall i\in\{0,\ldots,m\},\quad e_i(F_i)\le\epsilon_i,\\
  \forall i<m,\quad\|F_{i+1}-F_i\|_\infty\le\beta_i
 \end{gathered}\right\},\label{eq:finite-sequences}\\
 \mathsf{UG}_m(\lambda,\delta):={}&
                  [\mathcal P_m(\lambda,\delta)\ne\varnothing].
 \label{eq:ug-m}
\end{align}
Компактность \(\mathcal K_\delta\) и \(\mathcal P_m(\lambda,\delta)\)
следует из следующей леммы.
\end{definition}

\begin{lemma}[Компактность пространств допустимых конечных последовательностей]
\label{lem:compact-sequences}
\(\mathcal K_\delta\) компактно в равномерной топологии, а каждое
\(\mathcal P_m(\lambda,\delta)\) компактно в топологии конечного
произведения.
\end{lemma}
\begin{proof}
Условия на \(\delta_k\) и \(F\) задают общий модуль непрерывности
для \(\mathcal K_\delta\); все значения лежат в компактном \(\D\).
По теореме Арцела--Асколи семейство относительно компактно.
Равномерные пределы сохраняют условия определения, поэтому
\(\mathcal K_\delta\) замкнуто и компактно.

Для \(F,G\in C(\D,\D)\)
\[
 |e_i(F)-e_i(G)|\le\|F-G\|_\infty.
\]
Условия фиксации, на множество значений, на ошибку и на расстояние
между соседними отображениями задают замкнутые подмножества компактного
конечного произведения \(\mathcal K_\delta^{m+1}\). Значит,
\(\mathcal P_m(\lambda,\delta)\) компактно.
\end{proof}

\begin{proposition}[Существование допустимой последовательности при каждом \(m\)]
\label{prop:finite-sequences-exist}
\[
 \forall m\in\N\ \exists\lambda^{[m]}\in\Lambda\
 \exists\delta^{[m]}\in\Delta,\qquad
 \mathsf{UG}_m(\lambda^{[m]},\delta^{[m]}).
\]
\end{proposition}
\begin{proof}
При \(m=0\) возьмём \(\lambda^{[0]}=\lambda\) из теоремы
\ref{thm:constructed-external-maps} и \(F_0=\id_{\D}\). При \(m=1\)
возьмём \(\lambda^{[1]}=\lambda\) и
\((F_0,F_1)=(\id_{\D},\Pi|_{\D})\). В обоих случаях положим
\(\delta_k^{[m]}=2^{-k}\). Тогда \(\delta^{[m]}\in\Delta\), а все
отображения данной последовательности принадлежат
\(\mathcal K_{\delta^{[m]}}\). Имеем
\(e_0(F_0)\le2\sqrt2-2<1\), и при \(m=1\)
\(e_1(F_1)=0\), \(\|F_1-F_0\|_\infty<1\le\beta_0\).
Так выполнены условия определения~\ref{def:finite-sequences}.

Пусть теперь \(m\ge2\), и возьмём \(s_m,\lambda_m\) из теоремы
\ref{thm:constructed-external-maps}.
Положим \(F_0=\id_{\D}\), \(F_1=\Pi|_{\D}\) и
\(F_i=Q_{s_m}\) при \(2\le i\le m\). Положим
\(\lambda^{[m]}_0=0,\lambda^{[m]}_1=1,
\lambda^{[m]}_2=\lambda_m\) и
\[
 \lambda^{[m]}_i:=
 \max\{\lambda^{[m]}_{i-1}+1,\ell_i\}\quad(i\ge3).
\]
Так как \(\lambda\in\Lambda\) и \(\ell_m\ge\ell_2\), имеем
\(\lambda_m\ge\ell_m\ge\ell_2\) и \(\lambda_m\ge m\ge2\).
Рекурсия даёт \(\lambda^{[m]}_i\ge\ell_i\) при \(i\ge3\) и
строгое возрастание; следовательно, \(\lambda^{[m]}\in\Lambda\).
При \(2\le i\le m\) имеем \(\lambda^{[m]}_i\ge\lambda_m\), поэтому
\[
 O_{\lambda^{[m]}_i}\subseteq O_{\lambda_m}\subseteq E_{s_m}.
\]
Значит, \(Q_{s_m}\) фиксирует требуемые внешние множества, а ошибка
на уровне \(i\) ограничена сверху величиной
\(\kappa(s_m)\le2^{-m}\le2^{-i}\).
На нулевом уровне \(e_0(F_0)\le2\sqrt2-2<1\), а на первом
\(e_1(F_1)=0\). Начальные отображения
фиксируют \(O_0=O_1=K\), причём
\[
 \|F_1-F_0\|_\infty<1\le\beta_0,\qquad
 \|F_2-F_1\|_\infty\le\operatorname{diam}(K)=4=\beta_1.
\]
Все расстояния между соседними отображениями после второго равны нулю.
Конечное семейство
\(\{\id_{\D},\Pi|_{\D},Q_{s_m}\}\) равномерно непрерывно; рекурсивный
выбор достаточно малых \(\delta_k^{[m]}>0\), с
\(\delta_k^{[m]}\le2^{-k}\) и
\(\delta_{k+1}^{[m]}\le\delta_k^{[m]}/2\), задаёт
\(\delta^{[m]}\in\Delta\) и включает все \(F_i\) в
\(\mathcal K_{\delta^{[m]}}\).
\end{proof}

\begin{definition}[Последние отображения допустимых последовательностей]
\label{def:last-maps}
Для \(m\ge1\) положим
\[
 \mathcal T_m:=
 \left\{G\in C(\D,K):
  \exists\lambda\in\Lambda\ \exists\delta\in\Delta\
  \exists(F_0,\ldots,F_m)\in\mathcal P_m(\lambda,\delta),\
  G=F_m
 \right\}.
\]
\end{definition}

\begin{lemma}
\label{lem:last-maps-in-R}
\[
 \forall m\ge1,\qquad \mathcal T_m\subseteq\mathcal R_m.
\]
\end{lemma}
\begin{proof}
Пусть \(G\in\mathcal T_m\). По определению существуют
\(\lambda\in\Lambda\), \(\delta\in\Delta\) и
\((F_0,\ldots,F_m)\in\mathcal P_m(\lambda,\delta)\), для которых
\(G=F_m\). Положим \(N=\lambda_m\). Тогда \(N\ge m\),
\(G|_{O_N}=\id_{O_N}\) и
\[
 G(\D)\subseteq K,\qquad
 \sup_{z\in\D}\dist(G(z),O_N)=e_m(F_m)\le2^{-m}.
\]
Следовательно, \(G\in\mathcal R_m\).
\end{proof}

\begin{definition}[Функционал продолжения на следующий индекс]
\label{def:extension-functional}
Зафиксируем \(\lambda\in\Lambda\), \(\delta\in\Delta\), \(m\ge1\) и
\(\sigma=(F_0,\ldots,F_m)\in\mathcal P_m(\lambda,\delta)\). Положим
\[
 \mathcal H_{m+1}:=
 \{F\in\mathcal K_\delta:F(\D)\subseteq K,\
                    F|_{O_{\lambda_{m+1}}}=\id\},
\]
\[
 J_m(F;\sigma):=
 \max\left\{
 \frac{e_{m+1}(F)}{2^{-(m+1)}},
 \frac{\|F-F_m\|_\infty}{8\,2^{-m}}
 \right\},\qquad
 q_m(\sigma):=\inf_{F\in\mathcal H_{m+1}}J_m(F;\sigma).
\]
\[
 \{F\in\mathcal H_{m+1}:J_m(F;\sigma)\le1\}
 =
 \{F\in\mathcal K_\delta:
   (F_0,\ldots,F_m,F)\in\mathcal P_{m+1}(\lambda,\delta)\}.
\]
\end{definition}

\begin{proposition}[Минимум функционала продолжения]
\label{prop:extension-minimum}
Для каждого \(\sigma\in\mathcal P_m(\lambda,\delta)\) минимум
\(q_m(\sigma)\) достигается, и
\[
 \exists F\in\mathcal K_\delta:
 (F_0,\ldots,F_m,F)\in\mathcal P_{m+1}(\lambda,\delta)
 \quad\Leftrightarrow\quad q_m(\sigma)\le1.
\]
Для любых \(\sigma=(F_0,\ldots,F_m)\) и
\(\tau=(G_0,\ldots,G_m)\) из \(\mathcal P_m(\lambda,\delta)\)
\[
 |q_m(\sigma)-q_m(\tau)|
 \le\frac{\|F_m-G_m\|_\infty}{8\,2^{-m}},
 \qquad q_m(\sigma)\le2^{m+3}.
\]
\end{proposition}
\begin{proof}
\(\mathcal H_{m+1}\) замкнуто в компактном \(\mathcal K_\delta\).
Оно непусто: \(\Pi|_{\D}\in\mathcal K_\delta\), отображает \(\D\)
в \(K\) и фиксирует \(O_{\lambda_{m+1}}\subseteq K\).
Функционал \(J_m(\,\cdot\,;\sigma)\) непрерывен, поэтому его минимум
достигается. Из определения \(J_m\) условие \(J_m(F;\sigma)\le1\)
равносильно двум оценкам
\[
 e_{m+1}(F)\le2^{-(m+1)},\qquad
 \|F-F_m\|_\infty\le8\,2^{-m}.
\]
Вместе с условиями \(F\in\mathcal H_{m+1}\) они равносильны
\((F_0,\ldots,F_m,F)\in\mathcal P_{m+1}(\lambda,\delta)\).

Для фиксированного \(F\in\mathcal H_{m+1}\) неравенство треугольника
даёт
\[
 \left|\|F-F_m\|_\infty-\|F-G_m\|_\infty\right|
 \le\|F_m-G_m\|_\infty.
\]
Операция максимума сохраняет эту липшицеву оценку. Взятие минимумов по
общему компактному множеству \(\mathcal H_{m+1}\) даёт указанную оценку
для \(q_m\).

Наконец, \(\Pi(\D)=K\) и \(\operatorname{diam}(K)=4\), поэтому
\(e_{m+1}(\Pi)\le4\) и
\(\|\Pi-F_m\|_\infty\le4\), поскольку \(F_m(\D)\subseteq K\).
Следовательно,
\[
 q_m(\sigma)\le J_m(\Pi;\sigma)
 \le\max\{4\cdot2^{m+1},\,4/(8\,2^{-m})\}=2^{m+3}.
\]
\end{proof}

\begin{proposition}[Эквивалентность локальной связности и \(\mathsf{EA}\)]
\label{prop:equicontinuous-selection}
\[
 \LC(\M)\quad\Leftrightarrow\quad\mathsf{EA},
\]
\[
 \forall\rho,(m_j),(G_j),\qquad
 \mathsf{EAdata}\bigl(\rho,(m_j),(G_j)\bigr)
 \Rightarrow
 \forall\varepsilon>0,\qquad
 \omega_\M\!\left(\frac{\rho(\varepsilon/3)}2\right)
 \le\frac{2\varepsilon}3.
\]
\end{proposition}
\begin{proof}
Пусть выполнено \(\mathsf{EAdata}(\rho,(m_j),(G_j))\). Для каждого
\(j\) выберем \(N_j\ge m_j\), удовлетворяющее определению
\(\mathcal R_{m_j}\). Тогда
\[
 G_j(\D)\subseteq K,\qquad
 G_j|_{O_{N_j}}=\id_{O_{N_j}},\qquad
 \sup_{z\in\D}\dist(G_j(z),O_{N_j})\le2^{-m_j}.
\]
По теореме Арцела--Асколи существует равномерно сходящаяся
подпоследовательность \(G_{j_\nu}\to R\in C(\D,K)\).
Для фиксированного \(k\) при достаточно больших \(\nu\)
\(N_{j_\nu}\ge k\), поэтому \(O_{N_{j_\nu}}\subseteq O_k\), и
\[
 \dist(R(z),O_k)
 \le\|R-G_{j_\nu}\|_\infty+2^{-m_{j_\nu}}
 \longrightarrow0.
\]
Следовательно, \(R(\D)\subseteq\bigcap_kO_k=\M\). Поскольку
\(\M\subseteq O_{N_j}\) и \(G_j|_{O_{N_j}}=\id\), имеем
\(R|_\M=\id_\M\).

Равномерный предел сохраняет оценки модуля:
\[
 |z-w|\le\rho(\eta)\Rightarrow|R(z)-R(w)|\le\eta
 \qquad(\eta>0,\ z,w\in\D).
\]
Положим \(r_\varepsilon:=\rho(\varepsilon/3)/2\). При
\(x,y\in\M\), \(|x-y|<r_\varepsilon\), множество
\[
 C:=R\bigl(\D\cap\overline B(x,r_\varepsilon)\bigr)
\]
компактно и связно, содержит \(x,y\) и лежит в \(\M\). Для каждого
\(z\in\D\cap\overline B(x,r_\varepsilon)\)
\[
 |R(z)-x|=|R(z)-R(x)|\le\varepsilon/3.
\]
Отсюда \(\operatorname{diam}C\le2\varepsilon/3\), и
\[
 \omega_\M(r_\varepsilon)\le2\varepsilon/3<\varepsilon.
\]
Теорема~\ref{thm:small-connected-compacts} даёт \(\LC(\M)\).

Пусть теперь \(\LC(\M)\). По теореме Каратеодори
\(\Psi=\Phi^{-1}\) непрерывно продолжается на \(\{|w|\ge1\}\);
обозначим продолжение через \(\widehat\Psi\)
\cite[теорема 16.6]{Milnor}. Положим
\[
 R(c):=
 \begin{cases}
  c,&c\in\M,\\
  \widehat\Psi\!\left(\Phi(c)/|\Phi(c)|\right),&c\notin\M.
 \end{cases}
\]
Для \(c\notin\M\) положим \(\xi=\Phi(c)/|\Phi(c)|\). При
\(c\to\partial\M\) выполнено \(G(c)\to0\), поэтому
\[
 |R(c)-c|
 =|\widehat\Psi(\xi)-\widehat\Psi(e^{G(c)}\xi)|
 \longrightarrow0.
\]
Следовательно, \(R:\C\to\M\) непрерывно и \(R|_\M=\id_\M\).
На кольце \(A_a=\{w:1\le|w|\le e^a\}\) положим
\[
 \omega_a(r):=
 \sup_{\substack{w,v\in A_a\\|w-v|\le r}}
 |\widehat\Psi(w)-\widehat\Psi(v)|.
\]
Тогда \(\omega_a(r)\to0\) при \(r\downarrow0\). Для \(0<s\le a\) и
\(c\notin\M\) положим
\(\xi=\Phi(c)/|\Phi(c)|\) и \(r=\exp(\min\{s,G(c)\})\). Тогда
\(P_s(c)=\widehat\Psi(r\xi)\), \(R(c)=\widehat\Psi(\xi)\) и
\(|r\xi-\xi|\le e^s-1\). На \(\M\) отображения \(P_s\) и \(R\)
совпадают. Так как \(R(\D)\subseteq\M\subseteq K\) и отображение
\(\Pi\) является 1-липшицевым, для \(z\in\D\)
\[
 |Q_s(z)-R(z)|
 =|\Pi(P_s(z))-\Pi(R(z))|
 \le|P_s(z)-R(z)|
 \le\omega_a(e^s-1).
\]
Следовательно,
\[
 \|Q_s-R|_\D\|_\infty
 \le\omega_a(e^s-1)\longrightarrow0
 \qquad(s\downarrow0).
\]

Положим \(\lambda_0=0,\lambda_1=1,s_1=a\). Для каждого \(n\ge2\)
выберем \(s_n\) и \(\lambda_n\) так, чтобы
\[
 0<s_n<\min\{s_{n-1}/2,2^{-n}\},\qquad
 \kappa(s_n)\le2^{-n},\qquad
 \|Q_{s_n}-R|_\D\|_\infty\le2^{-n},
\]
\[
 \lambda_n>\lambda_{n-1},\qquad
 \lambda_n\ge\ell_n,\qquad
 2^{1-\lambda_n}\log3\le s_n.
\]
Тогда \(\lambda\in\Lambda\) и \(\lambda_n\ge n\) при \(n\ge1\).
Положим
\[
 F_1=\Pi|_\D,\qquad F_n=Q_{s_n}\quad(n\ge2).
\]
Тогда \(F_n\to R|_\D\) равномерно, откуда
\(\{F_n:n\ge1\}\) равностепенно непрерывно. Определим
\[
 \Omega(r):=
 \sup_{n\ge1}\ \sup_{\substack{z,w\in\D\\|z-w|\le r}}
 |F_n(z)-F_n(w)|,\qquad 0<r\le1,
\]
\[
 r_\eta:=\sup\{r\in(0,1]:\Omega(r)\le\eta\},\qquad
 \rho(\eta):=\min\{\eta,r_\eta/2\}\quad(\eta>0).
\]
Равностепенная непрерывность даёт \(r_\eta>0\). Монотонность
\(\Omega\) и определение супремума дают
\[
 0<\rho(\eta)\le\eta,\qquad \Omega(\rho(\eta))\le\eta.
\]
Лемма~\ref{lem:green-depth} даёт \(Q_{s_n}|_{O_{\lambda_n}}=\id\).
При \(n\ge2\)
\[
 \sup_{z\in\D}\dist(F_n(z),O_{\lambda_n})
 \le\sup_{z\in\D}\dist(Q_{s_n}(z),\M)
 \le\kappa(s_n)\le2^{-n}.
\]
Кроме того, \(F_n(\D)\subseteq K\). Отображение \(F_1=\Pi|_\D\)
фиксирует \(O_1=K\), отображает \(\D\) в \(K\) и имеет нулевое
расстояние до \(O_1\). Поэтому \(F_n\in\mathcal R_n\) для всех
\(n\ge1\). При \(m_j=j\) и \(G_j=F_j\) выполнено
\(\mathsf{EAdata}(\rho,(m_j),(G_j))\).
\end{proof}

\section{Другие рассмотренные подходы}
\label{app:other-approaches}

\noindent
В доказательство теоремы об аппроксимации включены функция \(q_m\), компактность
пространств допустимых последовательностей и критерий через
равностепенную непрерывность последовательности отображений
\(G_j\in\mathcal R_{m_j}\) при \(m_j\to\infty\). Ниже перечислены
альтернативные подходы и оценки для дальнейшего усиления доказательства.


\begin{description}
\item[Ретракции \(Q_s\).] Для отображений \(Q_s\) получена оценка расстояния между уровнями; конечность \(\int_0^aV(u)\,du\) обеспечила бы их равномерную сходимость при \(s\downarrow0\).
Этот маршрут сводится к оценке интеграла \(\int_0^aV(u)\,du\), конечность которого обеспечит равномерную сходимость ретракций.

\item[Продолжение последовательности.] Локальные изменения отображений и выпуклая интерполяция дают оценки ошибки и равномерных расстояний при наличии допустимого продолжения, а минимум \(q_m\) точно характеризует добавление следующего отображения.
Для продолжения при каждом \(m\) требуется равномерная оценка; имеющаяся граница \(q_m(\sigma)\le2^{m+3}\) растёт с \(m\) и оставляет этот количественный шаг открытым.

\item[Сумма расстояний.] Рассматривалось условие суммируемости равномерных расстояний между \(G_j\) и \(G_{j+1}\) при \(m_j\to\infty\):
\[
 \sum_{j\ge1}\|G_{j+1}-G_j\|_\infty<\infty.
\]
Равностепенная непрерывность уже даёт равномерно сходящуюся подпоследовательность по теореме Арцела--Асколи; суммируемость расстояний задаёт дополнительный количественный маршрут к равномерной сходимости.

\item[Гомотопические инварианты.] Изучались компоненты связности и высшие гомотопические группы \(\mathcal P_m\) при \(m\to\infty\).
Если при фиксированных \((\lambda,\delta)\) все множества
\(\mathcal P_m\) непусты, компактность соответствующих вложенных
множеств последовательностей даёт совместимую бесконечную последовательность.
При параметрах, зависящих от \(m\), потребовались бы обратная система
и количественные оценки.

\item[Разбиение границы и склейка.]
Рассматривались покрытия \(\partial\M\) компактными множествами с
одноточечными пересечениями, их образы и пределы склеенных комплексов.
Конечные объединения таких множеств имеют топологическую размерность
\(\le1\) (по
\(\dim_{\mathrm{top}}(\partial\M)\le1\) \cite{HurewiczWallman} и
теореме о замкнутых суммах размерности), а \(\M\) содержит
\(B(0,1/4)\). Для построения по этому маршруту задаются отображения,
определяющие отождествления, и оценивается стремление диаметров
склеиваемых множеств к нулю.
\end{description}
