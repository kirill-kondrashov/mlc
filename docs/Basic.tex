\section{Basic Definitions}

\subsection*{Def fc}
\textbf{Description:} The quadratic polynomial $f_c(z) = z^2 + c$.

\textbf{Details:} This is the fundamental map of the quadratic family. The parameter $c$ defines the specific polynomial.
This map is the simplest non-trivial holomorphic dynamical system.

\begin{lstlisting}[language=Lean]
def fc (c : ℂ) : ℂ → ℂ := fun z => z^2 + c
\end{lstlisting}

\subsection*{Def orbit}
\textbf{Description:} The forward orbit of $z_0$ under $f_c$.

\textbf{Details:} The orbit is the sequence of points $z_0, z_1, z_2, \dots$ where $z_{n+1} = f_c(z_n)$.
Formally, \texttt{orbit c z n} is the $n$-th iterate of $f_c$ applied to $z$.

\begin{lstlisting}[language=Lean]
def orbit (c : ℂ) (z0 : ℂ) : ℕ → ℂ := fun n => (Nat.iterate (fc c) n) z0
\end{lstlisting}

\subsection*{Def boundedOrbit}
\textbf{Description:} Boundedness of an orbit.

\textbf{Details:} An orbit is bounded if there exists a real number $M$ such that $|f_c^n(z)| \le M$ for all $n$.
This property determines whether a point belongs to the filled Julia set.

\end{lstlisting}

\subsection*{Def K}
\textbf{Description:} Filled Julia set.

\textbf{Details:} The filled Julia set $K(c)$ consists of all initial points $z$ whose orbit under $f_c$ remains bounded.
This is a compact subset of the complex plane.

\begin{lstlisting}[language=Lean]
def K (c : ℂ) : Set ℂ := { z | boundedOrbit c z }
\end{lstlisting}

\subsection*{Def J}
\textbf{Description:} Julia set as the topological boundary of $K(c)$.

\textbf{Details:} The Julia set $J(c)$ is the common boundary of the filled Julia set $K(c)$ and its complement (the basin of infinity).
It is the locus of chaotic dynamics.

\begin{lstlisting}[language=Lean]
def J (c : ℂ) : Set ℂ := frontier (K c)
\end{lstlisting}

\subsection*{Def MandelbrotSet}
\textbf{Description:} The Mandelbrot set.

\textbf{Details:} The Mandelbrot set $\mathcal{M}$ is the set of parameters $c$ for which the orbit of the critical point $0$ is bounded.
If $c \in \mathcal{M}$, the filled Julia set $K(c)$ is connected. If $c \notin \mathcal{M}$, $K(c)$ is a Cantor set.

\begin{lstlisting}[language=Lean]
def MandelbrotSet : Set ℂ := { c | boundedOrbit c 0 }
\end{lstlisting}

\subsection*{Def R}
\textbf{Description:} A convenient escape radius depending on $|c|$.

\textbf{Details:} $R(c) = \max(2, 1 + |c|)$ is chosen large enough so that if $|z| > R(c)$, the orbit of $z$ necessarily escapes to infinity.
See the Escape Lemma for the proof of this property.

\begin{lstlisting}[language=Lean]
def R (c : ℂ) : ℝ := max 2 (1 + ‖c‖)
\end{lstlisting}

\subsection*{Axiom mandelbrot\_set\_connected}
\textbf{Description:} The Mandelbrot set is connected.

\textbf{Details:} This is a fundamental theorem by Douady and Hubbard. It states that the set of parameters $c$ for which $0$ has a bounded orbit is connected.
Source: [Douady and Hubbard, Etude dynamique des polynômes complexes, 1984, Theorem 1, Chapter VIII] <https://pi.math.cornell.edu/\textasciitilde{}hubbard/OrsayEnglish.pdf> (p. 47)

\begin{lstlisting}[language=Lean]
axiom mandelbrot_set_connected : IsConnected MandelbrotSet
\end{lstlisting}

\subsection*{Axiom filled\_julia\_set\_connected}
\textbf{Description:} The filled Julia set is connected if $c$ is in the Mandelbrot set.

\textbf{Details:} If the critical point $0$ has a bounded orbit, then the filled Julia set $K(c)$ is connected.
Otherwise, $K(c)$ is a Cantor set (totally disconnected).
Source: [Douady and Hubbard, Etude dynamique des polynômes complexes, 1984, Proposition 1, Chapter VIII] <https://pi.math.cornell.edu/\textasciitilde{}hubbard/OrsayEnglish.pdf> (p. 47)

\begin{lstlisting}[language=Lean]
axiom filled_julia_set_connected {c : ℂ} (h : c ∈ MandelbrotSet) : IsConnected (K c)
\end{lstlisting}

\subsection*{Opaque modulus}
\textbf{Description:} The conformal modulus of a set in the complex plane.

\textbf{Details:} For an annulus $A = \{z \mid r < |z| < R\}$, the modulus is defined as $\frac{1}{2\pi} \ln(R/r)$.
The modulus is a conformal invariant: it is preserved under biholomorphic maps.
For general doubly connected domains, it is defined via conformal equivalence to a standard annulus.
We extend the definition to arbitrary sets (returning 0 for degenerate sets).

\begin{lstlisting}[language=Lean]
opaque modulus (S : Set ℂ) : ℝ
\end{lstlisting}

\subsection*{Def Annulus}
\textbf{Description:} In complex analysis an annulus $ann(a; r, R)$ in the complex plane is an open region defined as $r<|z-a|<R$.

\textbf{Details:} If $r=0$, the region is known as the punctured disk (a disk with a point hole in the center) of radius $R$ around the point $a$.
Wikipedia: https://en.wikipedia.org/wiki/Annulus\_(mathematics)\#:\textasciitilde{}:text=In\%20complex\%20analysis\%20an\%20annulus,may\%20take\%20inside\%20an\%20annulus.

\begin{lstlisting}[language=Lean]
def Annulus (a : ℂ) (r R : ℝ) : Set ℂ := { z | r < ‖z - a‖ ∧ ‖z - a‖ < R } -- open region between circles of radii r and R
\end{lstlisting}

\subsection*{Axiom modulus\_empty}
\textbf{Description:} The modulus of the empty set is 0.

\textbf{Details:} This is a basic property of the modulus. An empty set has no "width".

\begin{lstlisting}[language=Lean]
axiom modulus_empty : modulus ∅ = 0
\end{lstlisting}

\subsection*{Axiom modulus\_nonneg}
\textbf{Description:} Modulus is non-negative.

\textbf{Details:} The modulus measures the "thickness" of an annulus, which must be non-negative.

\begin{lstlisting}[language=Lean]
axiom modulus_nonneg (S : Set ℂ) : 0 ≤ modulus S
\end{lstlisting}

\subsection*{Axiom groetzsch\_inequality\_axiom}
\textbf{Description:} Axiom: Superadditivity of modulus for disjoint essential annuli (Grötzsch Inequality).

\textbf{Details:} If $A$ and $B$ are disjoint annuli nested within $S$, then the sum of their moduli is less than or equal to the modulus of $S$.
This inequality is crucial for proving convergence of moduli sums in the Yoccoz puzzle analysis.
Reference: Milnor, Dynamics in One Complex Variable, Corollary B.5

\begin{lstlisting}[language=Lean]
axiom groetzsch_inequality_axiom {A B S : Set ℂ}
  (h_disj : Disjoint A B) (h_sub : A ∪ B ⊆ S) :
  modulus A + modulus B ≤ modulus S
\end{lstlisting}
