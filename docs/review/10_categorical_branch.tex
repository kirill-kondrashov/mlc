\chapter{The categorical branch and the conditional MLC roots}

\section{What category theory changes}

The categorical development does not replace the analytic problem by a
formal trick.  It performs two more limited tasks:
\begin{enumerate}
\item it packages subsets of the parameter plane and their inclusions as
      objects and morphisms in a topological over-category;
\item it isolates the exact parameter--dynamical existence statement whose
      connected-image consequence is formal.
\end{enumerate}
The first task is elementary category theory in \texttt{TopCat}.  The second
task is still complex dynamics and is not supplied by the category
interfaces.

\section{The over-category of parameter approximations}

\begin{definition}[Approximation over the parameter plane]
\label{def:categorical-approximation}
Let
\[
 \mathsf{Plane}:=\operatorname{TopCat.of}\C.
\]
An approximation is an object of the over-category
\[
 \mathsf{Approx}:=\mathsf{TopCat}/\mathsf{Plane}.
\]
For $S\subseteq\C$, the canonical approximation is
\[
 \mathsf{ofSet}(S):=
 \bigl(\operatorname{TopCat.of}S\longrightarrow\mathsf{Plane}\bigr).
\]
\end{definition}

The implementation is:
\begin{itemize}
\item \lean{Mlc.CategoricalTopologicalApproximation.Approximation};
\item \lean{Mlc.CategoricalTopologicalApproximation.ofSet};
\item \lean{Mlc.CategoricalTopologicalApproximation.setInclusion}.
\end{itemize}
The underlying parameter subset of an arbitrary approximation is its range:
\[
 \operatorname{image}(A):=
 \operatorname{range}(A\to\mathsf{Plane}).
\]

\begin{theorem}[Images and categorical intersections]
\label{thm:categorical-image-intersection}
For approximations $A,B$,
\[
 \operatorname{image}(A\times_{\mathsf{Plane}}B)
 =
 \operatorname{image}(A)\cap\operatorname{image}(B).
\]
For subsets $S,T\subseteq\C$,
\[
 \mathsf{ofSet}(S)\longrightarrow\mathsf{ofSet}(T)
\quad\Longleftrightarrow\quad S\subseteq T.
\]
\end{theorem}

These are theorems
\lean{Categorical.image_intersection} and
\lean{Categorical.factorsThrough_ofSet_iff}.  They are important because
they show that the categorical pullback is not an abstract replacement for
set intersection: its image is exactly the ordinary set intersection.

\section{Orbit-bound and finite-orbit towers}

\begin{definition}[Outer and inner orbit approximations]
\label{def:categorical-orbit-tower}
For $N\in\N$ define
\[
 I_N:=\left\{c:\forall n,\ |p_n(c)|\le N\right\},
\qquad
 O_N:=\left\{c:|c|\le2,\ \forall n\le N,\ |p_n(c)|\le2\right\}.
\]
The inner sets are increasing and the outer sets are decreasing.
\end{definition}

The exact Lean declarations are
\lean{Categorical.Mandelbrot.innerOrbitSet} and
\lean{Categorical.Mandelbrot.outerOrbitSet}.  The following statements are
proved:
\begin{align*}
 I_N&\subseteq\M,
 &\bigcup_N I_N&=\M,\\
 \M&\subseteq O_N,
 &\bigcap_N O_N&=\M.
\end{align*}
The corresponding theorem names are
\begin{center}
\lean{Categorical.Mandelbrot.iUnion_innerOrbitSet_eq_set},
\qquad
\lean{Categorical.Mandelbrot.iInter_outerOrbitSet_eq_set}.
\end{center}

\begin{remark}
The inner definition contains infinitely many orbit inequalities.
Indeed, $I_N=\M$ for every $N\ge2$, by the universal critical-orbit
bound.  This is
\leanref{Mlc/CategoricalMandelbrot.lean}{innerOrbitSet_eq_set_of_two_le};
it must be distinguished from finite trapping certificates.

The outer and inner limits agree as sets for different reasons.  This
agreement does not say that the finite stages are connected, and it does not
imply local connectivity of the limit.  The missing information is
component control, represented in the repository by the buffer and
finite-component interfaces.
\end{remark}

\section{The categorical root}

\begin{definition}[Root input]
\label{def:categorical-root-input}
The current root input is the proposition
\[
\begin{gathered}
\mathsf{BUF}(\M,O):\quad
\forall c\in\M\ \forall\varepsilon>0\
\exists\,0<\delta<r<\varepsilon,\\
\forall N\ \exists L\ge N,\qquad
O_L\cap\overline B(c,\delta)
\subseteq
\operatorname{Comp}_c(O_N\cap\overline B(c,r)).
\end{gathered}
\]
\end{definition}

Its Lean packaging is
\lean{MLC.RootInput}; the field is
\lean{ParameterComponent.MandelbrotUniformOuterBuffer}.
The topological theorem
\lean{ParameterComponent.locallyConnectedSpace_of_uniformOuterBuffer}
proves
\[
 \mathsf{BUF}(\M,O)\Longrightarrow
 \operatorname{LocallyConnectedSpace}(\M).
\]

\begin{theorem}[Categorical compatibility root]
\label{thm:categorical-compatibility-root}
Let
\[
 \mathsf{MLC}_{\mathrm{cat}}
 :=
 \operatorname{LocallyConnectedSpace}
   \bigl(\operatorname{TopCat.of}\M\bigr).
\]
Then
\[
 \mathsf{MLC}_{\mathrm{cat}}
 \Longleftrightarrow
 \operatorname{LocallyConnectedSpace}(\M),
\]
and $\mathsf{BUF}(\M,O)$ implies both sides.
\end{theorem}

The exact declarations are:

\item \lean{Categorical.mlcConjecture_iff_set};
\item \lean{Categorical.categorical_mlc_conjecture};
\item \lean{MLC.mlc_conjecture}.
\end{itemize}
The literature-facing wrappers are:

\end{itemize}

\section{Finite covers and uniform geometric towers}

The finite-component branch replaces the pointwise buffer by finite
certificate data.  At scale $k$, a certificate supplies finitely many
rational parameter boxes and a stage $T$ covering $O_T$.  Each local box
comes with radii and a component-control implication.  The theorem
\[
 \mathsf{FC}\Longrightarrow\mathsf{BUF}(\M,O)
\Longrightarrow\mathsf{MLC}(\M)
\]
is implemented by
\lean{FiniteComponent.uniformOuterBuffer_of_finiteComponentCriterion}
and
\lean{FiniteComponent.rootInput_of_finiteComponentCriterion}.

The independent uniform-geometric branch uses the compact convex
square $D=[-2,2]^2\supseteq\M$ and continuous maps
\[
 \begin{gathered}
 N_n\ge n,\qquad R_n:D\longrightarrow D,\qquad
 R_n|_{O_{N_n}}=\id,\\
 \forall x\in D\quad\exists y\in O_{N_n},\qquad
 |R_n(x)-y|\le2^{-n},\\
 \|R_{n+1}-R_n\|_\infty\le8\cdot2^{-n}.
 \end{gathered}
\]
The summable adjacent-map bound gives a continuous uniform limit
$R$, with $\|R_n-R\|_\infty\le16\cdot2^{-n}$.
The proximity field places $R(x)$ in every closed outer stage,
because $N_n\ge n$ and the proximity error tends to zero.
The fixing field gives $R|_{\M}=\id$.
Thus the limit is a retraction $D\to\M$.  The proved implication is
\[
 \mathsf{UG}\Longrightarrow
 \text{continuous retraction }D\to\M
\Longrightarrow\mathsf{MLC}.
\]
The root theorem is
\lean{UniformGeometry.mandelbrot_locallyConnected_of_squareTower}.
Its tower existence is not asserted.

\begin{remark}
The categorical statement
\lean{UniformGeometry.categorical_mlc_of_squareTower}
is a packaging of the same set-level conclusion in \texttt{TopCat}; it is
not evidence that the tower exists.  Likewise,
\lean{FlowInterfaces.CoherentParametrization} and
\lean{FlowInterfaces.TerminalExtension} are explicit construction
obligations, not project axioms.
\end{remark}

\section{The categorical Green-sublevel branch}

For $c\in\M$ and $n\in\N$ define the frozen translate
\[
 U_n(c):=\{c':G_c(c'-c)<2^{-n}\},
\qquad
 V_n(c):=U_n(c)\cap\M.
\]
The un-intersected set $U_n(c)$ is connected.  The current code proves
\[
 \bigcap_n U_n(c)=(c+K_c),
\]
not $\{c\}$.  This is the theorem
\lean{Quadratic.iInter_green_sublevel_translate_eq_translate_filledJulia}.

The desired straddling statement would be
\[
 U_n(c)\not\subseteq\M
\Longrightarrow V_n(c)\text{ is connected}.
\]
Its categorical form is
\lean{GreenSublevelIntersectionCategoricalData}.
The equivalence with the set-level formulation is proved by
\lean{greenSublevelIntersectionCategoricalData_iff}.

\begin{theorem}[Frozen full-sublevel obstruction]
\label{thm:frozen-sublevel-obstruction}
The universal frozen Green-sublevel intersection statement is false.  There
are $c\in\M$ and $n$ for which $V_n(c)$ is disconnected.
\end{theorem}

The machine-checked witness is $c=i$, $n=16$.  The theorem is
\lean{not_greenSublevelIntersectionCategoricalData}; its set-level source is
\lean{GreenCounterexample.not_greenSublevelIntersectionSetData}.
Consequently, the full Green-sublevel categorical datum cannot be cited as
the classical Yoccoz parapuzzle theorem.

\section{Carving, Efimov-style interfaces, and the literature}

\begin{definition}[Categorical carving datum]
\label{def:categorical-carving}
A carving datum at $(c,n)$ is a surjective continuous morphism
\[
 \operatorname{TopCat.of}U_n(c)\longrightarrow
 \operatorname{TopCat.of}V_n(c).
\]
Since the source is connected, its image is connected.
\end{definition}

The exact native theorem is
\lean{Efimov.isConnected_of_topCatSurjectiveMorphism}.
The stronger analytic input
\lean{Efimov.SpaceHolomorphicCarvingData} supplies a differentiable map
with exact image $V_n(c)$ and is converted to
\lean{Efimov.DouadyHubbardYoccozCategoricalCarvingData}.
The existence proposition is
\lean{Efimov.DouadyHubbardYoccozCategoricalTheorem}.
Its universal version is refuted: it would imply the false
Green-intersection statement.  Likewise a universal family of the
stronger \lean{SpaceHolomorphicCarvingData} would contradict the
checked $c=i$, $n=16$ obstruction.  Individual carving data and
the connected-image implication remain meaningful; their universal
frozen realization is impossible.

The Efimov-inspired part is intentionally limited by available foundations:
\begin{itemize}
\item \lean{Efimov.DualizablePacmanTower} records rigid monoidal levels and
      adjoint transitions;
\item \lean{Efimov.StrongMittagLefflerData} records stabilization clauses;
\item \lean{Efimov.PacmanKTheory} records a graded additive shadow, not
      algebraic $K$-theory;
\item \lean{Efimov.PacmanRealization} records a realization into the
      topological approximation category.
\end{itemize}
Mathlib does not provide the stable infinity-categorical and continuous
$K$-theoretic machinery used in Efimov's papers.  Therefore the category
interfaces organize a future realization theorem but do not prove the
missing parameter carving.

\subsection{Literature of the categorical tools}
\label{sec:efimov-literature}

The categorical interfaces above use five papers of A.~I.~Efimov
\cite{EfimovContinuousK,EfimovInverseLimits,EfimovDevissage,EfimovRigidityMot,EfimovHeartKH}.
A case-insensitive search of their TeX sources for \emph{Mandelbrot}, \emph{Julia},
\emph{Green}, \emph{holomorphic motion}, \emph{Douady} and \emph{Yoccoz} returns no match,
so these papers supply categorical tools only.  The statements below are those of the
sources; the line numbers refer to the TeX files of the arXiv sources in \texttt{refs/}.

\begin{definition}[Strongly Mittag--Leffler sequence]
\label{def:strong-ml}
Let $\mathcal C_0\leftarrow\mathcal C_1\leftarrow\cdots$ be an inverse sequence of dualizable
categories over a rigid monoidal base $\mathcal E$ (in the corollary below, $\mathcal E=\mathrm{Sp}$),
with transition functors $F_{mn}\colon\mathcal C_m\to\mathcal C_n$ ($m\ge n$) and right adjoints
$F_{mn}^R$.  The sequence is \emph{strongly Mittag--Leffler} if
(i) for each $n$ the system $(F_{mn}F_{mn}^R)_{m\ge n}$ is essentially constant among the
$\mathcal E$-linear left adjoint endofunctors of $\mathcal C_n$, and
(ii) for all $n,k$ the limit functor $\Phi_{nk}=\varprojlim_{m\ge n,k}F_{mk}F_{mn}^R\colon\mathcal C_n\to\mathcal C_k$
is strongly continuous and has a left adjoint.
(\texttt{def:strong\_ML}, \texttt{arxiv-2502.04123.tex}, line 2681.)
\end{definition}

\begin{description}
\item[Sheaves on spaces~\cite{EfimovContinuousK}, lines 521--533.]
For a localizing invariant $F\colon\mathrm{Cat}^{\mathrm{perf}}\to\mathcal T$ into a presentable
stable category, commuting with filtered colimits, and a locally compact Hausdorff space $X$
with a presheaf $\underline{\mathcal C}$ of dualizable categories,
$F^{\mathrm{cont}}(\mathrm{Shv}(X;\underline{\mathcal C}))\cong\Gamma_c(X,F^{\mathrm{cont}}(\underline{\mathcal C}))$
(Theorem \texttt{th:main\_intro}).  For a finite CW complex $X$,
$K^{\mathrm{cont}}(\mathrm{Shv}(X;\mathcal C))\cong K^{\mathrm{cont}}(\mathcal C)^X$
(Corollary \texttt{cor:sheaves\_on\_finite\_CW\_complexes\_intro}).
\item[Localizing invariants of inverse limits~\cite{EfimovInverseLimits}, lines 2722--2725 and 3457--3482.]
For a strongly Mittag--Leffler sequence over $\mathrm{Sp}$,
$K^{\mathrm{cont}}(\varprojlim^{\mathrm{dual}}\mathcal C_n)\cong\varprojlim K^{\mathrm{cont}}(\mathcal C_n)$
(Corollary \texttt{cor:K\_theory\_of\_limit\_of\_ML}).  Theorem \texttt{th:hom\_epi\_for\_ML} shows that
such sequences satisfy the homological-epimorphism hypothesis.  Under that hypothesis,
Theorem \texttt{th:local\_invar\_of\_inverse\_limits} gives, for an accessible localizing
invariant $\Phi$, a cofiber sequence
$\Phi^{\mathrm{cont}}(\varprojlim^{\mathrm{dual}}\mathcal C_n)\to\Phi(\prod_n\mathcal C_n^{\omega})
\xrightarrow{\,1-\Phi(F)\,}\Phi(\prod_n\mathcal C_n^{\omega})$,
where $F$ is the product of the transition functors on compact objects.
\item[Dévissage and the theorem of the heart~\cite{EfimovDevissage}, lines 476--530.]
Quillen's dévissage theorem (Theorem \texttt{th:Quillen}, quoting Quillen~\cite{Quillen}, Theorem~4):
if a small abelian category $\mathcal A$ has a strictly full abelian subcategory $\mathcal B$,
closed under subobjects and quotients, such that every object of $\mathcal A$ has a finite
filtration with subquotients in $\mathcal B$, then $K_n(\mathcal B)\cong K_n(\mathcal A)$ for $n\ge0$.
Barwick's theorem of the heart (Theorem \texttt{th:Barwick}, \cite{Barwick}): for a stable
$\infty$-category $\mathcal T$ with a bounded $t$-structure, $K_n(\mathcal T^\heartsuit)\cong K_n(\mathcal T)$
for $n\ge0$.  Efimov's Theorem \texttt{th:key\_construction\_intro} gives a semi-orthogonal decomposition
$D^b(\mathcal E_{\mathcal A,\mathcal B})=\langle D^b(\mathcal B),D^b(\mathcal B)\rangle$ and a short exact sequence
$0\to\mathcal T_{\mathcal A,\mathcal B}\to D^b(\mathcal E_{\mathcal A,\mathcal B})\to D^b(\mathcal A)\to0$,
where $\mathcal E_{\mathcal A,\mathcal B}$ is the category of pairs $Y\subset X$ with $Y,\ X/Y\in\mathcal B$.
Quillen's theorem follows from these two statements and Barwick's theorem.
\item[Rigidity of localizing motives~\cite{EfimovRigidityMot}, introduction.]
The category $\mathrm{Mot}^{\mathrm{loc}}$ of localizing motives is rigid in the sense of Gaitsgory and
Rozenblyum, and hence dualizable (Theorem \texttt{th:rigidity\_over\_Sp\_intro}).  Consequently every
finitary localizing invariant $\mathrm{Cat}^{\mathrm{perf}}\to\mathrm{Sp}$ has the form $K(\mathcal C\otimes-)$
for some $\mathcal C$.  The relative version (Theorem \texttt{th:rigidity\_over\_E\_intro}) gives rigidity
of $\mathrm{Mot}^{\mathrm{loc}}_{\mathcal E}$ for a presentable stable rigid symmetric monoidal $\mathcal E$.
\item[Theorem of the heart for $KH$~\cite{EfimovHeartKH}, lines 530--551 and 2803--2804, 3017--3026.]
For a small stable category $\mathcal C$ with a bounded $t$-structure, the realization functor
$D^b(\mathcal C^\heartsuit)\to\mathcal C$ induces $KH(\mathcal C^\heartsuit)\cong KH(\mathcal C)$
(Theorem \texttt{th:theorem\_of\_the\_heart\_for\_KH\_intro}).  Under its hypotheses, dévissage for
coherently assembled abelian categories gives $K^{\mathrm{cont}}_j(\mathcal B)\cong K^{\mathrm{cont}}_j(\mathcal A)$
for $j\ge-1$, and $KH^{\mathrm{cont}}(\mathcal A)\simeq KH^{\mathrm{cont}}(\mathcal B)$
(Theorem \texttt{th:devissage\_for\_K\_and\_KH}).
\end{description}

\paragraph{Relation to the Lean interfaces.}
The \lean{Efimov.*} interfaces above are not instances of these theorems.
\lean{Efimov.StrongMittagLefflerData} records only stabilization clauses of
Definition~\ref{def:strong-ml}.  \lean{Efimov.DualizablePacmanTower} records rigid monoidal levels,
which is the notion in the rigidity theorem but not the rigidity of $\mathrm{Mot}^{\mathrm{loc}}$.
\lean{Efimov.PacmanKTheory} is a graded additive shadow, not continuous $K$-theory, and
\lean{Efimov.PacmanRealization} is a realization into the topological approximation category, which
none of the five papers constructs.  Mathlib does not contain the $\infty$-categorical framework
(presentable, dualizable, and stable categories) in which these theorems are stated.

\paragraph{Relation to the connectedness step.}
The five papers provide the categorical model side only: stable categories, $K$-theoretic invariants,
and their comparison under limits.  The realization of such a model in $\mathrm{TopCat}/\mathbb C$ and
the connectedness of its image require the Douady--Hubbard/Yoccoz phase--parameter input, which none
of the five papers provides.

\begin{conditionalstep}
The correct relation to the literature is:
\[
\begin{gathered}
\text{source-defined Douady--Hubbard/Yoccoz phase--parameter machinery}\\
\Downarrow\\
\text{genuine connected parameter pieces and a shrinking basis}\\
\Downarrow\\
\text{model-independent topological endpoint in Lean}\\
\Downarrow\\
\text{pointwise MLC; global MLC after covering every parameter}.
\end{gathered}
\]
The source construction uses moving graphs and marked
motions~\cite{DH,HubbardYoccoz,LyubichIII}.  A suitable continuous
surjection can express a connectedness step categorically when
one is actually available.  The historical ``Douady--Hubbard/Yoccoz''
name on a frozen carving interface does not identify it with that
classical construction.  In particular, the literature does not
supply the refuted universal frozen carving.
\end{conditionalstep}
