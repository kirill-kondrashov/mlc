\section{Green's Function}

\subsection*{Lemma green\_function\_functional\_eq}
\textbf{Description:} The Green's function satisfies the functional equation \texttt{G(f(z)) = 2 * G(z)}.

\textbf{Proof Description:} \texttt{G(f(z)) = lim\_\{n$\to$∞\} 1/2\^{}n log |f\^{}n(f(z))|}
    \texttt{= lim\_\{n$\to$∞\} 1/2\^{}n log |f\^{}\{n+1\}(z)|}
    \texttt{= lim\_\{n$\to$∞\} 2 * (1/2\^{}\{n+1\} log |f\^{}\{n+1\}(z)|)}
    \texttt{= 2 * G(z)}.

\begin{lstlisting}[language=Lean]

    green_function c (fc c z) = 2 * green_function c z := by
\end{lstlisting}

\subsection*{Lemma green\_function\_eq\_zero\_iff\_mem\_K}
\textbf{Description:} A point is in the filled Julia set iff its Green's function is zero.

\textbf{Proof Description:} *   \texttt{$\to$}: If \texttt{G(z) = 0}, suppose \texttt{z ∉ K(c)}. Then \texttt{z} escapes, and we can show \texttt{G(z) > 0}
        (using the positive growth rate outside \texttt{K(c)}). Contradiction.
    *   \texttt{←}: If \texttt{z $\in$ K(c)}, the orbit is bounded, so \texttt{potential\_seq} converges to 0.

\begin{lstlisting}[language=Lean]
lemma green_function_eq_zero_iff_mem_K (c z : ℂ) :
    green_function c z = 0 ↔ z ∈ K c := by
\end{lstlisting}
