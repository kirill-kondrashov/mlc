\title[Molecules]{A priori bounds for some\\ infinitely renormalizable quadratics:
  {\small III. Molecules.} }
\author {Jeremy Kahn and Mikhail Lyubich}
%\thanks{
%   This work was supported in part by Sloan Research Fellowship
% and NSF grants DMS-8920768 and DMS-9022140.}
\date{\today}



\begin{abstract} 
  In this paper we prove {\it a priori bounds} for infinitely renormalizable quadratic
polynomials satisfying a ``molecule condition''. Roughly speaking, this condition ensures that
the renormalization combinatorics stay away from the satellite types. These {\it a priori bounds}
imply local connectivity of the corresponding Julia sets and the Mandelbrot set at the corresponding parameter
values.   
\end{abstract}
